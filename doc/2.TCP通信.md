# 第2章: TCP通信の実装

Go言語バージョン: Go 1.22

## 2.1 TCPの基本概念と特性

TCPは「Transmission Control Protocol（伝送制御プロトコル）」の略で、インターネット上でデータを信頼性高く送受信するために使用される主要なプロトコルです。

### TCPの主要特性

1. **コネクション指向**
   - 通信前に3ウェイハンドシェイクによる接続確立が必要
   - 通信終了時に接続終了処理が必要
   - クライアントとサーバー間で双方向の通信チャネルを形成

2. **信頼性の高い通信**
   - パケットの到達保証（ACK機構）
   - パケットの順序保証（シーケンス番号による管理）
   - 重複パケットの除去
   - 破損データの検出と再送

3. **フロー制御とコンジェスション制御**
   - フロー制御: 受信側の処理能力に合わせた送信速度の調整
   - コンジェスション制御: ネットワーク混雑状況に応じた送信レートの調整

4. **全二重通信（Full-Duplex）**
   - 同時に双方向のデータ転送が可能

### Go言語におけるTCP通信

Go言語では`net`パッケージを使用してTCP通信を実装します。特に重要なコンポーネントは以下の通りです：

- `net.Dial()`: クライアント側でTCP接続を確立
- `net.Listen()`: サーバー側でTCPリスナーを作成
- `net.Conn`: 確立された接続を表すインターフェース
- `net.TCPConn`: TCP固有の機能を提供する構造体

## 2.2 TCPサーバーの実装

TCPサーバーの基本的な実装手順は以下の通りです：

1. ポート上でリッスンを開始
2. クライアント接続を受け付け
3. 各接続に対して処理を実行
4. 応答を返す

以下にシンプルかつ実用的なTCPサーバーの実装例を示します：

```go
package main

import (
	"bufio"
	"context"
	"errors"
	"fmt"
	"io"
	"log"
	"net"
	"os"
	"os/signal"
	"sync"
	"syscall"
	"time"
)

// Server は TCP サーバーを表す構造体
type Server struct {
	listener   net.Listener
	quit       chan struct{}
	wg         sync.WaitGroup
	connsMutex sync.Mutex
	conns      map[net.Conn]struct{}
}

// NewServer は新しい TCP サーバーを作成する
func NewServer(address string) (*Server, error) {
	listener, err := net.Listen("tcp", address)
	if err != nil {
		return nil, fmt.Errorf("リスナーの作成に失敗: %w", err)
	}

	server := &Server{
		listener: listener,
		quit:     make(chan struct{}),
		conns:    make(map[net.Conn]struct{}),
	}

	return server, nil
}

// Start はサーバーを開始し、クライアント接続を処理する
func (s *Server) Start() {
	s.wg.Add(1)
	go s.acceptLoop()

	log.Printf("サーバーがアドレス %s で開始しました\n", s.listener.Addr().String())
}

// acceptLoop は新しい接続を受け付けるループ
func (s *Server) acceptLoop() {
	defer s.wg.Done()

	for {
		conn, err := s.listener.Accept()
		if err != nil {
			select {
			case <-s.quit:
				return
			default:
				log.Printf("接続の受け入れに失敗: %v", err)
				continue
			}
		}

		s.connsMutex.Lock()
		s.conns[conn] = struct{}{}
		s.connsMutex.Unlock()

		s.wg.Add(1)
		go func() {
			s.handleConnection(conn)
			s.wg.Done()

			s.connsMutex.Lock()
			delete(s.conns, conn)
			s.connsMutex.Unlock()
		}()
	}
}

// handleConnection は個々のクライアント接続を処理する
func (s *Server) handleConnection(conn net.Conn) {
	defer conn.Close()
	
	// クライアントのアドレスをログに記録
	log.Printf("クライアント %s から接続を受け付けました\n", conn.RemoteAddr().String())

	// 10秒間のアイドルタイムアウトを設定
	if tcpConn, ok := conn.(*net.TCPConn); ok {
		if err := tcpConn.SetKeepAlive(true); err != nil {
			log.Printf("KeepAlive の設定に失敗: %v", err)
		}
		if err := tcpConn.SetKeepAlivePeriod(30 * time.Second); err != nil {
			log.Printf("KeepAlive 期間の設定に失敗: %v", err)
		}
	}

	reader := bufio.NewReader(conn)
	writer := bufio.NewWriter(conn)

	for {
		// タイムアウト設定
		if err := conn.SetDeadline(time.Now().Add(2 * time.Minute)); err != nil {
			log.Printf("デッドラインの設定に失敗: %v", err)
			return
		}

		// メッセージの読み取り
		message, err := reader.ReadString('\n')
		if err != nil {
			if errors.Is(err, io.EOF) {
				log.Printf("クライアント %s が接続を閉じました\n", conn.RemoteAddr().String())
			} else {
				log.Printf("メッセージの読み取りに失敗: %v", err)
			}
			return
		}

		// 受信データを表示して応答を返す
		log.Printf("クライアント %s からメッセージを受信: %s", conn.RemoteAddr().String(), message)
		
		response := fmt.Sprintf("サーバーが受信: %s", message)
		if _, err := writer.WriteString(response); err != nil {
			log.Printf("応答の書き込みに失敗: %v", err)
			return
		}
		if err := writer.Flush(); err != nil {
			log.Printf("バッファのフラッシュに失敗: %v", err)
			return
		}
	}
}

// Stop はサーバーを終了する
func (s *Server) Stop() error {
	// シャットダウン信号を送信
	close(s.quit)

	// リスナーを閉じる
	if err := s.listener.Close(); err != nil {
		return fmt.Errorf("リスナーを閉じる際にエラーが発生: %w", err)
	}

	// すべての接続を閉じる
	s.connsMutex.Lock()
	for conn := range s.conns {
		conn.Close()
	}
	s.connsMutex.Unlock()

	// すべてのゴルーチンが終了するのを待つ
	done := make(chan struct{})
	go func() {
		s.wg.Wait()
		close(done)
	}()

	// タイムアウト付きで待機
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()

	select {
	case <-done:
		return nil
	case <-ctx.Done():
		return errors.New("サーバー停止のタイムアウト")
	}
}

func main() {
	server, err := NewServer(":8080")
	if err != nil {
		log.Fatalf("サーバーの作成に失敗: %v", err)
	}

	// サーバーを開始
	server.Start()

	// シグナル処理
	sigChan := make(chan os.Signal, 1)
	signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
	<-sigChan

	log.Println("シャットダウン信号を受信しました。サーバーを停止します...")
	if err := server.Stop(); err != nil {
		log.Fatalf("サーバー停止エラー: %v", err)
	}
	log.Println("サーバーが正常に停止しました")
}
```

## 2.3 TCPクライアントの実装

TCPクライアントの基本的な実装手順は以下の通りです：

1. サーバーへの接続を確立
2. データを送信
3. サーバーからの応答を受信
4. 接続を閉じる

以下に実用的なTCPクライアントの実装例を示します：

```go
package main

import (
	"bufio"
	"context"
	"errors"
	"fmt"
	"io"
	"log"
	"net"
	"os"
	"strings"
	"time"
)

// Client は TCP クライアントを表す構造体
type Client struct {
	conn     net.Conn
	reader   *bufio.Reader
	writer   *bufio.Writer
	serverAddr string
}

// NewClient は新しい TCP クライアントを作成する
func NewClient(serverAddr string) (*Client, error) {
	return &Client{
		serverAddr: serverAddr,
	}, nil
}

// Connect はサーバーへの接続を確立する
func (c *Client) Connect() error {
	// コンテキストを使用したタイムアウト付き接続
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()

	// TCPアドレスの解決
	var d net.Dialer
	conn, err := d.DialContext(ctx, "tcp", c.serverAddr)
	if err != nil {
		return fmt.Errorf("サーバー %s への接続に失敗: %w", c.serverAddr, err)
	}

	c.conn = conn
	c.reader = bufio.NewReader(conn)
	c.writer = bufio.NewWriter(conn)

	// TCPコネクションの最適化
	if tcpConn, ok := conn.(*net.TCPConn); ok {
		// KeepAliveを有効化
		if err := tcpConn.SetKeepAlive(true); err != nil {
			return fmt.Errorf("KeepAlive の設定に失敗: %w", err)
		}
		
		// KeepAlive間隔を設定
		if err := tcpConn.SetKeepAlivePeriod(30 * time.Second); err != nil {
			return fmt.Errorf("KeepAlive 期間の設定に失敗: %w", err)
		}
		
		// Nagleアルゴリズムを無効化（必要に応じて）
		if err := tcpConn.SetNoDelay(true); err != nil {
			return fmt.Errorf("NoDelay の設定に失敗: %w", err)
		}
	}

	log.Printf("サーバー %s に接続しました\n", c.serverAddr)
	return nil
}

// SendMessage はサーバーにメッセージを送信し、応答を受信する
func (c *Client) SendMessage(message string) (string, error) {
	if c.conn == nil {
		return "", errors.New("サーバーに接続されていません")
	}

	// 5秒のタイムアウトを設定
	if err := c.conn.SetDeadline(time.Now().Add(5 * time.Second)); err != nil {
		return "", fmt.Errorf("デッドラインの設定に失敗: %w", err)
	}

	// メッセージが改行で終わることを確認
	if !strings.HasSuffix(message, "\n") {
		message += "\n"
	}

	// メッセージを送信
	if _, err := c.writer.WriteString(message); err != nil {
		return "", fmt.Errorf("メッセージの書き込みに失敗: %w", err)
	}
	if err := c.writer.Flush(); err != nil {
		return "", fmt.Errorf("バッファのフラッシュに失敗: %w", err)
	}

	log.Printf("サーバーにメッセージを送信: %s", message)

	// 応答を受信
	response, err := c.reader.ReadString('\n')
	if err != nil {
		if errors.Is(err, io.EOF) {
			return "", errors.New("サーバーが接続を閉じました")
		}
		return "", fmt.Errorf("応答の読み取りに失敗: %w", err)
	}

	return response, nil
}

// Close はクライアント接続を閉じる
func (c *Client) Close() error {
	if c.conn != nil {
		log.Println("クライアント接続を閉じています")
		return c.conn.Close()
	}
	return nil
}

func main() {
	client, err := NewClient("localhost:8080")
	if err != nil {
		log.Fatalf("クライアントの作成に失敗: %v", err)
	}

	// サーバーに接続
	if err := client.Connect(); err != nil {
		log.Fatalf("接続エラー: %v", err)
	}
	defer client.Close()

	// 標準入力からメッセージを読み取り
	scanner := bufio.NewScanner(os.Stdin)
	for {
		fmt.Print("サーバーに送信するメッセージを入力してください (exitで終了): ")
		if !scanner.Scan() {
			break
		}

		message := scanner.Text()
		if message == "exit" {
			break
		}

		// メッセージを送信して応答を受信
		response, err := client.SendMessage(message)
		if err != nil {
			log.Printf("エラー: %v", err)
			// 接続が失われた場合、再接続を試みる
			if errors.Is(err, io.EOF) || strings.Contains(err.Error(), "connection") {
				log.Println("サーバーへの再接続を試みます...")
				if err := client.Connect(); err != nil {
					log.Fatalf("再接続に失敗: %v", err)
				}
			}
			continue
		}

		fmt.Printf("サーバーからの応答: %s", response)
	}

	if err := scanner.Err(); err != nil {
		log.Printf("入力の読み取りエラー: %v", err)
	}
}
```

## 2.4 コネクション管理と並行処理

大規模なTCPサーバーでは、多数の接続を効率的に管理することが重要です。以下に、コネクションプールを実装した例を示します：

```go
package main

import (
	"context"
	"log"
	"net"
	"sync"
	"sync/atomic"
	"time"
)

// ConnectionPool は接続プールを表す構造体
type ConnectionPool struct {
	serverAddr     string
	maxConns       int
	minConns       int
	acquireTimeout time.Duration
	idleTimeout    time.Duration
	
	mu             sync.Mutex
	conns          []*pooledConnection
	connCount      int32
	
	// 接続プールの状態
	closed         bool
	wg             sync.WaitGroup
}

// pooledConnection はプール内の接続を表す構造体
type pooledConnection struct {
	conn      net.Conn
	createdAt time.Time
	lastUsed  time.Time
	inUse     bool
}

// NewConnectionPool は新しい接続プールを作成する
func NewConnectionPool(serverAddr string, minConns, maxConns int, acquireTimeout, idleTimeout time.Duration) (*ConnectionPool, error) {
	if minConns > maxConns {
		minConns = maxConns
	}
	
	pool := &ConnectionPool{
		serverAddr:     serverAddr,
		maxConns:       maxConns,
		minConns:       minConns,
		acquireTimeout: acquireTimeout,
		idleTimeout:    idleTimeout,
		conns:          make([]*pooledConnection, 0, maxConns),
	}
	
	// 初期接続を作成
	for i := 0; i < minConns; i++ {
		conn, err := pool.createConnection()
		if err != nil {
			pool.Close()
			return nil, err
		}
		
		pool.conns = append(pool.conns, &pooledConnection{
			conn:      conn,
			createdAt: time.Now(),
			lastUsed:  time.Now(),
			inUse:     false,
		})
	}
	
	// アイドル接続のクリーンアップを開始
	pool.wg.Add(1)
	go pool.cleanupIdleConnections()
	
	return pool, nil
}

// createConnection は新しい接続を作成する
func (p *ConnectionPool) createConnection() (net.Conn, error) {
	d := net.Dialer{Timeout: p.acquireTimeout}
	conn, err := d.Dial("tcp", p.serverAddr)
	if err != nil {
		return nil, err
	}
	
	// 接続数をインクリメント
	atomic.AddInt32(&p.connCount, 1)
	
	return conn, nil
}

// Acquire はプールから接続を取得する
func (p *ConnectionPool) Acquire(ctx context.Context) (net.Conn, error) {
	p.mu.Lock()
	
	if p.closed {
		p.mu.Unlock()
		return nil, net.ErrClosed
	}
	
	// 利用可能な接続を探す
	for i, pc := range p.conns {
		if !pc.inUse {
			pc.inUse = true
			pc.lastUsed = time.Now()
			conn := pc.conn
			p.mu.Unlock()
			return conn, nil
		}
	}
	
	// 接続数が最大に達していない場合は新しい接続を作成
	if int(p.connCount) < p.maxConns {
		conn, err := p.createConnection()
		if err != nil {
			p.mu.Unlock()
			return nil, err
		}
		
		pc := &pooledConnection{
			conn:      conn,
			createdAt: time.Now(),
			lastUsed:  time.Now(),
			inUse:     true,
		}
		
		p.conns = append(p.conns, pc)
		p.mu.Unlock()
		return conn, nil
	}
	
	// 全ての接続が使用中の場合はタイムアウトかキャンセルまで待機
	waitCh := make(chan struct{})
	
	go func() {
		for {
			p.mu.Lock()
			// もう一度利用可能な接続を探す
			for i, pc := range p.conns {
				if !pc.inUse {
					pc.inUse = true
					pc.lastUsed = time.Now()
					close(waitCh)
					p.mu.Unlock()
					return
				}
			}
			p.mu.Unlock()
			
			// 少し待ってから再試行
			time.Sleep(10 * time.Millisecond)
			
			// コンテキストがキャンセルまたはタイムアウトした場合
			select {
			case <-ctx.Done():
				close(waitCh)
				return
			default:
			}
		}
	}()
	
	// 接続が利用可能になるか、タイムアウトするまで待機
	p.mu.Unlock()
	
	select {
	case <-waitCh:
		// 利用可能な接続が見つかった
		for i, pc := range p.conns {
			if pc.inUse && pc.lastUsed.Equal(time.Now()) {
				return pc.conn, nil
			}
		}
		return nil, net.ErrClosed // 通常ここには到達しない
		
	case <-ctx.Done():
		return nil, ctx.Err()
	}
}

// Release は接続をプールに返却する
func (p *ConnectionPool) Release(conn net.Conn) {
	p.mu.Lock()
	defer p.mu.Unlock()
	
	if p.closed {
		conn.Close()
		return
	}
	
	// 該当する接続を探してフラグを更新
	for _, pc := range p.conns {
		if pc.conn == conn {
			pc.inUse = false
			pc.lastUsed = time.Now()
			return
		}
	}
	
	// 見つからない場合は閉じる
	conn.Close()
	atomic.AddInt32(&p.connCount, -1)
}

// cleanupIdleConnections はアイドル接続を定期的にクリーンアップする
func (p *ConnectionPool) cleanupIdleConnections() {
	defer p.wg.Done()
	
	ticker := time.NewTicker(30 * time.Second)
	defer ticker.Stop()
	
	for {
		select {
		case <-ticker.C:
			p.mu.Lock()
			
			if p.closed {
				p.mu.Unlock()
				return
			}
			
			now := time.Now()
			remainingConns := make([]*pooledConnection, 0)
			
			for _, pc := range p.conns {
				// 使用中の接続はそのまま残す
				if pc.inUse {
					remainingConns = append(remainingConns, pc)
					continue
				}
				
				// アイドルタイムアウトを超えた接続を閉じる（最小接続数は維持）
				if now.Sub(pc.lastUsed) > p.idleTimeout && len(remainingConns) >= p.minConns {
					pc.conn.Close()
					atomic.AddInt32(&p.connCount, -1)
					continue
				}
				
				// それ以外の接続は残す
				remainingConns = append(remainingConns, pc)
			}
			
			p.conns = remainingConns
			p.mu.Unlock()
			
		default:
			// プールが閉じられたかチェック
			p.mu.Lock()
			if p.closed {
				p.mu.Unlock()
				return
			}
			p.mu.Unlock()
			
			time.Sleep(100 * time.Millisecond)
		}
	}
}

// Close はプールを閉じ、全ての接続を解放する
func (p *ConnectionPool) Close() error {
	p.mu.Lock()
	
	if p.closed {
		p.mu.Unlock()
		return nil
	}
	
	p.closed = true
	
	// 全ての接続を閉じる
	for _, pc := range p.conns {
		pc.conn.Close()
	}
	
	p.conns = nil
	p.mu.Unlock()
	
	// クリーンアップゴルーチンの終了を待機
	p.wg.Wait()
	
	return nil
}

// Stats はプールの統計情報を返す
func (p *ConnectionPool) Stats() (total, inUse, idle int) {
	p.mu.Lock()
	defer p.mu.Unlock()
	
	total = len(p.conns)
	
	for _, pc := range p.conns {
		if pc.inUse {
			inUse++
		} else {
			idle++
		}
	}
	
	return total, inUse, idle
}

// 使用例
func main() {
	// 10〜30の接続を持つプールを作成、接続タイムアウト5秒、アイドルタイムアウト2分
	pool, err := NewConnectionPool("example.com:8080", 10, 30, 5*time.Second, 2*time.Minute)
	if err != nil {
		log.Fatalf("接続プールの作成に失敗: %v", err)
	}
	defer pool.Close()
	
	// コンテキストとタイムアウト
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()
	
	// 接続を取得
	conn, err := pool.Acquire(ctx)
	if err != nil {
		log.Fatalf("接続の取得に失敗: %v", err)
	}
	
	// 接続を使用 (例: データの送受信)
	// ...
	
	// 使用後に接続をプールに返却
	pool.Release(conn)
	
	// プールの統計情報を表示
	total, inUse, idle := pool.Stats()
	log.Printf("プール統計: 全体=%d, 使用中=%d, アイドル=%d", total, inUse, idle)
}
```

## 2.5 タイムアウト処理とキープアライブ

長時間の接続を管理する際は、タイムアウトとキープアライブの適切な設定が重要です：

```go
// タイムアウトとキープアライブの設定例
func configureConnection(conn *net.TCPConn) error {
	// 読み取りタイムアウト
	if err := conn.SetReadDeadline(time.Now().Add(30 * time.Second)); err != nil {
		return fmt.Errorf("読み取りタイムアウトの設定に失敗: %w", err)
	}
	
	// キープアライブの有効化
	if err := conn.SetKeepAlive(true); err != nil {
		return fmt.Errorf("キープアライブの設定に失敗: %w", err)
	}
	
	// キープアライブ間隔の設定
	if err := conn.SetKeepAlivePeriod(30 * time.Second); err != nil {
		return fmt.Errorf("キープアライブ間隔の設定に失敗: %w", err)
	}
	
	// Nagleアルゴリズムの無効化 (低遅延が必要な場合)
	if err := conn.SetNoDelay(true); err != nil {
		return fmt.Errorf("NoDelayの設定に失敗: %w", err)
	}
	
	return nil
}
```

## 2.6 エラー処理とリカバリー戦略

ネットワークプログラミングでは、接続断や様々なエラーに適切に対応することが重要です。以下にエラー処理とリカバリー戦略を示します：

```go
// 再接続ロジックの例
func connectWithRetry(addr string, maxRetries int, retryDelay time.Duration) (net.Conn, error) {
	var conn net.Conn
	var err error
	
	backoff := retryDelay
	
	for i := 0; i <= maxRetries; i++ {
		// 接続を試みる
		conn, err = net.DialTimeout("tcp", addr, 5*time.Second)
		if err == nil {
			return conn, nil
		}
		
		if i == maxRetries {
			break
		}
		
		log.Printf("接続に失敗（%d/%d）: %v - %s 後に再試行", i+1, maxRetries, err, backoff)
		time.Sleep(backoff)
		
		// 指数バックオフ (最大30秒)
		backoff *= 2
		if backoff > 30*time.Second {
			backoff = 30 * time.Second
		}
	}
	
	return nil, fmt.Errorf("%d回の試行後も接続に失敗: %w", maxRetries+1, err)
}

// エラー種別に応じた処理
func handleNetworkError(err error, conn net.Conn) {

		return
	}
	
	// タイムアウトエラー
	if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
		log.Printf("タイムアウトエラーが発生: %v", err)
		// タイムアウト固有の処理...
		return
	}
	
	// 接続が閉じられた
	if errors.Is(err, io.EOF) {
		log.Printf("リモート側が接続を閉じました")
		// 通常の接続終了処理...
		return
	}
	
	// ソケットの使用不可など、その他のエラー
	log.Printf("ネットワークエラー: %v", err)

	}
```

### メッセージプロトコル定義

まず、サーバーとクライアント間で交換されるメッセージのプロトコルを定義します：

```go
// message.go
package main

import (
    "encoding/binary"
    "encoding/json"
    "errors"
    "io"
)

// メッセージタイプの定義
const (
    TypePlainText = 1
    TypeJSON      = 2
    TypeBinary    = 3
    TypeHeartbeat = 4
)

// Message はプロトコルメッセージを表す
type Message struct {
    Type    uint8  // メッセージタイプ
    Length  uint32 // メッセージ長
    Payload []byte // メッセージ内容
}

// Encode はメッセージをワイヤフォーマットにエンコードする
func (m *Message) Encode(w io.Writer) error {
    // ヘッダー: タイプ(1バイト) + 長さ(4バイト)
    if err := binary.Write(w, binary.BigEndian, m.Type); err != nil {
        return err
    }
    
    if err := binary.Write(w, binary.BigEndian, m.Length); err != nil {
        return err
    }
    
    // ペイロード
    if m.Length > 0 {
        if _, err := w.Write(m.Payload); err != nil {
            return err
        }
    }
    
    return nil
}

// Decode はワイヤフォーマットからメッセージをデコードする
func DecodeMessage(r io.Reader) (*Message, error) {
    var msgType uint8
    if err := binary.Read(r, binary.BigEndian, &msgType); err != nil {
        return nil, err
    }
    
    var msgLength uint32
    if err := binary.Read(r, binary.BigEndian, &msgLength); err != nil {
        return nil, err
    }
    
    // サイズ制限チェック (例：10MB)
    if msgLength > 10*1024*1024 {
        return nil, errors.New("メッセージが大きすぎます")
    }
    
    var payload []byte
    if msgLength > 0 {
        payload = make([]byte, msgLength)
        if _, err := io.ReadFull(r, payload); err != nil {
            return nil, err
        }
    }
    
    return &Message{
        Type:    msgType,
        Length:  msgLength,
        Payload: payload,
    }, nil
}

// NewTextMessage はテキストメッセージを作成する
func NewTextMessage(text string) *Message {
    payload := []byte(text)
    return &Message{
        Type:    TypePlainText,
        Length:  uint32(len(payload)),
        Payload: payload,
    }
}

// NewJSONMessage はJSONメッセージを作成する
func NewJSONMessage(v interface{}) (*Message, error) {
    payload, err := json.Marshal(v)
    if err != nil {
        return nil, err
    }
    
    return &Message{
        Type:    TypeJSON,
        Length:  uint32(len(payload)),
        Payload: payload,
    }, nil
}

// NewBinaryMessage はバイナリメッセージを作成する
func NewBinaryMessage(data []byte) *Message {
    return &Message{
        Type:    TypeBinary,
        Length:  uint32(len(data)),
        Payload: data,
    }
}

// NewHeartbeatMessage はハートビートメッセージを作成する
func NewHeartbeatMessage() *Message {
    return &Message{
        Type:    TypeHeartbeat,
        Length:  0,
        Payload: nil,
    }
}

// GetTextPayload はテキストペイロードを取得する
func (m *Message) GetTextPayload() string {
    return string(m.Payload)
}

// GetJSONPayload はJSONペイロードを取得する
func (m *Message) GetJSONPayload(v interface{}) error {
    return json.Unmarshal(m.Payload, v)
}
```

### 改良されたTCPサーバー

```go
// server.go
package main

import (
    "context"
    "fmt"
    "io"
    "log"
    "net"
    "os"
    "os/signal"
    "sync"
    "syscall"
    "time"
)

// Client はクライアント接続を表す
type Client struct {
    conn     net.Conn
    id       string
    lastSeen time.Time
    send     chan *Message
}

// Server はTCPサーバーを表す
type Server struct {
    listener   net.Listener
    clients    map[string]*Client
    clientsMu  sync.RWMutex
    register   chan *Client
    unregister chan *Client
    broadcast  chan *Message
    quit       chan struct{}
    wg         sync.WaitGroup
}

// NewServer は新しいTCPサーバーを作成する
func NewServer(address string) (*Server, error) {
    listener, err := net.Listen("tcp", address)
    if err != nil {
        return nil, fmt.Errorf("リスナーの作成に失敗: %w", err)
    }
    
    server := &Server{
        listener:   listener,
        clients:    make(map[string]*Client),
        register:   make(chan *Client),
        unregister: make(chan *Client),
        broadcast:  make(chan *Message),
        quit:       make(chan struct{}),
    }
    
    return server, nil
}

// Start はサーバーを開始する
func (s *Server) Start() {
    // クライアント管理ルーチン
    s.wg.Add(1)
    go func() {
        defer s.wg.Done()
        s.run()
    }()
    
    // 接続受付ルーチン
    s.wg.Add(1)
    go func() {
        defer s.wg.Done()
        s.acceptConnections()
    }()
    
    log.Printf("サーバーを開始しました: %s\n", s.listener.Addr())
}

// acceptConnections は新しいクライアント接続を受け付ける
func (s *Server) acceptConnections() {
    for {
        conn, err := s.listener.Accept()
        if err != nil {
            select {
            case <-s.quit:
                return
            default:
                log.Printf("接続の受け入れエラー: %v", err)
                time.Sleep(100 * time.Millisecond)
                continue
            }
        }
        
        clientID := fmt.Sprintf("%s-%d", conn.RemoteAddr(), time.Now().UnixNano())
        
        client := &Client{
            conn:     conn,
            id:       clientID,
            lastSeen: time.Now(),
            send:     make(chan *Message, 100), // バッファ付きチャネル
        }
        
        // クライアント登録
        s.register <- client
        
        // クライアントのgoroutineを開始
        s.wg.Add(2)
        go s.readPump(client)
        go s.writePump(client)
    }
}

// run はクライアント管理ループを実行する
func (s *Server) run() {
    heartbeatTicker := time.NewTicker(30 * time.Second)
    defer heartbeatTicker.Stop()
    
    for {
        select {
        case <-s.quit:
            return
            
        case client := <-s.register:
            s.clientsMu.Lock()
            s.clients[client.id] = client
            s.clientsMu.Unlock()
            log.Printf("クライアント登録: %s (%s)", client.id, client.conn.RemoteAddr())
            
        case client := <-s.unregister:
            s.clientsMu.Lock()
            if _, ok := s.clients[client.id]; ok {
                close(client.send)
                delete(s.clients, client.id)
                client.conn.Close()
                log.Printf("クライアント削除: %s", client.id)
            }
            s.clientsMu.Unlock()
            
        case message := <-s.broadcast:
            s.clientsMu.RLock()
            for _, client := range s.clients {
                select {
                case client.send <- message:
                    // メッセージを送信キューに追加
                default:
                    // チャネルがいっぱいの場合、クライアントを切断
                    close(client.send)
                    s.unregister <- client
                }
            }
            s.clientsMu.RUnlock()
            
        case <-heartbeatTicker.C:
            // 全クライアントにハートビートを送信
            s.clientsMu.RLock()
            for _, client := range s.clients {
                select {
                case client.send <- NewHeartbeatMessage():
                    // ハートビート送信
                default:
                    // チャネルがいっぱいの場合はスキップ
                }
            }
            s.clientsMu.RUnlock()
            
            // 非アクティブクライアントの検出と削除
            s.cleanupInactiveClients()
        }
    }
}

// cleanupInactiveClients は一定時間アクティブでないクライアントを削除する
func (s *Server) cleanupInactiveClients() {
    deadline := time.Now().Add(-5 * time.Minute)
    
    s.clientsMu.Lock()
    defer s.clientsMu.Unlock()
    
    for id, client := range s.clients {
        if client.lastSeen.Before(deadline) {
            log.Printf("非アクティブクライアントを削除: %s (最終活動: %v)", id, client.lastSeen)
            close(client.send)
            delete(s.clients, id)
            client.conn.Close()
        }
    }
}

// readPump はクライアントからメッセージを読み取る
func (s *Server) readPump(client *Client) {
    defer func() {
        s.unregister <- client
        s.wg.Done()
    }()
    
    // タイムアウト設定
    if tcpConn, ok := client.conn.(*net.TCPConn); ok {
        tcpConn.SetKeepAlive(true)
        tcpConn.SetKeepAlivePeriod(60 * time.Second)
        tcpConn.SetReadBuffer(32 * 1024)
    }
    
    for {
        client.conn.SetReadDeadline(time.Now().Add(5 * time.Minute))
        
        // メッセージを読み取る
        msg, err := DecodeMessage(client.conn)
        if err != nil {
            if err != io.EOF {
                log.Printf("クライアント %s の読み取りエラー: %v", client.id, err)
            }
            return
        }
        
        // クライアントのlastSeenを更新
        client.lastSeen = time.Now()
        
        // ハートビートは処理せず、他のクライアントには転送しない
        if msg.Type == TypeHeartbeat {
            continue
        }
        
        // メッセージを処理
        log.Printf("クライアント %s からメッセージを受信: タイプ=%d, 長さ=%d", client.id, msg.Type, msg.Length)
        
        // ブロードキャスト
        s.broadcast <- msg
    }
}

// writePump はクライアントにメッセージを送信する
func (s *Server) writePump(client *Client) {
    defer s.wg.Done()
    
    // 送信バッファサイズの最適化
    if tcpConn, ok := client.conn.(*net.TCPConn); ok {
        tcpConn.SetWriteBuffer(32 * 1024)
        tcpConn.SetNoDelay(true) // Nagleアルゴリズムを無効化
    }
    
    for {
        select {
        case message, ok := <-client.send:
            if !ok {
                // チャネルが閉じられた
                return
            }
            
            // 書き込みタイムアウト設定
            client.conn.SetWriteDeadline(time.Now().Add(10 * time.Second))
            
            // メッセージを送信
            if err := message.Encode(client.conn); err != nil {
                log.Printf("クライアント %s への書き込みエラー: %v", client.id, err)
                return
            }
        }
    }
}

// Broadcast はメッセージをブロードキャストする
func (s *Server) Broadcast(message *Message) {
    s.broadcast <- message
}

// ClientCount は現在接続しているクライアント数を返す
func (s *Server) ClientCount() int {
    s.clientsMu.RLock()
    defer s.clientsMu.RUnlock()
    return len(s.clients)
}

// Stop はサーバーを停止する
func (s *Server) Stop(ctx context.Context) error {
    // シャットダウン信号を送信
    close(s.quit)
    
    // リスナーを閉じる
    if err := s.listener.Close(); err != nil {
        return fmt.Errorf("リスナーを閉じる際にエラーが発生: %w", err)
    }
    
    // 全クライアントを切断
    s.clientsMu.Lock()
    for _, client := range s.clients {
        close(client.send)
        client.conn.Close()
    }
    s.clients = make(map[string]*Client)
    s.clientsMu.Unlock()
    
    // ゴルーチンの終了を待つ
    done := make(chan struct{})
    go func() {
        s.wg.Wait()
        close(done)
    }()
    
    // タイムアウト付きで待機
    select {
    case <-done:
        return nil
    case <-ctx.Done():
        return ctx.Err()
    }
}

func main() {
    // サーバーを作成
    server, err := NewServer(":8080")
    if err != nil {
        log.Fatalf("サーバーの作成に失敗: %v", err)
    }
    
    // サーバーを開始
    server.Start()
    
    // シグナル処理
    sigChan := make(chan os.Signal, 1)
    signal.Notify(sigChan, os.Interrupt, syscall.SIGTERM)
    <-sigChan
    
    log.Println("シャットダウン中...")
    
    // 30秒のタイムアウトでサーバーを停止
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()
    
    if err := server.Stop(ctx); err != nil {
        log.Fatalf("サーバー停止エラー: %v", err)
    }
    
    log.Println("サーバーは正常に停止しました")
}
```

### TCPクライアント実装

```go
// client.go
package main

import (
    "context"
    "errors"
    "fmt"
    "io"
    "log"
    "net"
    "sync"
    "time"
)

// MessageHandler はメッセージハンドラ関数の型
type MessageHandler func(*Message)

// TCPClient はTCPクライアントを表す
type TCPClient struct {
    serverAddr  string
    conn        net.Conn
    sendMu      sync.Mutex  // 送信用ミューテックス
    handler     MessageHandler
    heartbeatInterval time.Duration
    
    reconnect   bool
    maxRetries  int
    retryDelay  time.Duration
    
    quit        chan struct{}
    wg          sync.WaitGroup
    
    connected   bool
    connMu      sync.RWMutex
}

// NewTCPClient は新しいTCPクライアントを作成する
func NewTCPClient(serverAddr string, handler MessageHandler) *TCPClient {
    return &TCPClient{
        serverAddr:        serverAddr,
        handler:           handler,
        heartbeatInterval: 30 * time.Second,
        reconnect:         true,
        maxRetries:        10,
        retryDelay:        time.Second,
        quit:              make(chan struct{}),
    }
}

// SetReconnect は再接続設定を行う
func (c *TCPClient) SetReconnect(reconnect bool, maxRetries int, retryDelay time.Duration) {
    c.reconnect = reconnect
    c.maxRetries = maxRetries
    c.retryDelay = retryDelay
}

// SetHeartbeatInterval はハートビート間隔を設定する
func (c *TCPClient) SetHeartbeatInterval(interval time.Duration) {
    c.heartbeatInterval = interval
}

// Connect はサーバーへの接続を確立する
func (c *TCPClient) Connect() error {
    c.connMu.Lock()
    defer c.connMu.Unlock()
    
    if c.connected {
        return nil
    }
    
    // 接続タイムアウト付きのDialer
    dialer := &net.Dialer{
        Timeout: 10 * time.Second,
    }
    
    var err error
    c.conn, err = dialer.Dial("tcp", c.serverAddr)
    if err != nil {
        return fmt.Errorf("サーバー %s への接続に失敗: %w", c.serverAddr, err)
    }
    
    // TCPコネクションの最適化
    if tcpConn, ok := c.conn.(*net.TCPConn); ok {
        if err := tcpConn.SetKeepAlive(true); err != nil {
            log.Printf("KeepAlive の設定に失敗: %v", err)
        }
        if err := tcpConn.SetKeepAlivePeriod(30 * time.Second); err != nil {
            log.Printf("KeepAlive 期間の設定に失敗: %v", err)
        }
        if err := tcpConn.SetNoDelay(true); err != nil {
            log.Printf("NoDelay の設定に失敗: %v", err)
        }
    }
    
    c.connected = true
    
    // 受信用goroutineを開始
    c.wg.Add(1)
    go c.readPump()
    
    // ハートビート用goroutineを開始
    if c.heartbeatInterval > 0 {
        c.wg.Add(1)
        go c.heartbeatPump()
    }
    
    return nil
}

// ConnectWithRetry は再試行付きでサーバーへの接続を確立する
func (c *TCPClient) ConnectWithRetry() error {
    var err error
    
    backoff := c.retryDelay
    
    for i := 0; i <= c.maxRetries; i++ {
        err = c.Connect()
        if err == nil {
            return nil
        }
        
        if i == c.maxRetries {
            break
        }
        
        log.Printf("接続に失敗（%d/%d）: %v - %s 後に再試行", i+1, c.maxRetries, err, backoff)
        time.Sleep(backoff)
        
        // 指数バックオフ (最大30秒)
        backoff *= 2
        if backoff > 30*time.Second {
            backoff = 30 * time.Second
        }
    }
    
    return fmt.Errorf("%d回の試行後も接続に失敗: %w", c.maxRetries+1, err)
}

// IsConnected は接続状態を返す
func (c *TCPClient) IsConnected() bool {
    c.connMu.RLock()
    defer c.connMu.RUnlock()
    return c.connected
}

// readPump はサーバーからメッセージを読み取る
func (c *TCPClient) readPump() {
    defer c.wg.Done()
    defer func() {
        c.connMu.Lock()
        c.connected = false
        c.connMu.Unlock()
        
        // 再接続が有効な場合、切断後に再接続を試みる
        if c.reconnect {
            select {
            case <-c.quit:
                // シャットダウン中は再接続しない
                return
            default:
                log.Println("接続が失われました。再接続を試みています...")
                go func() {
                    if err := c.ConnectWithRetry(); err != nil {
                        log.Printf("再接続に失敗しました: %v", err)
                    }
                }()
            }
        }
    }()
    
    for {
        select {
        case <-c.quit:
            return
        default:
            // 読み取りタイムアウトを設定
            if err := c.conn.SetReadDeadline(time.Now().Add(2 * time.Minute)); err != nil {
                log.Printf("読み取りタイムアウトの設定に失敗: %v", err)
            }
            
            // メッセージを読み取る
            msg, err := DecodeMessage(c.conn)
            if err != nil {
                if err != io.EOF {
                    log.Printf("メッセージの読み取りに失敗: %v", err)
                }
                return
            }
            
            // ハートビートメッセージは無視
            if msg.Type == TypeHeartbeat {
                continue
            }
            
            // メッセージを処理
            if c.handler != nil {
                c.handler(msg)
            }
        }
    }
}

// heartbeatPump は定期的にハートビートを送信する
func (c *TCPClient) heartbeatPump() {
    defer c.wg.Done()
    
    ticker := time.NewTicker(c.heartbeatInterval)
    defer ticker.Stop()
    
    for {
        select {
        case <-c.quit:
            return
        case <-ticker.C:
            if c.IsConnected() {
                if err := c.SendMessage(NewHeartbeatMessage()); err != nil {
                    log.Printf("ハートビート送信エラー: %v", err)
                }
            }
        }
    }
}

// SendMessage はサーバーにメッセージを送信する
func (c *TCPClient) SendMessage(msg *Message) error {
    if !c.IsConnected() {
        return errors.New("サーバーに接続されていません")
    }
    
    c.sendMu.Lock()
    defer c.sendMu.Unlock()
    
    // 書き込みタイムアウトを設定
    if err := c.conn.SetWriteDeadline(time.Now().Add(10 * time.Second)); err != nil {
        return fmt.Errorf("書き込みタイムアウトの設定に失敗: %w", err)
    }
    
    // メッセージを送信
    if err := msg.Encode(c.conn); err != nil {
        // 送信エラーの場合、接続状態をリセット
        c.connMu.Lock()
        c.connected = false
        c.connMu.Unlock()
        return fmt.Errorf("メッセージの送信に失敗: %w", err)
    }
    
    return nil
}

// SendText はテキストメッセージを送信する
func (c *TCPClient) SendText(text string) error {
    return c.SendMessage(NewTextMessage(text))
}

// SendJSON はJSONメッセージを送信する
func (c *TCPClient) SendJSON(v interface{}) error {
    msg, err := NewJSONMessage(v)
    if err != nil {
        return err
    }
    return c.SendMessage(msg)
}

// SendBinary はバイナリメッセージを送信する
func (c *TCPClient) SendBinary(data []byte) error {
    return c.SendMessage(NewBinaryMessage(data))
}

// Close はクライアント接続を閉じる
func (c *TCPClient) Close() error {
    c.connMu.Lock()
    
    if !c.connected {
        c.connMu.Unlock()
        return nil
    }
    
    // シャットダウン信号を送信
    close(c.quit)
    
    // 接続を閉じる
    if c.conn != nil {
        if err := c.conn.Close(); err != nil {
            c.connMu.Unlock()
            return fmt.Errorf("接続を閉じる際にエラーが発生: %w", err)
        }
    }
    
    c.connected = false
    c.connMu.Unlock()
    
    // すべてのゴルーチンが終了するのを待つ
    done := make(chan struct{})
    go func() {
        c.wg.Wait()
        close(done)
    }()
    
    // タイムアウト付きで待機
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()
    
    select {
    case <-done:
        return nil
    case <-ctx.Done():
        return errors.New("クライアント停止のタイムアウト")
    }
}

// クライアント使用例
func main() {
    // メッセージハンドラを定義
```

```go
// file_protocol.go
package main

import (
    "encoding/binary"
    "errors"
    "fmt"
    "io"
    "path/filepath"
)

// コマンド定義
const (
    CmdRequestFileList = 1
    CmdFileList        = 2
    CmdRequestFile     = 3
    CmdFileData        = 4
    CmdFileComplete    = 5
    CmdError           = 6
)

// FileHeader はファイル転送プロトコルのヘッダー
type FileHeader struct {
    Command     uint8  // コマンド
    PayloadSize uint32 // ペイロードサイズ
}

// FileInfo はファイル情報
type FileInfo struct {
    Name  string
    Size  int64
    IsDir bool
}

// EncodeFileList はファイルリストをエンコードする
func EncodeFileList(w io.Writer, files []FileInfo) error {
    // ファイル数を書き込む
    if err := binary.Write(w, binary.BigEndian, uint32(len(files))); err != nil {
        return err
    }
    
    // 各ファイル情報を書き込む
    for _, file := range files {
        // ファイル名のサイズ
        nameBytes := []byte(file.Name)
        if err := binary.Write(w, binary.BigEndian, uint16(len(nameBytes))); err != nil {
            return err
        }
        
        // ファイル名
        if _, err := w.Write(nameBytes); err != nil {
            return err
        }
        
        // ファイルサイズ
        if err := binary.Write(w, binary.BigEndian, file.Size); err != nil {
            return err
        }
        
        // ディレクトリフラグ
        isDir := uint8(0)
        if file.IsDir {
            isDir = 1
        }
        if err := binary.Write(w, binary.BigEndian, isDir); err != nil {
            return err
        }
    }
    
    return nil
}

// DecodeFileList はファイルリストをデコードする
func DecodeFileList(r io.Reader) ([]FileInfo, error) {
    // ファイル数を読み取る
    var count uint32
    if err := binary.Read(r, binary.BigEndian, &count); err != nil {
        return nil, err
    }
    
    // ファイル数の妥当性チェック
    if count > 10000 {
        return nil, errors.New("ファイル数が多すぎます")
    }
    
    files := make([]FileInfo, 0, count)
    
    // 各ファイル情報を読み取る
    for i := uint32(0); i < count; i++ {
        // ファイル名のサイズ
        var nameLen uint16
        if err := binary.Read(r, binary.BigEndian, &nameLen); err != nil {
            return nil, err
        }
        
        // ファイル名のサイズの妥当性チェック
        if nameLen > 1024 {
            return nil, errors.New("ファイル名が長すぎます")
        }
        
        // ファイル名
        nameBytes := make([]byte, nameLen)
        if _, err := io.ReadFull(r, nameBytes); err != nil {
            return nil, err
        }
        
        // ファイルサイズ
        var size int64
        if err := binary.Read(r, binary.BigEndian, &size); err != nil {
            return nil, err
        }
        
        // ディレクトリフラグ
        var isDir uint8
        if err := binary.Read(r, binary.BigEndian, &isDir); err != nil {
            return nil, err
        }
        
        files = append(files, FileInfo{
            Name:  string(nameBytes),
            Size:  size,
            IsDir: isDir == 1,
        })
    }
    
    return files, nil
}

// 安全なファイルパスかどうかチェック
func IsValidPath(basePath, filePath string) bool {
    // パスのクリーニング
    cleanBase := filepath.Clean(basePath)
    requestedPath := filepath.Clean(filepath.Join(cleanBase, filePath))
    
    // パスがベースパス内に収まっているかチェック
    return len(requestedPath) >= len(cleanBase) && requestedPath[:len(cleanBase)] == cleanBase
}
```

### ファイル転送サーバー

```go
// file_server.go
package main

import (
    "context"
    "encoding/binary"
    "errors"
    "fmt"
    "io"
    "log"
    "net"
    "os"
    "path/filepath"
    "sync"
    "time"
)

// FileServer はファイル転送サーバーを表す
type FileServer struct {
    listener    net.Listener
    basePath    string
    chunkSize   int
    maxFileSize int64
    quit        chan struct{}
    wg          sync.WaitGroup
}

// NewFileServer は新しいファイル転送サーバーを作成する
func NewFileServer(address, basePath string, chunkSize int, maxFileSize int64) (*FileServer, error) {
    // ベースディレクトリの存在確認
    info, err := os.Stat(basePath)
    if err != nil {
        return nil, fmt.Errorf("ベースパスが無効です: %w", err)
    }
    
    if !info.IsDir() {
        return nil, errors.New("ベースパスはディレクトリでなければなりません")
    }
    
    // リスナーの作成
    listener, err := net.Listen("tcp", address)
    if err != nil {
        return nil, fmt.Errorf("リスナーの作成に失敗: %w", err)
    }
    
    server := &FileServer{
        listener:    listener,
        basePath:    basePath,
        chunkSize:   chunkSize,
        maxFileSize: maxFileSize,
        quit:        make(chan struct{}),
    }
    
    return server, nil
}

// Start はサーバーを開始する
func (s *FileServer) Start() {
    s.wg.Add(1)
    go s.acceptConnections()
    
    log.Printf("ファイル転送サーバーを開始しました: %s, ベースパス: %s", s.listener.Addr(), s.basePath)
}

// acceptConnections は新しいクライアント接続を受け付ける
func (s *FileServer) acceptConnections() {
    defer s.wg.Done()
    
    for {
        conn, err := s.listener.Accept()
        if err != nil {
            select {
            case <-s.quit:
                return
            default:
                log.Printf("接続の受け入れエラー: %v", err)
                time.Sleep(100 * time.Millisecond)
                continue
            }
        }
        
        // 各クライアント接続に対してgoroutineを起動
        s.wg.Add(1)
        go func() {
            defer s.wg.Done()
            defer conn.Close()
            
            log.Printf("クライアント接続: %s", conn.RemoteAddr())
            
            if tcpConn, ok := conn.(*net.TCPConn); ok {
                tcpConn.SetKeepAlive(true)
                tcpConn.SetKeepAlivePeriod(60 * time.Second)
                tcpConn.SetNoDelay(true)
            }
            
            s.handleClient(conn)
            
            log.Printf("クライアント切断: %s", conn.RemoteAddr())
        }()
    }
}

// handleClient はクライアント接続を処理する
func (s *FileServer) handleClient(conn net.Conn) {
    for {
        // タイムアウト設定
        if err := conn.SetReadDeadline(time.Now().Add(2 * time.Minute)); err != nil {
            log.Printf("読み取りタイムアウト設定エラー: %v", err)
            return
        }
        
        // ヘッダーを読み取る
        var header FileHeader
        if err := binary.Read(conn, binary.BigEndian, &header.Command); err != nil {
            if err != io.EOF {
                log.Printf("コマンド読み取りエラー: %v", err)
            }
            return
        }
        
        if err := binary.Read(conn, binary.BigEndian, &header.PayloadSize); err != nil {
            log.Printf("ペイロードサイズ読み取りエラー: %v", err)
            return
        }
        
        // ペイロードサイズの妥当性チェック (最大100MB)
        if header.PayloadSize > 100*1024*1024 {
            s.sendError(conn, "ペイロードが大きすぎます")
            continue
        }
        
        // コマンドに応じた処理
        switch header.Command {
        case CmdRequestFileList:
            s.handleFileListRequest(conn)
            
        case CmdRequestFile:
            // ファイル名の読み取り
            if header.PayloadSize > 1024 {
                s.sendError(conn, "ファイル名が長すぎます")
                continue
            }
            
            filenameBytes := make([]byte, header.PayloadSize)
            if _, err := io.ReadFull(conn, filenameBytes); err != nil {
                log.Printf("ファイル名読み取りエラー: %v", err)
                return
            }
            
            filename := string(filenameBytes)
            s.handleFileRequest(conn, filename)
            
        default:
            log.Printf("不明なコマンド: %d", header.Command)
            s.sendError(conn, "不明なコマンド")
        }
    }
}

// handleFileListRequest はファイルリストリクエストを処理する
func (s *FileServer) handleFileListRequest(conn net.Conn) {
    // ベースディレクトリのファイル一覧を取得
    var files []FileInfo
    
    err := filepath.Walk(s.basePath, func(path string, info os.FileInfo, err error) error {
        if err != nil {
            return err
        }
        
        // ベースパスからの相対パスを取得
        relPath, err := filepath.Rel(s.basePath, path)
        if err != nil {
            return err
        }
        
        // ルートディレクトリは除外
        if relPath == "." {
            return nil
        }
        
        files = append(files, FileInfo{
            Name:  relPath,
            Size:  info.Size(),
            IsDir: info.IsDir(),
        })
        
        return nil
    })
    
    if err != nil {
        log.Printf("ファイル一覧の取得エラー: %v", err)
        s.sendError(conn, "ファイル一覧の取得に失敗しました")
        return
    }
    
    // レスポンスデータを作成
    buffer := make([]byte, 0, 1024*1024) // 1MB初期バッファ
    w := NewBufferWriter(buffer)
    
    if err := EncodeFileList(w, files); err != nil {
        log.Printf("ファイルリストのエンコードエラー: %v", err)
        s.sendError(conn, "ファイルリストのエンコードに失敗しました")
        return
    }
    
    responseData := w.Bytes()
    
    // レスポンスヘッダーを送信
    if err := conn.SetWriteDeadline(time.Now().Add(30 * time.Second)); err != nil {
        log.Printf("書き込みタイムアウト設定エラー: %v", err)
        return
    }
    
    if err := binary.Write(conn, binary.BigEndian, uint8(CmdFileList)); err != nil {
        log.Printf("コマンド書き込みエラー: %v", err)
        return
    }
    
    if err := binary.Write(conn, binary.BigEndian, uint32(len(responseData))); err != nil {
        log.Printf("ペイロードサイズ書き込みエラー: %v", err)
        return
    }
    
    // レスポンスデータを送信
    if _, err := conn.Write(responseData); err != nil {
        log.Printf("ファイルリスト送信エラー: %v", err)
        return
    }
    
    log.Printf("ファイルリストを送信しました: %d ファイル", len(files))
}

// handleFileRequest はファイルリクエストを処理する
func (s *FileServer) handleFileRequest(conn net.Conn, filename string) {
    // パスが安全かチェック
    if !IsValidPath(s.basePath, filename) {
        log.Printf("不正なファイルパス: %s", filename)
        s.sendError(conn, "不正なファイルパス")
        return
    }
    
    filePath := filepath.Join(s.basePath, filename)
    
    // ファイル情報を取得
    info, err := os.Stat(filePath)
    if err != nil {
        log.Printf("ファイル情報取得エラー: %v", err)
        s.sendError(conn, "ファイルが見つかりません")
        return
    }
    
    // ディレクトリの場合はエラー
    if info.IsDir() {
        s.sendError(conn, "指定されたパスはディレクトリです")
        return
    }
    
    // ファイルサイズチェック
    if info.Size() > s.maxFileSize {
        s.sendError(conn, fmt.Sprintf("ファイルが大きすぎます (最大: %d バイト)", s.maxFileSize))
        return
    }
    
    // ファイルを開く
    file, err := os.Open(filePath)
    if err != nil {
        log.Printf("ファイルを開くエラー: %v", err)
        s.sendError(conn, "ファイルを開けません")
        return
    }
    defer file.Close()
    
    // ファイルデータを送信
    buffer := make([]byte, s.chunkSize)
    totalSent := int64(0)
    
    for {
        n, err := file.Read(buffer)
        if err != nil && err != io.EOF {
            log.Printf("ファイル読み取りエラー: %v", err)
            s.sendError(conn, "ファイル読み取りエラー")
            return
        }
        
        if n > 0 {
            // データチャンクを送信
            if err := conn.SetWriteDeadline(time.Now().Add(30 * time.Second)); err != nil {
                log.Printf("書き込みタイムアウト設定エラー: %v", err)
                return
            }
            
            if err := binary.Write(conn, binary.BigEndian, uint8(CmdFileData)); err != nil {
                log.Printf("コマンド書き込みエラー: %v", err)
                return
            }
            
            if err := binary.Write(conn, binary.BigEndian, uint32(n)); err != nil {
                log.Printf("データサイズ書き込みエラー: %v", err)
                return
            }
            
            if _, err := conn.Write(buffer[:n]); err != nil {
                log.Printf("データ送信エラー: %v", err)
                return
            }
            
            totalSent += int64(n)
            
            // 進捗ログ (大きなファイルの場合)
            if info.Size() > 1024*1024 && totalSent%(1024*1024) < int64(n) {
                log.Printf("ファイル送信中: %s - %.1f%% (%d/%d バイト)",
                    filename, float64(totalSent)/float64(info.Size())*100, totalSent, info.Size())
            }
        }
        
        if err == io.EOF {
            break
        }
    }
    
    // 転送完了メッセージを送信
    if err := conn.SetWriteDeadline(time.Now().Add(30 * time.Second)); err != nil {
        log.Printf("書き込みタイムアウト設定エラー: %v", err)
        return
    }
    
    if err := binary.Write(conn, binary.BigEndian, uint8(CmdFileComplete)); err != nil {
        log.Printf("完了コマンド書き込みエラー: %v", err)
        return
    }
    
    if err := binary.Write(conn, binary.BigEndian, uint32(0)); err != nil {
        log.Printf("ペイロードサイズ書き込みエラー: %v", err)
        return
    }
    
    log.Printf("ファイル転送完了: %s (%d バイト)", filename, totalSent)
}

// sendError はエラーメッセージを送信する
func (s *FileServer) sendError(conn net.Conn, errMsg string) {
    if err := conn.SetWriteDeadline(time.Now().Add(10 * time.Second)); err != nil {
        log.Printf("書き込みタイムアウト設定エラー: %v", err)
        return
    }
    
    errBytes := []byte(errMsg)
    
    if err := binary.Write(conn, binary.BigEndian, uint8(CmdError)); err != nil {
        log.Printf("エラーコマンド書き込みエラー: %v", err)
        return
    }
    
    if err := binary.Write(conn, binary.BigEndian, uint32(len(errBytes))); err != nil {
        log.Printf("エラーサイズ書き込みエラー: %v", err)
        return
    }
    
    if _, err := conn.Write(errBytes); err != nil {
        log.Printf("エラーメッセージ送信エラー: %v", err)
        return
    }
    
    log.Printf("エラーを送信: %s", errMsg)
}

// Stop はサーバーを停止する
func (s *FileServer) Stop(ctx context.Context) error {
    // シャットダウン信号を送信
    close(s.quit)
    
    // リスナーを閉じる
    if err := s.listener.Close(); err != nil {
        return fmt.Errorf("リスナーを閉じる際にエラーが発生: %w", err)
    }
    
    // ゴルーチンの終了を待つ
    done := make(chan struct{})
    go func() {
        s.wg.Wait()
        close(done)
    }()
    
    // タイムアウト付きで待機
    select {
    case <-done:
        return nil
    case <-ctx.Done():
        return ctx.Err()
    }
}

type BufferWriter struct {
    buffer []byte
    pos    int
}

// NewBufferWriter は新しいバッファライターを作成する
func NewBufferWriter(buffer []byte) *BufferWriter {
    return &BufferWriter{
        buffer: buffer,
        pos:    0,
    }
}

// Write はio.Writer実装
func (w *BufferWriter) Write(p []byte) (n int, err error) {
    // バッファの拡張が必要かチェック
    neededSize := w.pos + len(p)
    if neededSize > cap(w.buffer) {
        // 新しいバッファサイズ（2倍に拡張）
        newSize := cap(w.buffer) * 2
        if newSize < neededSize {
            newSize = neededSize
        }
        
        // 新しいバッファを作成して内容をコピー
        newBuffer := make([]byte, newSize)
        copy(newBuffer, w.buffer[:w.pos])
        w.buffer = newBuffer
    }
    
    // データをバッファにコピー
    if w.pos+len(p) > len(w.buffer) {
        w.buffer = w.buffer[:w.pos+len(p)]
    }
    copy(w.buffer[w.pos:], p)
    w.pos += len(p)
    
    return len(p), nil
}

// Bytes は現在のバッファ内容を返す
func (w *BufferWriter) Bytes() []byte {
    return w.buffer[:w.pos]
}
```

```go
// GetFileList はサーバーからファイルリストを取得する
func (c *FileClient) GetFileList() ([]FileInfo, error) {
    if c.conn == nil {
        return nil, errors.New("サーバーに接続されていません")
    }
    
    // リクエストを送信
    if err := c.conn.SetWriteDeadline(time.Now().Add(c.writeTimeout)); err != nil {
        return nil, fmt.Errorf("書き込みタイムアウト設定エラー: %w", err)
    }
    
    if err := binary.Write(c.conn, binary.BigEndian, uint8(CmdRequestFileList)); err != nil {
        return nil, fmt.Errorf("コマンド送信エラー: %w", err)
    }
    
    if err := binary.Write(c.conn, binary.BigEndian, uint32(0)); err != nil {
        return nil, fmt.Errorf("ペイロードサイズ送信エラー: %w", err)
    }
    
    // レスポンスヘッダーを受信
    if err := c.conn.SetReadDeadline(time.Now().Add(c.readTimeout)); err != nil {
        return nil, fmt.Errorf("読み取りタイムアウト設定エラー: %w", err)
    }
    
    var cmdType uint8
    if err := binary.Read(c.conn, binary.BigEndian, &cmdType); err != nil {
        return nil, fmt.Errorf("レスポンスタイプ読み取りエラー: %w", err)
    }
    
    var payloadSize uint32
    if err := binary.Read(c.conn, binary.BigEndian, &payloadSize); err != nil {
        return nil, fmt.Errorf("ペイロードサイズ読み取りエラー: %w", err)
    }
    
    // ペイロードサイズの妥当性チェック (最大100MB)
    if payloadSize > 100*1024*1024 {
        return nil, errors.New("ペイロードが大きすぎます")
    }
    
    // レスポンスタイプをチェック
    switch cmdType {
    case CmdFileList:
        // ファイルリストを読み取る
        payload := make([]byte, payloadSize)
        if _, err := io.ReadFull(c.conn, payload); err != nil {
            return nil, fmt.Errorf("ファイルリスト読み取りエラー: %w", err)
        }
        
        files, err := DecodeFileList(NewBufferReader(payload))
        if err != nil {
            return nil, fmt.Errorf("ファイルリストのデコードエラー: %w", err)
        }
        
        return files, nil
        
    case CmdError:
        // エラーメッセージを読み取る
        errMsg := make([]byte, payloadSize)
        if _, err := io.ReadFull(c.conn, errMsg); err != nil {
            return nil, fmt.Errorf("エラーメッセージ読み取りエラー: %w", err)
        }
        
        return nil, fmt.Errorf("サーバーエラー: %s", string(errMsg))
        
    default:
        return nil, fmt.Errorf("不明なレスポンスタイプ: %d", cmdType)
    }
}

// DownloadFile はサーバーからファイルをダウンロードする
func (c *FileClient) DownloadFile(filename string, progressCB func(int64, int64)) error {
    if c.conn == nil {
        return errors.New("サーバーに接続されていません")
    }
    
    log.Printf("ファイルのダウンロードを開始: %s", filename)
    
    // リクエストを送信
    if err := c.conn.SetWriteDeadline(time.Now().Add(c.writeTimeout)); err != nil {
        return fmt.Errorf("書き込みタイムアウト設定エラー: %w", err)
    }
    
    filenameBytes := []byte(filename)
    
    if err := binary.Write(c.conn, binary.BigEndian, uint8(CmdRequestFile)); err != nil {
        return fmt.Errorf("コマンド送信エラー: %w", err)
    }
    
    if err := binary.Write(c.conn, binary.BigEndian, uint32(len(filenameBytes))); err != nil {
        return fmt.Errorf("ファイル名サイズ送信エラー: %w", err)
    }
    
    if _, err := c.conn.Write(filenameBytes); err != nil {
        return fmt.Errorf("ファイル名送信エラー: %w", err)
    }
    
    // ダウンロード先ファイルを準備
    downloadPath := filepath.Join(c.downloadDir, filepath.Base(filename))
    file, err := os.Create(downloadPath)
    if err != nil {
        return fmt.Errorf("ファイル作成エラー: %w", err)
    }
    defer file.Close()
    
    // ファイルデータを受信
    totalReceived := int64(0)
    startTime := time.Now()
    
    for {
        // ヘッダーを読み取る
        if err := c.conn.SetReadDeadline(time.Now().Add(c.readTimeout)); err != nil {
            return fmt.Errorf("読み取りタイムアウト設定エラー: %w", err)
        }
        
        var cmdType uint8
        if err := binary.Read(c.conn, binary.BigEndian, &cmdType); err != nil {
            return fmt.Errorf("レスポンスタイプ読み取りエラー: %w", err)
        }
        
        var payloadSize uint32
        if err := binary.Read(c.conn, binary.BigEndian, &payloadSize); err != nil {
            return fmt.Errorf("ペイロードサイズ読み取りエラー: %w", err)
        }
        
        switch cmdType {
        case CmdFileData:
            // データチャンクを読み取る
            if payloadSize > 0 {
                buffer := make([]byte, payloadSize)
                if _, err := io.ReadFull(c.conn, buffer); err != nil {
                    return fmt.Errorf("データチャンク読み取りエラー: %w", err)
                }
                
                // ファイルに書き込む
                if _, err := file.Write(buffer); err != nil {
                    return fmt.Errorf("ファイル書き込みエラー: %w", err)
                }
                
                totalReceived += int64(payloadSize)
                
                // 進捗コールバック
                if progressCB != nil {
                    progressCB(totalReceived, -1) // 合計サイズは不明
                }
            }
            
        case CmdFileComplete:
            // 転送完了
            elapsedTime := time.Since(startTime)
            speedMBps := float64(totalReceived) / 1024 / 1024 / elapsedTime.Seconds()
            
            log.Printf("ファイルダウンロード完了: %s (%.2f MB, %.2f MB/s)",
                filename, float64(totalReceived)/1024/1024, speedMBps)
            
            return nil
            
        case CmdError:
            // エラーメッセージを読み取る
            errMsg := make([]byte, payloadSize)
            if _, err := io.ReadFull(c.conn, errMsg); err != nil {
                return fmt.Errorf("エラーメッセージ読み取りエラー: %w", err)
            }
            
            // 不完全なファイルを削除
            file.Close()
            os.Remove(downloadPath)
            
            return fmt.Errorf("サーバーエラー: %s", string(errMsg))
            
        default:
            return fmt.Errorf("不明なレスポンスタイプ: %d", cmdType)
        }
    }
}

    r.pos += n
```

## 2.9 TCP通信における主な課題と対策

TCPプログラミングにおける一般的な課題と、その対策について説明します。

### 1. ヘッドオブラインブロッキング（Head-of-Line Blocking）

**問題**: TCPはストリーム指向のプロトコルであり、データパケットの順序を保証するために、あるパケットが失われると、それ以降のパケットはそのパケットが再送されるまで配信されません。

**対策**:
- 複数のTCP接続を使用して並列化
- 重要なデータとそうでないデータを別々の接続で送信
- プロトコル設計時に、独立した処理が可能なメッセージ形式を採用

```go
// 複数接続を管理する例
type MultiConnectionClient struct {
    connections []*net.TCPConn
    nextConn    int
    mu          sync.Mutex
}

func (c *MultiConnectionClient) Send(data []byte) error {
    c.mu.Lock()
    conn := c.connections[c.nextConn]
    c.nextConn = (c.nextConn + 1) % len(c.connections)
    c.mu.Unlock()
    
    _, err := conn.Write(data)
    return err
}
```

### 4. バックプレッシャー処理

**問題**: 送信側が受信側の処理能力を超える速度でデータを送信すると、メモリ使用量の増加やパフォーマンス低下が発生する。

**対策**:

- フロー制御の実装
- カスタムプロトコルにフロー制御メカニズムを組み込む

```go
// バックプレッシャー処理の例
type FlowControlledSender struct {
    conn           net.Conn
    sendQueue      chan []byte
    ackCh          chan struct{}
    maxOutstanding int
    outstanding    int
    mu             sync.Mutex
}

func NewFlowControlledSender(conn net.Conn, maxOutstanding int) *FlowControlledSender {
    s := &FlowControlledSender{
        conn:           conn,
        sendQueue:      make(chan []byte, 100),
        ackCh:          make(chan struct{}, maxOutstanding),
        maxOutstanding: maxOutstanding,
    }
    
    go s.sendLoop()
    go s.receiveAcks()
    
    return s
}

func (s *FlowControlledSender) Send(data []byte) error {
    select {
    case s.sendQueue <- data:
        return nil
    case <-time.After(5 * time.Second):
        return errors.New("送信キューがいっぱいです")
    }
}

func (s *FlowControlledSender) sendLoop() {
    for data := range s.sendQueue {
        // 未確認メッセージの数が上限に達したら待機
        s.mu.Lock()
        for s.outstanding >= s.maxOutstanding {
            s.mu.Unlock()
            <-s.ackCh
            s.mu.Lock()
            s.outstanding--
        }
        
        s.outstanding++
        s.mu.Unlock()
        
        // データを送信
        if err := writeMessage(s.conn, data); err != nil {
            log.Printf("送信エラー: %v", err)
            return
        }
    }
}

func (s *FlowControlledSender) receiveAcks() {
    for {
        // 確認応答を受信
        ack, err := readMessage(s.conn)
        if err != nil {
            log.Printf("確認応答受信エラー: %v", err)
            return
        }
        
        // 確認応答を処理
        if string(ack) == "ACK" {
            s.ackCh <- struct{}{}
        }
    }
}
```

### 3. バッチ処理

複数の小さなメッセージをバッチとして送信することで、ネットワークオーバーヘッドを削減できます。

```go
// バッチ処理の例
type BatchSender struct {
    conn        net.Conn
    batchSize   int
    buffer      []byte
    bufferPos   int
    sendTimeout time.Duration
    lastSend    time.Time
    mu          sync.Mutex
}

func NewBatchSender(conn net.Conn, batchSize int, timeout time.Duration) *BatchSender {
    bs := &BatchSender{
        conn:        conn,
        batchSize:   batchSize,
        buffer:      make([]byte, batchSize),
        sendTimeout: timeout,
        lastSend:    time.Now(),
    }
    
    // タイムアウトによる送信を処理するゴルーチン
    go func() {
        ticker := time.NewTicker(timeout / 2)
        defer ticker.Stop()
        
        for range ticker.C {
            bs.flushIfNeeded()
        }
    }()
    
    return bs
}

func (bs *BatchSender) Send(data []byte) error {
    bs.mu.Lock()
    defer bs.mu.Unlock()
    
    // バッファに収まらない場合は先にフラッシュ
    if bs.bufferPos+len(data) > bs.batchSize {
        if err := bs.flushLocked(); err != nil {
            return err
        }
    }
    
    // データをバッファにコピー
    copy(bs.buffer[bs.bufferPos:], data)
    bs.bufferPos += len(data)
    
    // バッファがいっぱいになったらフラッシュ
    if bs.bufferPos >= bs.batchSize {
        return bs.flushLocked()
    }
    
    return nil
}

func (bs *BatchSender) flushIfNeeded() {
    bs.mu.Lock()
    defer bs.mu.Unlock()
    
    if bs.bufferPos > 0 && time.Since(bs.lastSend) >= bs.sendTimeout {
        bs.flushLocked()
    }
}

func (bs *BatchSender) flushLocked() error {
    if bs.bufferPos == 0 {
        return nil
    }
    
    // バッファを送信
    _, err := bs.conn.Write(bs.buffer[:bs.bufferPos])
    if err != nil {
        return err
    }
    
    bs.bufferPos = 0
    bs.lastSend = time.Now()
    return nil
}

func (bs *BatchSender) Flush() error {
    bs.mu.Lock()
    defer bs.mu.Unlock()
    return bs.flushLocked()
}
```

### 6. プロファイリングとベンチマーク

```go
// ベンチマーク用のヘルパー関数
func benchmark(name string, iterations int, fn func() error) {
    log.Printf("%s ベンチマーク開始 (%d イテレーション)", name, iterations)
    start := time.Now()
    
    successCount := 0
    errorCount := 0
    
    for i := 0; i < iterations; i++ {
        if err := fn(); err != nil {
            errorCount++
        } else {
            successCount++
        }
    }
    
    elapsed := time.Since(start)
    opsPerSec := float64(iterations) / elapsed.Seconds()
    
    log.Printf("%s ベンチマーク結果:", name)
    log.Printf("  合計時間: %v", elapsed)
    log.Printf("  スループット: %.2f ops/sec", opsPerSec)
    log.Printf("  成功: %d, エラー: %d", successCount, errorCount)
}

// 使用例
benchmark("TCPエコーテスト", 1000, func() error {
    conn, err := net.Dial("tcp", "localhost:8080")
    if err != nil {
        return err
    }
    defer conn.Close()
    
    // テストデータを送信
    if _, err := conn.Write([]byte("テストデータ")); err != nil {
        return err
    }
    
    // 応答を読み取る
    buffer := make([]byte, 1024)
    _, err = conn.Read(buffer)
    return err
})
```
