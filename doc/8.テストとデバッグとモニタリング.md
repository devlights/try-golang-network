# 第8章: テスト、デバッグ、モニタリング

## Go言語（バージョン 1.22）でのネットワークアプリケーションの品質保証

本章では、ネットワークアプリケーションの信頼性と品質を確保するための重要な側面であるテスト、デバッグ、およびモニタリングについて詳細に解説します。ネットワークアプリケーションは分散性、非同期性、外部依存性という特性があるため、これらの分野に特化した知識が必要となります。

## 1. ネットワークアプリケーションのテスト戦略

### 1.1 テストの階層と種類

ネットワークアプリケーションのテストは、以下の階層で考えることが重要です：

1. **ユニットテスト**: 個々の関数やメソッドの動作を確認
2. **統合テスト**: 複数のコンポーネントの相互作用を確認
3. **エンドツーエンドテスト**: システム全体の動作を確認
4. **負荷テスト**: システムのパフォーマンスと耐久性を確認

ネットワークアプリケーション特有のテスト考慮事項：
- コネクション管理
- タイムアウト処理
- 再接続ロジック
- エラー処理
- 同時接続処理
- プロトコル適合性

### 1.2 テーブル駆動テスト

Goのテスト手法として定評のあるテーブル駆動テストをネットワーク処理にも適用します。

```go
// Go 1.22
package network

import (
	"testing"
	"net"
	"context"
	"time"
)

func TestConnect(t *testing.T) {
	tests := []struct {
		name        string
		address     string
		timeout     time.Duration
		shouldFail  bool
		expectedErr error
	}{
		{
			name:        "正常な接続",
			address:     "localhost:8080",
			timeout:     time.Second * 5,
			shouldFail:  false,
			expectedErr: nil,
		},
		{
			name:        "タイムアウトエラー",
			address:     "10.255.255.1:8080", // 応答しないアドレス
			timeout:     time.Millisecond * 100,
			shouldFail:  true,
			expectedErr: context.DeadlineExceeded,
		},
		{
			name:        "無効なアドレス",
			address:     "invalid-address",
			timeout:     time.Second * 5,
			shouldFail:  true,
			expectedErr: &net.DNSError{},
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			ctx, cancel := context.WithTimeout(context.Background(), tt.timeout)
			defer cancel()

			conn, err := Connect(ctx, tt.address)
			
			// 失敗ケースのテスト
			if tt.shouldFail {
				if err == nil {
					t.Errorf("期待されるエラーが発生しませんでした")
					if conn != nil {
						conn.Close()
					}
					return
				}
				
				// エラータイプの検証
				switch expected := tt.expectedErr.(type) {
				case *net.DNSError:
					if _, ok := err.(*net.DNSError); !ok {
						t.Errorf("期待されるエラータイプ %T ではなく %T が返されました", expected, err)
					}
				default:
					if err != expected {
						t.Errorf("期待されるエラー %v ではなく %v が返されました", expected, err)
					}
				}
				return
			}
			
			// 成功ケースのテスト
			if err != nil {
				t.Errorf("接続エラー: %v", err)
				return
			}
			
			if conn == nil {
				t.Errorf("接続が確立されましたが、接続オブジェクトがnilです")
				return
			}
			
			defer conn.Close()
			
			// 接続に関する追加検証
			localAddr := conn.LocalAddr()
			if localAddr == nil {
				t.Errorf("ローカルアドレスがnilです")
			}
			
			remoteAddr := conn.RemoteAddr()
			if remoteAddr == nil {
				t.Errorf("リモートアドレスがnilです")
			}
		})
	}
}

// Connect関数はコンテキストとアドレスを受け取りTCP接続を確立する
func Connect(ctx context.Context, address string) (net.Conn, error) {
	var d net.Dialer
	return d.DialContext(ctx, "tcp", address)
}
```

### 1.3 シナリオベースのテスト

実際のユースケースに基づいたシナリオベースのテストも効果的です。

```go
// Go 1.22
func TestHTTPClientRetry(t *testing.T) {
	// テスト用HTTPサーバーの起動
	attempts := 0
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		attempts++
		if attempts <= 2 {
			// 最初の2回はサーバーエラーを返す
			w.WriteHeader(http.StatusInternalServerError)
			return
		}
		// 3回目は成功レスポンスを返す
		w.WriteHeader(http.StatusOK)
		w.Write([]byte(`{"status":"success"}`))
	}))
	defer server.Close()
	
	// リトライ機能付きHTTPクライアントのテスト
	client := NewRetryableHTTPClient(3, time.Millisecond*50)
	resp, err := client.Get(server.URL)
	
	if err != nil {
		t.Fatalf("リクエストエラー: %v", err)
	}
	defer resp.Body.Close()
	
	if resp.StatusCode != http.StatusOK {
		t.Errorf("期待されるステータスコード 200 ではなく %d が返されました", resp.StatusCode)
	}
	
	body, err := io.ReadAll(resp.Body)
	if err != nil {
		t.Fatalf("レスポンスボディの読み取りエラー: %v", err)
	}
	
	expected := `{"status":"success"}`
	if string(body) != expected {
		t.Errorf("期待されるレスポンスボディ %s ではなく %s が返されました", expected, string(body))
	}
	
	if attempts != 3 {
		t.Errorf("期待される試行回数 3 ではなく %d でした", attempts)
	}
}
```

### 1.4 並行テスト

Goの優れた並行処理機能を活かした並行テストは、ネットワークアプリケーションの安定性評価に有効です。

```go
// Go 1.22
func TestConcurrentConnections(t *testing.T) {
	server := NewTCPServer(":0") // 任意のポートでリスン
	defer server.Close()
	
	port := server.Addr().(*net.TCPAddr).Port
	address := fmt.Sprintf("localhost:%d", port)
	
	const concurrentClients = 100
	const messagesPerClient = 10
	
	var wg sync.WaitGroup
	wg.Add(concurrentClients)
	
	errorCh := make(chan error, concurrentClients*messagesPerClient)
	
	// 複数クライアントの並行実行
	for i := 0; i < concurrentClients; i++ {
		go func(clientID int) {
			defer wg.Done()
			
			client := NewTCPClient()
			err := client.Connect(address)
			if err != nil {
				errorCh <- fmt.Errorf("クライアント %d の接続エラー: %v", clientID, err)
				return
			}
			defer client.Close()
			
			for j := 0; j < messagesPerClient; j++ {
				message := fmt.Sprintf("Client %d Message %d", clientID, j)
				err := client.Send(message)
				if err != nil {
					errorCh <- fmt.Errorf("クライアント %d のメッセージ %d 送信エラー: %v", clientID, j, err)
					return
				}
				
				response, err := client.Receive()
				if err != nil {
					errorCh <- fmt.Errorf("クライアント %d のメッセージ %d 受信エラー: %v", clientID, j, err)
					return
				}
				
				expected := fmt.Sprintf("ECHO: %s", message)
				if response != expected {
					errorCh <- fmt.Errorf("クライアント %d のメッセージ %d: 期待値 %q ではなく %q が返されました", 
						clientID, j, expected, response)
				}
			}
		}(i)
	}
	
	// すべてのクライアントが完了するまで待機
	wg.Wait()
	close(errorCh)
	
	// エラーの収集と報告
	var errors []error
	for err := range errorCh {
		errors = append(errors, err)
	}
	
	if len(errors) > 0 {
		t.Errorf("%d 件のエラーが発生しました:", len(errors))
		for i, err := range errors {
			t.Errorf("  %d: %v", i+1, err)
		}
	}
}
```

## 2. モックとスタブを使用したテスト

### 2.1 インターフェースによるモック化

Goのインターフェースを活用することで、ネットワーク依存部分を容易にモック化できます。

```go
// Go 1.22
// ネットワーク接続のインターフェース
type Connection interface {
	Send(data []byte) error
	Receive() ([]byte, error)
	Close() error
}

// 実際のTCP接続の実装
type TCPConnection struct {
	conn net.Conn
}

func (c *TCPConnection) Send(data []byte) error {
	_, err := c.conn.Write(data)
	return err
}

func (c *TCPConnection) Receive() ([]byte, error) {
	buffer := make([]byte, 4096)
	n, err := c.conn.Read(buffer)
	if err != nil {
		return nil, err
	}
	return buffer[:n], nil
}

func (c *TCPConnection) Close() error {
	return c.conn.Close()
}

// モック接続の実装
type MockConnection struct {
	SendFunc    func(data []byte) error
	ReceiveFunc func() ([]byte, error)
	CloseFunc   func() error
	
	SendCalls    [][]byte
	ReceiveCalls int
	CloseCalls   int
}

func (m *MockConnection) Send(data []byte) error {
	m.SendCalls = append(m.SendCalls, data)
	if m.SendFunc != nil {
		return m.SendFunc(data)
	}
	return nil
}

func (m *MockConnection) Receive() ([]byte, error) {
	m.ReceiveCalls++
	if m.ReceiveFunc != nil {
		return m.ReceiveFunc()
	}
	return []byte("mock response"), nil
}

func (m *MockConnection) Close() error {
	m.CloseCalls++
	if m.CloseFunc != nil {
		return m.CloseFunc()
	}
	return nil
}
```

```go
// Go 1.22
func TestClientWithMockServer(t *testing.T) {
	// モックHTTPサーバー
	responses := map[string]struct {
		statusCode int
		body       string
	}{
		"/api/users": {
			statusCode: http.StatusOK,
			body:       `[{"id":1,"name":"User1"},{"id":2,"name":"User2"}]`,
		},

			statusCode: http.StatusOK,
			body:       `{"id":1,"name":"User1"}`,

		resp, ok := responses[r.URL.Path]
		if !ok {
			w.WriteHeader(http.StatusNotFound)
			w.Write([]byte(`{"error":"Not found"}`))
			return
		}
		
		w.WriteHeader(resp.statusCode)
		w.Write([]byte(resp.body))
	}))
	defer server.Close()
	
	// クライアントの初期化
	client := NewAPIClient(server.URL)
	
	// ユーザー一覧取得のテスト
	users, err := client.GetUsers()
	if err != nil {
		t.Fatalf("ユーザー一覧取得エラー: %v", err)
	}
	
	if len(users) != 2 {
		t.Errorf("期待されるユーザー数 2 ではなく %d が返されました", len(users))
	}
	
	// 単一ユーザー取得のテスト
	user, err := client.GetUser(1)
	if err != nil {
		t.Fatalf("ユーザー取得エラー: %v", err)
	}
	
	if user.ID != 1 || user.Name != "User1" {
		t.Errorf("期待されるユーザー {ID:1 Name:\"User1\"} ではなく {ID:%d Name:\"%s\"} が返されました", 
			user.ID, user.Name)
	}
	
	// エラーハンドリングのテスト
	_, err = client.GetErrorEndpoint()
	if err == nil {
		t.Fatal("エラーエンドポイントでエラーが返されませんでした")
	}
	
	// カスタムエラータイプの検証
	apiErr, ok := err.(*APIError)
	if !ok {
		t.Fatalf("期待されるエラータイプ *APIError ではなく %T が返されました", err)
	}
	
	if apiErr.StatusCode != http.StatusInternalServerError {
		t.Errorf("期待されるステータスコード 500 ではなく %d が返されました", apiErr.StatusCode)
	}
}
```

### 2.3 高度なモック技術：動的な挙動

ランダム性やタイミングの変化など、実際のネットワークの動的特性をシミュレートするモックも作成できます。

```go
// Go 1.22
func TestWithUnreliableNetwork(t *testing.T) {
	// 不安定なネットワーク接続をシミュレートするモック
	mock := &MockConnection{
		SendFunc: func(data []byte) error {
			// 20%の確率で接続エラーをシミュレート
			if rand.Float32() < 0.2 {
				return fmt.Errorf("シミュレートされた接続エラー")
			}
			
			// 10%の確率でタイムアウトをシミュレート
			if rand.Float32() < 0.1 {
				time.Sleep(time.Second * 2)
				return context.DeadlineExceeded
			}
			
			return nil
		},
		ReceiveFunc: func() ([]byte, error) {
			// 15%の確率でデータ損失をシミュレート
			if rand.Float32() < 0.15 {
				return []byte{}, fmt.Errorf("シミュレートされたデータ損失")
			}
			
			// 5%の確率で破損データをシミュレート
			if rand.Float32() < 0.05 {
				return []byte("corrupted data"), nil
			}
			
			// 正常レスポンス
			return []byte(`{"status":"success"}`), nil
		},
	}
	
	// リトライ機能を持つクライアント
	client := NewReliableClient(mock, RetryOptions{
		MaxRetries:    5,
		InitialDelay:  time.Millisecond * 10,
		MaxDelay:      time.Millisecond * 500,
		BackoffFactor: 2.0,
	})
	
	// 100回のオペレーションをテスト
	successCount := 0
	const operations = 100
	
	for i := 0; i < operations; i++ {
		err := client.SendOperation([]byte("test data"))
		if err == nil {
			successCount++
		}
	}
	
	// 成功率の検証（理論上80%以上のオペレーションが成功するはず）
	successRate := float64(successCount) / float64(operations)
	t.Logf("成功率: %.2f%%", successRate*100)
	
	if successRate < 0.8 {
		t.Errorf("期待される成功率 80%以上 ではなく %.2f%% でした", successRate*100)
	}
}
```

### 2.4 依存性の注入によるテスト

よりクリーンなテストのためには、依存性の注入を活用します。

```go
// Go 1.22
// ダイアラーインターフェース
type Dialer interface {
	Dial(network, address string) (net.Conn, error)
}

// 実際のダイアラー
type TCPDialer struct{}

func (d *TCPDialer) Dial(network, address string) (net.Conn, error) {
	return net.Dial(network, address)
}

// モックダイアラー
type MockDialer struct {
	DialFunc func(network, address string) (net.Conn, error)
	Calls    []struct {
		Network string
		Address string
	}
}

func (m *MockDialer) Dial(network, address string) (net.Conn, error) {
	m.Calls = append(m.Calls, struct {
		Network string
		Address string
	}{network, address})
	
	if m.DialFunc != nil {
		return m.DialFunc(network, address)
	}
	return nil, fmt.Errorf("モックダイアラー: 接続未実装")
}

// 依存性注入を使用したクライアント
type Client struct {
	dialer  Dialer
	timeout time.Duration
}

func NewClient(dialer Dialer, timeout time.Duration) *Client {
	return &Client{
		dialer:  dialer,
		timeout: timeout,
	}
}

func (c *Client) Connect(address string) (net.Conn, error) {
	return c.dialer.Dial("tcp", address)
}

// 依存性注入を使用したテスト
func TestClientWithMockDialer(t *testing.T) {
	mockConn := &MockConn{}
	mockDialer := &MockDialer{
		DialFunc: func(network, address string) (net.Conn, error) {
			if network != "tcp" {
				return nil, fmt.Errorf("サポートされていないネットワークタイプ: %s", network)
			}
			if address != "example.com:80" {
				return nil, fmt.Errorf("無効なアドレス: %s", address)
			}
			return mockConn, nil
		},
	}
	
	client := NewClient(mockDialer, time.Second*10)
	conn, err := client.Connect("example.com:80")
	
	if err != nil {
		t.Fatalf("接続エラー: %v", err)
	}
	
	if conn != mockConn {
		t.Errorf("期待される接続オブジェクトが返されませんでした")
	}
	
	if len(mockDialer.Calls) != 1 {
		t.Fatalf("Dial メソッドが1回呼ばれるべきですが、%d回呼ばれました", len(mockDialer.Calls))
	}
	
	call := mockDialer.Calls[0]
	if call.Network != "tcp" || call.Address != "example.com:80" {
		t.Errorf("Dial(%q, %q)が呼ばれるべきですが、Dial(%q, %q)が呼ばれました", 
			"tcp", "example.com:80", call.Network, call.Address)
	}
	
	// エラーケースのテスト
	_, err = client.Connect("invalid:999999")
	if err == nil {
		t.Error("無効なアドレスでエラーが返されませんでした")
	}
}
```

## 3. デバッグ手法とツール

### 3.1 効果的なロギング戦略

ネットワークアプリケーションのデバッグには、適切なロギングが不可欠です。

```go
// Go 1.22
package networking

import (
	"fmt"
	"log/slog"
	"net"
	"os"
	"time"
)

// ネットワークロガー
type NetworkLogger struct {
	logger *slog.Logger
}

// ログレベル
const (
	LevelDebug = slog.LevelDebug
	LevelInfo  = slog.LevelInfo
	LevelWarn  = slog.LevelWarn
	LevelError = slog.LevelError
)

// 新規ロガーの作成
func NewNetworkLogger(level slog.Level) *NetworkLogger {
	opts := &slog.HandlerOptions{
		Level: level,
		ReplaceAttr: func(groups []string, a slog.Attr) slog.Attr {
			// タイムスタンプのフォーマット変更
			if a.Key == slog.TimeKey {
				return slog.Attr{
					Key:   slog.TimeKey,
					Value: slog.StringValue(a.Value.Time().Format(time.RFC3339Nano)),
				}
			}
			return a
		},
	}
	
	handler := slog.NewJSONHandler(os.Stdout, opts)
	logger := slog.New(handler)
	
	return &NetworkLogger{
		logger: logger,
	}
}

// コネクション確立をログ
func (nl *NetworkLogger) LogConnectionEstablished(local, remote net.Addr, connID string) {
	nl.logger.Info("コネクション確立",
		"conn_id", connID,
		"local_addr", local.String(),
		"remote_addr", remote.String(),
		"network", local.Network(),
	)
}

// コネクションクローズをログ
func (nl *NetworkLogger) LogConnectionClosed(connID string, duration time.Duration, bytesRx, bytesTx int64, err error) {
	attrs := []any{
		"conn_id", connID,
		"duration_ms", duration.Milliseconds(),
		"bytes_received", bytesRx,
		"bytes_sent", bytesTx,
	}
	
	if err != nil {
		attrs = append(attrs, "error", err.Error())
		nl.logger.Warn("コネクションクローズ（エラー発生）", attrs...)
	} else {
		nl.logger.Info("コネクションクローズ（正常）", attrs...)
	}
}

// パケット送信をログ
func (nl *NetworkLogger) LogPacketSent(connID string, packetType string, packetID int, size int) {
	nl.logger.Debug("パケット送信",
		"conn_id", connID,
		"packet_type", packetType,
		"packet_id", packetID,
		"size_bytes", size,

	)
}

// パケット受信をログ
func (nl *NetworkLogger) LogPacketReceived(connID string, packetType string, packetID int, size int, latency time.Duration) {
	nl.logger.Debug("パケット受信",

		"packet_type", packetType,
		"packet_id", packetID,
		"size_bytes", size,

	)
}
```

### 3.2 パケットキャプチャとネットワークスニッファー

ネットワーク問題の診断には、パケットキャプチャが有効です。Goからtcpdumpなどのツールを連携したデバッグ環境を構築できます。

```go
// Go 1.22
package debugging

import (
	"bufio"
	"fmt"
	"io"
	"os/exec"
	"strings"
	"sync"
)

// パケットキャプチャマネージャー
type PacketCaptureManager struct {
	cmd      *exec.Cmd
	stdout   io.ReadCloser
	stderr   io.ReadCloser
	stopCh   chan struct{}
	waitDone chan struct{}
	mutex    sync.Mutex
	running  bool
}

// 新規パケットキャプチャマネージャーの作成
func NewPacketCaptureManager() *PacketCaptureManager {
	return &PacketCaptureManager{
		stopCh:   make(chan struct{}),
		waitDone: make(chan struct{}),
	}
}

// インターフェースとポートでパケットキャプチャを開始
func (pcm *PacketCaptureManager) Start(iface string, port int, outputFile string) error {
	pcm.mutex.Lock()
	defer pcm.mutex.Unlock()
	
	if pcm.running {
		return fmt.Errorf("パケットキャプチャは既に実行中です")
	}
	
	// tcpdumpコマンドの構築
	filter := fmt.Sprintf("port %d", port)
	args := []string{"-i", iface, "-n", "-w", outputFile, filter}
	
	pcm.cmd = exec.Command("tcpdump", args...)
	
	var err error
	pcm.stdout, err = pcm.cmd.StdoutPipe()
	if err != nil {
		return fmt.Errorf("stdout パイプ作成エラー: %w", err)
	}
	
	pcm.stderr, err = pcm.cmd.StderrPipe()
	if err != nil {
		return fmt.Errorf("stderr パイプ作成エラー: %w", err)
	}
	
	// 標準出力の処理
	go pcm.processOutput(pcm.stdout, "STDOUT")
	
	// 標準エラー出力の処理
	go pcm.processOutput(pcm.stderr, "STDERR")
	
	// コマンド実行
	if err := pcm.cmd.Start(); err != nil {
		return fmt.Errorf("tcpdump 起動エラー: %w", err)
	}
	
	pcm.running = true
	
	// 非同期でコマンド終了を監視
	go func() {
		defer close(pcm.waitDone)
		err := pcm.cmd.Wait()
		pcm.mutex.Lock()
		pcm.running = false
		pcm.mutex.Unlock()
		
		if err != nil && !strings.Contains(err.Error(), "killed") {
			fmt.Printf("tcpdump 終了エラー: %v\n", err)
		}
	}()
	
	fmt.Printf("パケットキャプチャ開始: インターフェース=%s, ポート=%d, 出力ファイル=%s\n",
		iface, port, outputFile)
	
	return nil
}

// パケットキャプチャの停止
func (pcm *PacketCaptureManager) Stop() error {
	pcm.mutex.Lock()
	running := pcm.running
	pcm.mutex.Unlock()
	
	if !running {
		return fmt.Errorf("パケットキャプチャは実行中ではありません")
	}
	
	// プロセスを終了
	if err := pcm.cmd.Process.Kill(); err != nil {
		return fmt.Errorf("tcpdump プロセス終了エラー: %w", err)
	}
	
	// 終了完了を待機
	<-pcm.waitDone
	
	fmt.Println("パケットキャプチャ停止")
	return nil
}

// 出力処理
func (pcm *PacketCaptureManager) processOutput(r io.Reader, prefix string) {
	scanner := bufio.NewScanner(r)
	for scanner.Scan() {
		line := scanner.Text()
		fmt.Printf("[%s] %s\n", prefix, line)
	}
}

// 使用例
func ExampleUsage() {
	captureManager := NewPacketCaptureManager()
	
	// HTTPサーバー起動（例）
	// httpServer := startHTTPServer(":8080")
	
	// パケットキャプチャ開始

	if err != nil {
		fmt.Printf("キャプチャ開始エラー: %v\n", err)
		return
	}
	
	// テスト実行...
	
	// パケットキャプチャ停止
	err = captureManager.Stop()
	if err != nil {
		fmt.Printf("キャプチャ停止エラー: %v\n", err)
	}
	
	// HTTPサーバー停止（例）
	// httpServer.Shutdown(context.Background())
	
	fmt.Println("キャプチャファイル: http_traffic.pcap を Wireshark で分析できます")
}
```

### 3.3 実行時ネットワーク状態監視

アプリケーションの実行時にネットワーク状態を監視するデバッグツールも重要です。

```go
// Go 1.22
package debugging

import (
	"context"
	"fmt"
	"net"
	"sync"
	"time"
)

// コネクション状態
type ConnectionState struct {
	ID            string
	LocalAddr     string
	RemoteAddr    string
	Protocol      string
	EstablishedAt time.Time
	LastActivity  time.Time
	BytesSent     int64
	BytesReceived int64
	RTT           time.Duration
	State         string
	ErrorCount    int
}

// ネットワーク状態モニター
type NetworkStateMonitor struct {
	connections map[string]*ConnectionState
	mutex       sync.RWMutex
	ctx         context.Context
	cancel      context.CancelFunc
}

// 新規ネットワーク状態モニターの作成
func NewNetworkStateMonitor() *NetworkStateMonitor {
	ctx, cancel := context.WithCancel(context.Background())
	return &NetworkStateMonitor{
		connections: make(map[string]*ConnectionState),
		ctx:         ctx,
		cancel:      cancel,
	}
}

// コネクションの追跡開始
func (nsm *NetworkStateMonitor) TrackConnection(id string, conn net.Conn) {
	nsm.mutex.Lock()
	defer nsm.mutex.Unlock()
	
	nsm.connections[id] = &ConnectionState{
		ID:            id,
		LocalAddr:     conn.LocalAddr().String(),
		RemoteAddr:    conn.RemoteAddr().String(),
		Protocol:      conn.LocalAddr().Network(),
		EstablishedAt: time.Now(),
		LastActivity:  time.Now(),
		State:         "established",
	}
	
	// 定期的なRTT測定（TCPの場合）
	if tcpConn, ok := conn.(*net.TCPConn); ok {
		go nsm.monitorTCPConnection(id, tcpConn)
	}
}

// TCPコネクションのモニタリング
func (nsm *NetworkStateMonitor) monitorTCPConnection(id string, conn *net.TCPConn) {
	ticker := time.NewTicker(5 * time.Second)
	defer ticker.Stop()
	
	for {
		select {
		case <-nsm.ctx.Done():
			return
		case <-ticker.C:
			// 接続の存在確認
			nsm.mutex.RLock()
			_, exists := nsm.connections[id]
			nsm.mutex.RUnlock()
			
			if !exists {
				return
			}
			
			// RTT測定のソケットオプションを使用
			err := conn.SetReadDeadline(time.Now().Add(500 * time.Millisecond))
			if err != nil {
				nsm.updateConnectionError(id, "set_deadline_error")
				continue
			}
			
			start := time.Now()
			
			// 1バイトのデータでTCPのACKを誘発（ping）
			_, err = conn.Write([]byte{0})
			if err != nil {
				nsm.updateConnectionError(id, "write_error")
				continue
			}
			
			// レスポンス読み取り
			buffer := make([]byte, 1)
			_, err = conn.Read(buffer)
			
			elapsed := time.Since(start)
			
			if err != nil {
				if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
					// タイムアウトは予想されるので無視
				} else {
					nsm.updateConnectionError(id, "read_error")
					continue
				}
			}
			
			// RTT更新
			nsm.mutex.Lock()
			if conn, ok := nsm.connections[id]; ok {
				conn.RTT = elapsed
				conn.LastActivity = time.Now()
			}
			nsm.mutex.Unlock()
		}
	}
}

// データ送信を記録
func (nsm *NetworkStateMonitor) RecordDataSent(id string, bytes int64) {
	nsm.mutex.Lock()
	defer nsm.mutex.Unlock()
	
	if conn, ok := nsm.connections[id]; ok {
		conn.BytesSent += bytes
		conn.LastActivity = time.Now()
	}
}

// データ受信を記録
func (nsm *NetworkStateMonitor) RecordDataReceived(id string, bytes int64) {
	nsm.mutex.Lock()
	defer nsm.mutex.Unlock()
	
	if conn, ok := nsm.connections[id]; ok {
		conn.BytesReceived += bytes
		conn.LastActivity = time.Now()
	}
}

// コネクションエラーを更新
func (nsm *NetworkStateMonitor) updateConnectionError(id string, state string) {
	nsm.mutex.Lock()
	defer nsm.mutex.Unlock()
	
	if conn, ok := nsm.connections[id]; ok {
		conn.State = state
		conn.ErrorCount++
	}
}

// コネクション状態を取得
func (nsm *NetworkStateMonitor) GetConnectionState(id string) (*ConnectionState, bool) {
	nsm.mutex.RLock()
	defer nsm.mutex.RUnlock()
	
	conn, ok := nsm.connections[id]
	if !ok {
		return nil, false
	}
	
	// オブジェクトをコピーして返す
	stateCopy := *conn
	return &stateCopy, true
}

// すべてのコネクション状態を取得
func (nsm *NetworkStateMonitor) GetAllConnectionStates() []*ConnectionState {
	nsm.mutex.RLock()
	defer nsm.mutex.RUnlock()
	
	states := make([]*ConnectionState, 0, len(nsm.connections))
	for _, conn := range nsm.connections {
		stateCopy := *conn
		states = append(states, &stateCopy)
	}
	
	return states
}

// コネクションのクローズを記録
func (nsm *NetworkStateMonitor) RecordConnectionClosed(id string) {
	nsm.mutex.Lock()
	defer nsm.mutex.Unlock()
	
	if conn, ok := nsm.connections[id]; ok {
		conn.State = "closed"
	}
}

// コネクションの追跡を終了
func (nsm *NetworkStateMonitor) StopTracking(id string) {
	nsm.mutex.Lock()
	defer nsm.mutex.Unlock()
	
	delete(nsm.connections, id)
}

// モニターの停止
func (nsm *NetworkStateMonitor) Stop() {
	nsm.cancel()
}

// デバッグレポートの生成
func (nsm *NetworkStateMonitor) GenerateReport() string {
	states := nsm.GetAllConnectionStates()
	
	report := "ネットワーク接続状態レポート:\n"
	report += "================================\n\n"
	
	if len(states) == 0 {
		report += "アクティブな接続はありません。\n"
		return report
	}
	
	report += fmt.Sprintf("アクティブな接続数: %d\n\n", len(states))
	
	for i, state := range states {
		report += fmt.Sprintf("接続 #%d (ID: %s)\n", i+1, state.ID)
		report += fmt.Sprintf("  プロトコル: %s\n", state.Protocol)
		report += fmt.Sprintf("  ローカルアドレス: %s\n", state.LocalAddr)
		report += fmt.Sprintf("  リモートアドレス: %s\n", state.RemoteAddr)
		report += fmt.Sprintf("  確立時刻: %s\n", state.EstablishedAt.Format(time.RFC3339))
		report += fmt.Sprintf("  最終アクティビティ: %s (%s前)\n", 
			state.LastActivity.Format(time.RFC3339),
			time.Since(state.LastActivity).Round(time.Second))
		report += fmt.Sprintf("  送信バイト数: %d\n", state.BytesSent)
		report += fmt.Sprintf("  受信バイト数: %d\n", state.BytesReceived)
		report += fmt.Sprintf("  RTT: %s\n", state.RTT)
		report += fmt.Sprintf("  状態: %s\n", state.State)
		report += fmt.Sprintf("  エラー数: %d\n", state.ErrorCount)
		report += "\n"
	}
	
	return report
}
```

### 3.4 サーキットブレーカーの実装とデバッグ

障害検出と回復のためのサーキットブレーカーパターンの実装と、その挙動確認のためのツールを提供します。

```go
// Go 1.22
package debugging

import (
	"errors"
	"fmt"
	"sync"
	"sync/atomic"
	"time"
)

// サーキットブレーカーの状態
type State int32

const (
	StateClosed State = iota
	StateHalfOpen
	StateOpen
)

// サーキットブレーカーの設定
type CircuitBreakerConfig struct {
	FailureThreshold   uint32        // 失敗閾値
	SuccessThreshold   uint32        // 成功閾値
	Timeout            time.Duration // 開放状態のタイムアウト
	HalfOpenMaxCalls   uint32        // 半開状態での最大呼び出し数
	WindowSize         uint32        // エラー率計算のウィンドウサイズ
	MinimumRequestRate uint32        // 最小リクエスト率
}

// デフォルト設定を返す
func DefaultCircuitBreakerConfig() CircuitBreakerConfig {
	return CircuitBreakerConfig{
		FailureThreshold:   5,                  // 5回の失敗でオープン
		SuccessThreshold:   2,                  // 2回の成功でクローズ
		Timeout:            time.Second * 30,   // 30秒後に半開状態に移行
		HalfOpenMaxCalls:   10,                 // 半開状態で最大10回の呼び出しを許可
		WindowSize:         100,                // 直近100リクエストでエラー率を計算
		MinimumRequestRate: 3,                  // 少なくとも3リクエスト/秒
	}
}

// サーキットブレーカー
type CircuitBreaker struct {
	name              string
	config            CircuitBreakerConfig
	state             atomic.Int32
	failures          atomic.Uint32
	successes         atomic.Uint32
	halfOpenAllowed   atomic.Uint32
	lastStateChange   atomic.Int64
	lastFailure       atomic.Int64
	mutex             sync.RWMutex
	requestCounts     []uint32 // 単位時間あたりのリクエスト数
	currentBucketIdx  int
	lastBucketRotation time.Time
}

// ErrCircuitOpen はサーキットが開いている場合のエラー
var ErrCircuitOpen = errors.New("circuit breaker is open")

// 新規サーキットブレーカーの作成
func NewCircuitBreaker(name string, config CircuitBreakerConfig) *CircuitBreaker {
	cb := &CircuitBreaker{
		name:             name,
		config:           config,
		requestCounts:    make([]uint32, 10), // 10秒間のリクエスト数を記録
		lastBucketRotation: time.Now(),
	}
	cb.state.Store(int32(StateClosed))
	cb.lastStateChange.Store(time.Now().UnixNano())
	
	// リクエスト率を監視するゴルーチンを開始
	go cb.monitorRequestRate()
	
	return cb
}

// リクエスト率の監視
func (cb *CircuitBreaker) monitorRequestRate() {
	ticker := time.NewTicker(time.Second)
	defer ticker.Stop()
	
	for range ticker.C {
		cb.rotateBucket()
	}
}

// バケットのローテーション
func (cb *CircuitBreaker) rotateBucket() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()
	
	cb.currentBucketIdx = (cb.currentBucketIdx + 1) % len(cb.requestCounts)
	cb.requestCounts[cb.currentBucketIdx] = 0
	cb.lastBucketRotation = time.Now()
}

// 現在のリクエスト率を計算
func (cb *CircuitBreaker) currentRequestRate() uint32 {
	cb.mutex.RLock()
	defer cb.mutex.RUnlock()
	
	var total uint32
	for _, count := range cb.requestCounts {
		total += count
	}
	return total / uint32(len(cb.requestCounts))
}

// リクエストを記録
func (cb *CircuitBreaker) recordRequest() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()
	
	cb.requestCounts[cb.currentBucketIdx]++
}

// 実行
func (cb *CircuitBreaker) Execute(fn func() error) error {
	// リクエストを記録
	cb.recordRequest()
	
	// 現在の状態を確認
	state := State(cb.state.Load())
	
	switch state {
	case StateClosed:
		// サーキットがクローズしている場合、関数を実行
		err := fn()
		if err != nil {
			cb.recordFailure()
			return err
		}
		
		cb.recordSuccess()
		return nil
		
	case StateHalfOpen:
		// 半開状態では、制限された数のリクエストのみ許可
		allowed := cb.halfOpenAllowed.Add(1)
		if allowed > cb.config.HalfOpenMaxCalls {
			return ErrCircuitOpen
		}
		
		err := fn()
		if err != nil {
			cb.tripOpen()
			return err
		}
		
		// 成功をカウント
		successes := cb.successes.Add(1)
		if successes >= cb.config.SuccessThreshold {
			cb.closeCircuit()
		}
		
		return nil
		
	case StateOpen:
		// タイムアウトチェック
		lastChange := time.Unix(0, cb.lastStateChange.Load())
		if time.Since(lastChange) > cb.config.Timeout {
			cb.attemptHalfOpen()
		}
		
		return ErrCircuitOpen
	}
	
	// コードが到達することはないが、コンパイラを満足させる
	return errors.New("unexpected circuit breaker state")
}

// 失敗を記録
func (cb *CircuitBreaker) recordFailure() {
	failures := cb.failures.Add(1)
	cb.lastFailure.Store(time.Now().UnixNano())
	
	// エラー率を計算
	rate := cb.currentRequestRate()
	if rate >= cb.config.MinimumRequestRate && failures >= cb.config.FailureThreshold {
		cb.tripOpen()
	}
}

// 成功を記録
func (cb *CircuitBreaker) recordSuccess() {
	cb.failures.Store(0)
}

// サーキットを開放
func (cb *CircuitBreaker) tripOpen() {
	// CASTによるアトミックな状態遷移
	if cb.state.CompareAndSwap(int32(StateClosed), int32(StateOpen)) ||
	   cb.state.CompareAndSwap(int32(StateHalfOpen), int32(StateOpen)) {
		cb.lastStateChange.Store(time.Now().UnixNano())
		cb.failures.Store(0)
		cb.successes.Store(0)
		fmt.Printf("[CircuitBreaker:%s] サーキットブレーカーがOPEN状態に遷移しました\n", cb.name)
	}
}

// サーキットをクローズ
func (cb *CircuitBreaker) closeCircuit() {
	if cb.state.CompareAndSwap(int32(StateHalfOpen), int32(StateClosed)) {
		cb.lastStateChange.Store(time.Now().UnixNano())
		cb.failures.Store(0)
		cb.successes.Store(0)
		fmt.Printf("[CircuitBreaker:%s] サーキットブレーカーがCLOSED状態に遷移しました\n", cb.name)
	}
}

// 半開状態への移行を試みる
func (cb *CircuitBreaker) attemptHalfOpen() {
	if cb.state.CompareAndSwap(int32(StateOpen), int32(StateHalfOpen)) {
		cb.lastStateChange.Store(time.Now().UnixNano())
		cb.halfOpenAllowed.Store(0)
		cb.failures.Store(0)
		cb.successes.Store(0)
		fmt.Printf("[CircuitBreaker:%s] サーキットブレーカーがHALF-OPEN状態に遷移しました\n", cb.name)
	}
}

// 現在の状態を取得
func (cb *CircuitBreaker) GetState() State {
	return State(cb.state.Load())
}

// 状態の文字列表現を取得
func (cb *CircuitBreaker) GetStateString() string {
	state := cb.GetState()
	switch state {
	case StateClosed:
		return "CLOSED"
	case StateHalfOpen:
		return "HALF-OPEN"
	case StateOpen:
		return "OPEN"
	default:
		return "UNKNOWN"
	}
}

// メトリクスを取得
func (cb *CircuitBreaker) GetMetrics() map[string]interface{} {
	state := cb.GetState()
	lastChange := time.Unix(0, cb.lastStateChange.Load())
	lastFailure := time.Unix(0, cb.lastFailure.Load())
	
	return map[string]interface{}{
		"name":                 cb.name,
		"state":                cb.GetStateString(),
		"failures":             cb.failures.Load(),
		"successes":            cb.successes.Load(),
		"request_rate":         cb.currentRequestRate(),
		"since_last_state_change_ms": time.Since(lastChange).Milliseconds(),
		"since_last_failure_ms":     time.Since(lastFailure).Milliseconds(),
	}
}

// サーキットブレーカーのデバッグ出力を生成
func (cb *CircuitBreaker) Debug() string {
	metrics := cb.GetMetrics()
	
	output := fmt.Sprintf("サーキットブレーカー: %s\n", metrics["name"])
	output += fmt.Sprintf("  状態: %s\n", metrics["state"])
	output += fmt.Sprintf("  失敗カウント: %d/%d\n", metrics["failures"], cb.config.FailureThreshold)
	output += fmt.Sprintf("  成功カウント: %d/%d\n", metrics["successes"], cb.config.SuccessThreshold)
	output += fmt.Sprintf("  リクエスト率: %d/秒 (最小閾値: %d/秒)\n", 
		metrics["request_rate"], cb.config.MinimumRequestRate)
	output += fmt.Sprintf("  最終状態変更からの経過時間: %dms\n", metrics["since_last_state_change_ms"])
	output += fmt.Sprintf("  最終障害からの経過時間: %dms\n", metrics["since_last_failure_ms"])
	
	if state := State(cb.state.Load()); state == StateOpen {
		remaining := cb.config.Timeout - time.Since(time.Unix(0, cb.lastStateChange.Load()))
		if remaining > 0 {
			output += fmt.Sprintf("  半開状態への移行まで: %.1f秒\n", remaining.Seconds())
		} else {
			output += "  次回の実行で半開状態に移行します\n"
		}
	} else if state == StateHalfOpen {
		output += fmt.Sprintf("  半開状態での実行数: %d/%d\n", 
			cb.halfOpenAllowed.Load(), cb.config.HalfOpenMaxCalls)
	}
	
	return output
}
```

## 4. パフォーマンスのプロファイリングと測定

### 4.1 Goのプロファイリングツール

Goにはパフォーマンス解析のための強力なツールが組み込まれています。

```go
// Go 1.22
package profiling

import (
	"fmt"
	"log"
	"net/http"
	_ "net/http/pprof" // pprof エンドポイントの自動登録
	"os"
	"runtime"
	"runtime/pprof"
	"time"
)

// プロファイラーマネージャー
type ProfilerManager struct {
	cpuProfile   *os.File
	memProfile   *os.File
	httpEnabled  bool
	httpAddr     string
	httpServer   *http.Server
	startTime    time.Time
}

// 新規プロファイラーマネージャーの作成
func NewProfilerManager() *ProfilerManager {
	return &ProfilerManager{}
}

// CPUプロファイリングの開始
func (pm *ProfilerManager) StartCPUProfiling(filename string) error {
	var err error
	pm.cpuProfile, err = os.Create(filename)
	if err != nil {
		return fmt.Errorf("CPUプロファイル作成エラー: %w", err)
	}
	
	if err := pprof.StartCPUProfile(pm.cpuProfile); err != nil {
		pm.cpuProfile.Close()
		return fmt.Errorf("CPUプロファイリング開始エラー: %w", err)
	}
	
	log.Printf("CPUプロファイリング開始: %s", filename)
	pm.startTime = time.Now()
	return nil
}

// CPUプロファイリングの停止
func (pm *ProfilerManager) StopCPUProfiling() {
	if pm.cpuProfile != nil {
		pprof.StopCPUProfile()
		pm.cpuProfile.Close()
		pm.cpuProfile = nil
		log.Printf("CPUプロファイリング停止 (実行時間: %s)", time.Since(pm.startTime))
	}
}

// メモリプロファイリングの取得
func (pm *ProfilerManager) CaptureMemoryProfile(filename string) error {
	var err error
	pm.memProfile, err = os.Create(filename)
	if err != nil {
		return fmt.Errorf("メモリプロファイル作成エラー: %w", err)
	}
	
	runtime.GC() // 測定前にGCを強制実行
	
	if err := pprof.WriteHeapProfile(pm.memProfile); err != nil {
		pm.memProfile.Close()
		return fmt.Errorf("メモリプロファイル書き込みエラー: %w", err)
	}
	
	pm.memProfile.Close()
	pm.memProfile = nil
	
	log.Printf("メモリプロファイル保存: %s", filename)
	return nil
}

// ゴルーチンプロファイルの取得
func (pm *ProfilerManager) CaptureGoroutineProfile(filename string) error {
	f, err := os.Create(filename)
	if err != nil {
		return fmt.Errorf("ゴルーチンプロファイル作成エラー: %w", err)
	}
	defer f.Close()
	
	if err := pprof.Lookup("goroutine").WriteTo(f, 0); err != nil {
		return fmt.Errorf("ゴルーチンプロファイル書き込みエラー: %w", err)
	}
	
	log.Printf("ゴルーチンプロファイル保存: %s", filename)
	return nil
}
```

### 4.2 ネットワークパフォーマンスベンチマーク

ネットワークアプリケーションのパフォーマンスを測定するためのベンチマークツールを実装します。

```go
// Go 1.22
package benchmarking

import (
	"context"
	"fmt"
	"math"
	"net"
	"sync"
	"sync/atomic"
	"time"
)

// ベンチマーク結果
type BenchmarkResult struct {
	TotalRequests      int64
	SuccessfulRequests int64
	FailedRequests     int64
	TotalDuration      time.Duration
	MinLatency         time.Duration
	MaxLatency         time.Duration
	AvgLatency         time.Duration
	Percentiles        map[float64]time.Duration
	RequestsPerSecond  float64
	TotalBytes         int64
	BytesPerSecond     float64
}

// ネットワークベンチマーカー
type NetworkBenchmarker struct {
	connections      int
	requestsPerConn  int
	concurrency      int
	target           string
	requestGenerator func() []byte
	timeout          time.Duration
	results          *BenchmarkResult
	latencies        []time.Duration
	latenciesMutex   sync.Mutex
}

// 新規ネットワークベンチマーカーの作成
func NewNetworkBenchmarker(target string, connections, requestsPerConn, concurrency int) *NetworkBenchmarker {
	return &NetworkBenchmarker{
		connections:     connections,
		requestsPerConn: requestsPerConn,
		concurrency:     concurrency,
		target:          target,
		timeout:         time.Second * 10,
		requestGenerator: func() []byte {
			return []byte("GET / HTTP/1.1\r\nHost: example.com\r\nConnection: keep-alive\r\n\r\n")
		},
	}
}

// リクエスト生成関数の設定
func (nb *NetworkBenchmarker) SetRequestGenerator(generator func() []byte) {
	nb.requestGenerator = generator
}

// タイムアウトの設定
func (nb *NetworkBenchmarker) SetTimeout(timeout time.Duration) {
	nb.timeout = timeout
}

// ベンチマークの実行
func (nb *NetworkBenchmarker) Run() (*BenchmarkResult, error) {
	results := &BenchmarkResult{
		Percentiles: make(map[float64]time.Duration),
		MinLatency:  time.Hour, // 初期値を十分大きな値に設定
	}
	
	var wg sync.WaitGroup
	semaphore := make(chan struct{}, nb.concurrency)
	
	start := time.Now()
	
	// 各コネクションに対してワーカーを起動
	for i := 0; i < nb.connections; i++ {
		wg.Add(1)
		
		go func(connID int) {
			defer wg.Done()
			
			// コネクション確立
			ctx, cancel := context.WithTimeout(context.Background(), nb.timeout)
			defer cancel()
			
			var d net.Dialer
			conn, err := d.DialContext(ctx, "tcp", nb.target)
			if err != nil {
				fmt.Printf("コネクション %d 確立エラー: %v\n", connID, err)
				atomic.AddInt64(&results.FailedRequests, int64(nb.requestsPerConn))
				atomic.AddInt64(&results.TotalRequests, int64(nb.requestsPerConn))
				return
			}
			defer conn.Close()
			
			// リクエスト送信
			for j := 0; j < nb.requestsPerConn; j++ {
				// 同時実行数の制限
				semaphore <- struct{}{}
				
				reqData := nb.requestGenerator()
				
				reqStart := time.Now()
				
				// リクエスト送信
				err := conn.SetWriteDeadline(time.Now().Add(nb.timeout))
				if err != nil {
					fmt.Printf("コネクション %d のリクエスト %d でデッドライン設定エラー: %v\n", 
						connID, j, err)
					atomic.AddInt64(&results.FailedRequests, 1)
					<-semaphore // セマフォアの解放
					continue
				}
				
				_, err = conn.Write(reqData)
				if err != nil {
					fmt.Printf("コネクション %d のリクエスト %d で送信エラー: %v\n", 
						connID, j, err)
					atomic.AddInt64(&results.FailedRequests, 1)
					<-semaphore // セマフォアの解放
					continue
				}
				
				// レスポンス受信
				err = conn.SetReadDeadline(time.Now().Add(nb.timeout))
				if err != nil {
					fmt.Printf("コネクション %d のリクエスト %d でデッドライン設定エラー: %v\n", 
						connID, j, err)
					atomic.AddInt64(&results.FailedRequests, 1)
					<-semaphore // セマフォアの解放
					continue
				}
				
				buf := make([]byte, 4096)
				n, err := conn.Read(buf)
				
				reqEnd := time.Now()
				latency := reqEnd.Sub(reqStart)
				
				// レイテンシのトラッキング
				nb.latenciesMutex.Lock()
				nb.latencies = append(nb.latencies, latency)
				nb.latenciesMutex.Unlock()
				
				// 最小レイテンシの更新
				for {
					current := results.MinLatency
					if latency >= current {
						break
					}
					if atomic.CompareAndSwapInt64((*int64)(&results.MinLatency), int64(current), int64(latency)) {
						break
					}
				}
				
				// 最大レイテンシの更新
				for {
					current := results.MaxLatency
					if latency <= current {
						break
					}
					if atomic.CompareAndSwapInt64((*int64)(&results.MaxLatency), int64(current), int64(latency)) {
						break
					}
				}
				
				if err != nil {
					if n > 0 {
						atomic.AddInt64(&results.TotalBytes, int64(n))
					}
					fmt.Printf("コネクション %d のリクエスト %d で受信エラー: %v\n", 
						connID, j, err)
					atomic.AddInt64(&results.FailedRequests, 1)
					<-semaphore // セマフォアの解放
					continue
				}
				
				atomic.AddInt64(&results.TotalBytes, int64(n))
				atomic.AddInt64(&results.SuccessfulRequests, 1)
				
				<-semaphore // セマフォアの解放
			}
			
			atomic.AddInt64(&results.TotalRequests, int64(nb.requestsPerConn))
			
		}(i)
	}
	
	// すべてのワーカーの完了を待機
	wg.Wait()
	
	// 総実行時間の計測
	results.TotalDuration = time.Since(start)
	
	// 平均レイテンシを計算
	var totalLatency time.Duration
	for _, l := range nb.latencies {
		totalLatency += l
	}
	
	if len(nb.latencies) > 0 {
		results.AvgLatency = totalLatency / time.Duration(len(nb.latencies))
	}
	
	// パーセンタイルの計算
	if len(nb.latencies) > 0 {
		// ソート
		sortDurations(nb.latencies)
		
		// 各パーセンタイルを計算
		percentiles := []float64{50, 75, 90, 95, 99, 99.9}
		for _, p := range percentiles {
			idx := int(math.Ceil(float64(len(nb.latencies))*p/100.0)) - 1
			if idx >= 0 && idx < len(nb.latencies) {
				results.Percentiles[p] = nb.latencies[idx]
			}
		}
	}
	
	// スループットを計算
	results.RequestsPerSecond = float64(results.SuccessfulRequests) / results.TotalDuration.Seconds()
	results.BytesPerSecond = float64(results.TotalBytes) / results.TotalDuration.Seconds()
	
	nb.results = results
	return results, nil
}

// ベンチマーク結果のレポート生成
func (nb *NetworkBenchmarker) GenerateReport() string {
	if nb.results == nil {
		return "ベンチマークを実行していません"
	}
	
	r := nb.results
	
	report := "ネットワークベンチマークレポート\n"
	report += "============================\n\n"
	
	report += fmt.Sprintf("ターゲット: %s\n", nb.target)
	report += fmt.Sprintf("実行設定:\n")
	report += fmt.Sprintf("  コネクション数: %d\n", nb.connections)
	report += fmt.Sprintf("  コネクションあたりのリクエスト: %d\n", nb.requestsPerConn)
	report += fmt.Sprintf("  同時実行数: %d\n", nb.concurrency)
	report += fmt.Sprintf("  タイムアウト: %s\n\n", nb.timeout)
	
	report += fmt.Sprintf("実行結果:\n")
	report += fmt.Sprintf("  総リクエスト数: %d\n", r.TotalRequests)
	report += fmt.Sprintf("  成功リクエスト数: %d (%.1f%%)\n", 
		r.SuccessfulRequests, float64(r.SuccessfulRequests)/float64(r.TotalRequests)*100)
	report += fmt.Sprintf("  失敗リクエスト数: %d (%.1f%%)\n", 
		r.FailedRequests, float64(r.FailedRequests)/float64(r.TotalRequests)*100)
	report += fmt.Sprintf("  総実行時間: %s\n\n", r.TotalDuration)
	
	report += fmt.Sprintf("レイテンシ:\n")
	report += fmt.Sprintf("  最小: %s\n", r.MinLatency)
	report += fmt.Sprintf("  最大: %s\n", r.MaxLatency)
	report += fmt.Sprintf("  平均: %s\n", r.AvgLatency)
	
	// パーセンタイルの表示
	report += fmt.Sprintf("  パーセンタイル:\n")
	for _, p := range []float64{50, 75, 90, 95, 99, 99.9} {
		if val, ok := r.Percentiles[p]; ok {
			report += fmt.Sprintf("    p%g: %s\n", p, val)
		}
	}
	
	report += fmt.Sprintf("\nスループット:\n")
	report += fmt.Sprintf("  リクエスト/秒: %.1f\n", r.RequestsPerSecond)
	report += fmt.Sprintf("  バイト/秒: %.1f (%.2f MB/s)\n", 
		r.BytesPerSecond, r.BytesPerSecond/1024/1024)
	
	return report
}

// 時間のソート（単純なバブルソート）
func sortDurations(durations []time.Duration) {
	n := len(durations)
	for i := 0; i < n-1; i++ {
		for j := 0; j < n-i-1; j++ {
			if durations[j] > durations[j+1] {
				durations[j], durations[j+1] = durations[j+1], durations[j]
			}
		}
	}
}
```

### 4.3 メモリとリソース使用量の監視

ネットワークアプリケーションがメモリやリソースをどのように使用しているかを監視するツールを実装します。

```go
// Go 1.22
package monitoring

import (
	"fmt"
	"runtime"
	"sort"
	"sync"
	"time"
)

// メモリ統計情報
type MemoryStats struct {
	Time         time.Time
	Alloc        uint64
	TotalAlloc   uint64
	Sys          uint64
	HeapAlloc    uint64
	HeapSys      uint64
	HeapIdle     uint64
	HeapInuse    uint64
	HeapReleased uint64
	HeapObjects  uint64
	StackInuse   uint64
	StackSys     uint64
	NumGC        uint32
	PauseTotalNs uint64
	NumGoRoutines int
}

// リソースモニター
type ResourceMonitor struct {
	interval    time.Duration
	samples     []MemoryStats
	samplesMu   sync.RWMutex
	stopCh      chan struct{}
	startTime   time.Time
	ticker      *time.Ticker
	running     bool
	maxSamples  int
}

// 新規リソースモニターの作成
func NewResourceMonitor(interval time.Duration, maxSamples int) *ResourceMonitor {
	if maxSamples <= 0 {
		maxSamples = 10000 // デフォルト値
	}
	
	return &ResourceMonitor{
		interval:   interval,
		samples:    make([]MemoryStats, 0, 1000),
		stopCh:     make(chan struct{}),
		startTime:  time.Time{},
		maxSamples: maxSamples,
	}
}

// モニタリングの開始
func (rm *ResourceMonitor) Start() {
	rm.samplesMu.Lock()
	defer rm.samplesMu.Unlock()
	
	if rm.running {
		return
	}
	
	rm.startTime = time.Now()
	rm.samples = rm.samples[:0] // サンプルをクリア
	rm.ticker = time.NewTicker(rm.interval)
	rm.running = true
	
	go func() {
		for {
			select {
			case <-rm.ticker.C:
				rm.collectStats()
			case <-rm.stopCh:
				return
			}
		}
	}()
}

// モニタリングの停止
func (rm *ResourceMonitor) Stop() {
	rm.samplesMu.Lock()
	defer rm.samplesMu.Unlock()
	
	if !rm.running {
		return
	}
	
	rm.ticker.Stop()
	rm.stopCh <- struct{}{}
	rm.running = false
}

// 統計情報の収集
func (rm *ResourceMonitor) collectStats() {
	var stats runtime.MemStats
	runtime.ReadMemStats(&stats)
	
	sample := MemoryStats{
		Time:          time.Now(),
		Alloc:         stats.Alloc,
		TotalAlloc:    stats.TotalAlloc,
		Sys:           stats.Sys,
		HeapAlloc:     stats.HeapAlloc,
		HeapSys:       stats.HeapSys,
		HeapIdle:      stats.HeapIdle,
		HeapInuse:     stats.HeapInuse,
		HeapReleased:  stats.HeapReleased,
		HeapObjects:   stats.HeapObjects,
		StackInuse:    stats.StackInuse,
		StackSys:      stats.StackSys,
		NumGC:         stats.NumGC,
		PauseTotalNs:  stats.PauseTotalNs,
		NumGoRoutines: runtime.NumGoroutine(),
	}
	
	rm.samplesMu.Lock()
	defer rm.samplesMu.Unlock()
	
	// 最大サンプル数を超えた場合、古いサンプルを削除
	if len(rm.samples) >= rm.maxSamples {
		// 配列の先頭を削除（効率的な方法）
		copy(rm.samples, rm.samples[1:])
		rm.samples = rm.samples[:len(rm.samples)-1]
	}
	
	rm.samples = append(rm.samples, sample)
}

// レポートの生成
func (rm *ResourceMonitor) GenerateReport() string {
	rm.samplesMu.RLock()
	samples := make([]MemoryStats, len(rm.samples))
	copy(samples, rm.samples)
	running := rm.running
	startTime := rm.startTime
	rm.samplesMu.RUnlock()
	
	if len(samples) == 0 {
		return "サンプルデータがありません"
	}
	
	// 各メトリクスの統計値を計算
	var (
		minHeapAlloc, maxHeapAlloc, avgHeapAlloc uint64
		minSysAlloc, maxSysAlloc, avgSysAlloc     uint64
		minNumGoroutines, maxNumGoroutines        int
		avgNumGoroutines                          float64
		gcCount                                    uint32
		totalGCPauseNs                             uint64
	)
	
	// 最初のサンプルで初期化
	minHeapAlloc = samples[0].HeapAlloc
	maxHeapAlloc = samples[0].HeapAlloc
	minSysAlloc = samples[0].Sys
	maxSysAlloc = samples[0].Sys
	minNumGoroutines = samples[0].NumGoRoutines
	maxNumGoroutines = samples[0].NumGoRoutines

	heapAllocs := make([]uint64, len(samples))
	sysAllocs := make([]uint64, len(samples))
	numGoroutines := make([]int, len(samples))

	var totalNumGoroutines int
	
	for i, s := range samples {
		heapAllocs[i] = s.HeapAlloc
		sysAllocs[i] = s.Sys
		numGoroutines[i] = s.NumGoRoutines
		
		totalHeapAlloc += s.HeapAlloc
		totalSysAlloc += s.Sys
		totalNumGoroutines += s.NumGoRoutines
		
		if s.HeapAlloc < minHeapAlloc {
			minHeapAlloc = s.HeapAlloc
		}
		if s.HeapAlloc > maxHeapAlloc {
			maxHeapAlloc = s.HeapAlloc
		}
		
		if s.Sys < minSysAlloc {
			minSysAlloc = s.Sys
		}
		if s.Sys > maxSysAlloc {
			maxSysAlloc = s.Sys
		}
		
		if s.NumGoRoutines < minNumGoroutines {
			minNumGoroutines = s.NumGoRoutines
		}
		if s.NumGoRoutines > maxNumGoroutines {
			maxNumGoroutines = s.NumGoRoutines
		}
	}
	
	// 最終サンプルのGC統計
	lastSample := samples[len(samples)-1]
	gcCount = lastSample.NumGC
	totalGCPauseNs = lastSample.PauseTotalNs
	
	// 平均値の計算
	avgHeapAlloc = totalHeapAlloc / uint64(len(samples))
	avgSysAlloc = totalSysAlloc / uint64(len(samples))
	avgNumGoroutines = float64(totalNumGoroutines) / float64(len(samples))
	
	// パーセンタイルの計算のためにソート
	sort.Slice(heapAllocs, func(i, j int) bool { return heapAllocs[i] < heapAllocs[j] })
	sort.Slice(sysAllocs, func(i, j int) bool { return sysAllocs[i] < sysAllocs[j] })
	sort.Slice(numGoroutines, func(i, j int) bool { return numGoroutines[i] < numGoroutines[j] })
	
	p95HeapAlloc := percentile(heapAllocs, 95)
	p99HeapAlloc := percentile(heapAllocs, 99)
	p95SysAlloc := percentile(sysAllocs, 95)
	p99SysAlloc := percentile(sysAllocs, 99)
	p95NumGoroutines := percentileInt(numGoroutines, 95)
	p99NumGoroutines := percentileInt(numGoroutines, 99)
	
	// レポート生成
	report := "リソース使用量レポート\n"
	report += "====================\n\n"
	
	if running {
		report += fmt.Sprintf("実行中: Yes\n")
	} else {
		report += fmt.Sprintf("実行中: No\n")
	}
	
	monitorDuration := samples[len(samples)-1].Time.Sub(startTime)
	report += fmt.Sprintf("モニタリング期間: %s\n", monitorDuration)
	report += fmt.Sprintf("サンプル数: %d\n", len(samples))
	report += fmt.Sprintf("サンプリング間隔: %s\n\n", rm.interval)
	
	// ヒープアロケーション
	report += "ヒープアロケーション:\n"
	report += fmt.Sprintf("  最小: %s\n", formatBytes(minHeapAlloc))
	report += fmt.Sprintf("  最大: %s\n", formatBytes(maxHeapAlloc))
	report += fmt.Sprintf("  平均: %s\n", formatBytes(avgHeapAlloc))
	report += fmt.Sprintf("  95パーセンタイル: %s\n", formatBytes(p95HeapAlloc))
	report += fmt.Sprintf("  99パーセンタイル: %s\n\n", formatBytes(p99HeapAlloc))
	
	// システムメモリ
	report += "システムメモリ使用量:\n"
	report += fmt.Sprintf("  最小: %s\n", formatBytes(minSysAlloc))
	report += fmt.Sprintf("  最大: %s\n", formatBytes(maxSysAlloc))
	report += fmt.Sprintf("  平均: %s\n", formatBytes(avgSysAlloc))
	report += fmt.Sprintf("  95パーセンタイル: %s\n", formatBytes(p95SysAlloc))
	report += fmt.Sprintf("  99パーセンタイル: %s\n\n", formatBytes(p99SysAlloc))
	
	// ゴルーチン
	report += "ゴルーチン数:\n"
	report += fmt.Sprintf("  最小: %d\n", minNumGoroutines)
	report += fmt.Sprintf("  最大: %d\n", maxNumGoroutines)
	report += fmt.Sprintf("  平均: %.1f\n", avgNumGoroutines)
	report += fmt.Sprintf("  95パーセンタイル: %d\n", p95NumGoroutines)
	report += fmt.Sprintf("  99パーセンタイル: %d\n\n", p99NumGoroutines)
	
	// GC統計
	report += "ガベージコレクション:\n"
	report += fmt.Sprintf("  GC回数: %d\n", gcCount)
	report += fmt.Sprintf("  総GCポーズ時間: %s\n", time.Duration(totalGCPauseNs))
	if monitorDuration > 0 {
		gcPerMinute := float64(gcCount) / monitorDuration.Minutes()
		report += fmt.Sprintf("  GC頻度: %.1f回/分\n", gcPerMinute)
	}
	
	return report
}

// パーセンタイルの計算（uint64用）
func percentile(values []uint64, p float64) uint64 {
	if len(values) == 0 {
		return 0
	}
	
	idx := int(float64(len(values)) * p / 100)
	if idx >= len(values) {
		idx = len(values) - 1
	}
	
	return values[idx]
}

// パーセンタイルの計算（int用）
func percentileInt(values []int, p float64) int {
	if len(values) == 0 {
		return 0
	}
	
	idx := int(float64(len(values)) * p / 100)
	if idx >= len(values) {
		idx = len(values) - 1
	}
	
	return values[idx]
}

// バイト数のフォーマット
func formatBytes(bytes uint64) string {
	const (
		_        = iota
		KB uint64 = 1 << (10 * iota)
		MB
		GB
	)
	
	unit := ""
	value := float64(bytes)
	
	switch {
	case bytes >= GB:
		unit = "GB"
		value = float64(bytes) / float64(GB)
	case bytes >= MB:
		unit = "MB"
		value = float64(bytes) / float64(MB)
	case bytes >= KB:
		unit = "KB"
		value = float64(bytes) / float64(KB)
	default:
		unit = "B"
	}
	
	return fmt.Sprintf("%.2f %s", value, unit)
}
```

### 4.4 コネクションプールとスループットの最適化

ネットワークアプリケーションのパフォーマンスを向上させるためのコネクションプール実装とその調整方法を提供します。

```go
// Go 1.22
package performance

import (
	"context"
	"errors"
	"fmt"
	"net"
	"sync"
	"sync/atomic"
	"time"
)

// コネクションの状態
type connectionState int

const (
	connectionIdle connectionState = iota
	connectionInUse
	connectionClosed
)

// プールされるコネクション
type pooledConnection struct {
	conn      net.Conn
	createdAt time.Time
	lastUsed  time.Time
	state     connectionState
	id        uint64
}

// コネクションプールの設定
type ConnectionPoolConfig struct {
	// 初期コネクション数
	InitialSize int
	
	// 最大コネクション数
	MaxSize int
	
	// アイドル状態のコネクション最大数
	MaxIdle int
	
	// コネクションのタイムアウト
	DialTimeout time.Duration
	
	// コネクション最大寿命
	MaxLifetime time.Duration
	
	// アイドルタイムアウト（この時間以上使われていないコネクションは閉じる）
	IdleTimeout time.Duration
	
	// ヘルスチェックの間隔
	HealthCheckInterval time.Duration
	
	// ヘルスチェック用の関数
	HealthCheck func(net.Conn) bool
}

// デフォルト設定
func DefaultConnectionPoolConfig() ConnectionPoolConfig {
	return ConnectionPoolConfig{
		InitialSize:         5,
		MaxSize:             100,
		MaxIdle:             20,
		DialTimeout:         time.Second * 5,
		MaxLifetime:         time.Hour * 1,
		IdleTimeout:         time.Minute * 10,
		HealthCheckInterval: time.Minute * 1,
		HealthCheck: func(conn net.Conn) bool {
			// デフォルトのヘルスチェックはTCPコネクションの状態を確認
			if tcpConn, ok := conn.(*net.TCPConn); ok {
				// 1バイト書き込みを試みる
				originalDeadline, err := tcpConn.GetWriteDeadline()
				if err != nil {
					return false
				}
				
				err = tcpConn.SetWriteDeadline(time.Now().Add(time.Second))
				if err != nil {
					return false
				}
				
				_, err = tcpConn.Write([]byte{0})
				
				// 元のデッドラインに戻す
				tcpConn.SetWriteDeadline(originalDeadline)
				
				return err == nil
			}
			return true // TCP以外のコネクションは常に健全と見なす
		},
	}
}

// コネクションプール
type ConnectionPool struct {
	config ConnectionPoolConfig
	
	// ネットワークとアドレス
	network string
	address string
	
	// アクティブコネクション数
	activeCount int32
	
	// 次のコネクションID
	nextConnID uint64
	
	// アイドルコネクションのリスト
	idleConns []*pooledConnection
	
	// 内部状態の保護用ミューテックス
	mu sync.Mutex
	
	// ダイアル関数
	dialFunc func(context.Context) (net.Conn, error)
	
	// プールのクローズ状態
	closed bool
	
	// リソース管理用WaitGroup
	wg sync.WaitGroup
	
	// ヘルスチェックとクリーンアップ用の停止チャネル
	stopCh chan struct{}
	
	// メトリクス
	metrics struct {
		created        int64
		closed         int64
		reused         int64
		idleTimeout    int64
		maxLifetime    int64
		healthCheckFail int64
		failedDials    int64
		acquireWaits   int64
	}
}

// エラー定義
var (
	ErrPoolClosed      = errors.New("connection pool is closed")
	ErrPoolMaxSize     = errors.New("connection pool reached max size")
	ErrAcquireTimeout  = errors.New("timeout while acquiring connection")
	ErrDialFailed      = errors.New("failed to dial new connection")
	ErrConnectionClosed = errors.New("connection is closed")
)

// 新規コネクションプールの作成
func NewConnectionPool(network, address string, config ConnectionPoolConfig) (*ConnectionPool, error) {
	if config.MaxSize <= 0 {
		return nil, errors.New("MaxSize must be greater than 0")
	}
	
	if config.MaxIdle > config.MaxSize {
		config.MaxIdle = config.MaxSize
	}
	
	pool := &ConnectionPool{
		config:  config,
		network: network,
		address: address,
		idleConns: make([]*pooledConnection, 0, config.MaxIdle),
		stopCh:  make(chan struct{}),
		dialFunc: func(ctx context.Context) (net.Conn, error) {
			dialer := &net.Dialer{Timeout: config.DialTimeout}
			return dialer.DialContext(ctx, network, address)
		},
	}
	
	// 初期コネクションの確立
	for i := 0; i < config.InitialSize; i++ {
		conn, err := pool.createConnection(context.Background())
		if err != nil {
			// 初期化中にエラーが発生した場合、すでに作成されたコネクションをクローズ
			pool.Close()
			return nil, fmt.Errorf("failed to create initial connection: %w", err)
		}
		
		pool.idleConns = append(pool.idleConns, conn)
	}
	
	// バックグラウンドのヘルスチェックとクリーンアップを開始
	pool.wg.Add(1)
	go pool.periodicCleanup()
	
	return pool, nil
}

// コネクションの作成
func (p *ConnectionPool) createConnection(ctx context.Context) (*pooledConnection, error) {
	conn, err := p.dialFunc(ctx)
	if err != nil {
		atomic.AddInt64(&p.metrics.failedDials, 1)
		return nil, fmt.Errorf("%w: %v", ErrDialFailed, err)
	}
	
	atomic.AddInt64(&p.metrics.created, 1)
	atomic.AddInt32(&p.activeCount, 1)
	
	return &pooledConnection{
		conn:      conn,
		createdAt: time.Now(),
		lastUsed:  time.Now(),
		state:     connectionIdle,
		id:        atomic.AddUint64(&p.nextConnID, 1) - 1,
	}, nil
}

// コネクションの取得
func (p *ConnectionPool) Acquire(ctx context.Context) (net.Conn, error) {
	// タイムアウトまたはキャンセル検出用
	var timer *time.Timer
	var timerCh <-chan time.Time
	
	deadline, hasDeadline := ctx.Deadline()
	if hasDeadline {
		timer = time.NewTimer(time.Until(deadline))
		timerCh = timer.C
	}
	
	for {
		// クローズされたプールからは取得不可
		if p.closed {
			if timer != nil {
				timer.Stop()
			}
			return nil, ErrPoolClosed
		}
		
		// アイドルコネクションの取得を試みる
		poolConn, err := p.getIdleConnection()
		if err == nil {
			if timer != nil {
				timer.Stop()
			}
			return &PooledConn{pool: p, poolConn: poolConn}, nil
		}
		
		// まだ最大数に達していなければ新しいコネクションを作成
		currentCount := atomic.LoadInt32(&p.activeCount)
		if currentCount < int32(p.config.MaxSize) {
			if atomic.CompareAndSwapInt32(&p.activeCount, currentCount, currentCount+1) {
				poolConn, err := p.createConnection(ctx)
				if err != nil {
					// 作成に失敗した場合、カウントを戻す
					atomic.AddInt32(&p.activeCount, -1)
					if timer != nil {
						timer.Stop()
					}
					return nil, err
				}
				
				poolConn.state = connectionInUse
				return &PooledConn{pool: p, poolConn: poolConn}, nil
			}
			// CASが失敗した場合は再試行
			continue
		}
		
		// 最大数に達している場合、タイムアウトかキャンセルを待つ
		atomic.AddInt64(&p.metrics.acquireWaits, 1)
		
		select {
		case <-ctx.Done():
			if timer != nil {
				timer.Stop()
			}
			return nil, ctx.Err()
		case <-timerCh:
			return nil, ErrAcquireTimeout
		case <-time.After(time.Millisecond * 100):
			// 短い待機後に再試行
			continue
		}
	}
}

// アイドルコネクションの取得
func (p *ConnectionPool) getIdleConnection() (*pooledConnection, error) {
	p.mu.Lock()
	defer p.mu.Unlock()
	
	if len(p.idleConns) == 0 {
		return nil, errors.New("no idle connections")
	}
	
	// 最後のコネクションを取得（LIFO）
	lastIdx := len(p.idleConns) - 1
	poolConn := p.idleConns[lastIdx]
	p.idleConns = p.idleConns[:lastIdx]
	
	// コネクションの状態を「使用中」に変更
	poolConn.state = connectionInUse
	poolConn.lastUsed = time.Now()
	
	atomic.AddInt64(&p.metrics.reused, 1)
	return poolConn, nil
}

// コネクションのリリース
func (p *ConnectionPool) release(poolConn *pooledConnection) error {
	p.mu.Lock()
	defer p.mu.Unlock()
	
	if p.closed {
		// プールがクローズされている場合は接続を閉じる
		poolConn.state = connectionClosed
		atomic.AddInt32(&p.activeCount, -1)
		atomic.AddInt64(&p.metrics.closed, 1)
		return poolConn.conn.Close()
	}
	
	// 最大寿命チェック
	if p.config.MaxLifetime > 0 && time.Since(poolConn.createdAt) > p.config.MaxLifetime {
		poolConn.state = connectionClosed
		atomic.AddInt32(&p.activeCount, -1)
		atomic.AddInt64(&p.metrics.maxLifetime, 1)
		return poolConn.conn.Close()
	}
	
	// アイドルコネクション最大数チェック
	if len(p.idleConns) >= p.config.MaxIdle {
		poolConn.state = connectionClosed
		atomic.AddInt32(&p.activeCount, -1)
		atomic.AddInt64(&p.metrics.closed, 1)
		return poolConn.conn.Close()
	}
	
	// 健全性チェック
	if p.config.HealthCheck != nil && !p.config.HealthCheck(poolConn.conn) {
		poolConn.state = connectionClosed
		atomic.AddInt32(&p.activeCount, -1)
		atomic.AddInt64(&p.metrics.healthCheckFail, 1)
		return poolConn.conn.Close()
	}
	
	// コネクションをアイドル状態に戻す
	poolConn.state = connectionIdle
	poolConn.lastUsed = time.Now()
	p.idleConns = append(p.idleConns, poolConn)
	
	return nil
}

// 定期的なクリーンアップ処理
func (p *ConnectionPool) periodicCleanup() {
	defer p.wg.Done()
	
	ticker := time.NewTicker(p.config.HealthCheckInterval)
	defer ticker.Stop()
	
	for {
		select {
		case <-ticker.C:
			p.cleanup()
		case <-p.stopCh:
			return
		}
	}
}

// クリーンアップ処理
func (p *ConnectionPool) cleanup() {
	p.mu.Lock()
	defer p.mu.Unlock()
	
	if p.closed {
		return
	}
	
	var remainingConns []*pooledConnection
	now := time.Now()
	
	for _, poolConn := range p.idleConns {
		// アイドルタイムアウトの確認
		if p.config.IdleTimeout > 0 && now.Sub(poolConn.lastUsed) > p.config.IdleTimeout {
			poolConn.state = connectionClosed
			atomic.AddInt32(&p.activeCount, -1)
			atomic.AddInt64(&p.metrics.idleTimeout, 1)
			poolConn.conn.Close()
			continue
		}
		
		// 最大寿命の確認
		if p.config.MaxLifetime > 0 && now.Sub(poolConn.createdAt) > p.config.MaxLifetime {
			poolConn.state = connectionClosed
			atomic.AddInt32(&p.activeCount, -1)
			atomic.AddInt64(&p.metrics.maxLifetime, 1)
			poolConn.conn.Close()
			continue
		}
		
		// ヘルスチェック
		if p.config.HealthCheck != nil && !p.config.HealthCheck(poolConn.conn) {
			poolConn.state = connectionClosed
			atomic.AddInt32(&p.activeCount, -1)
			atomic.AddInt64(&p.metrics.healthCheckFail, 1)
			poolConn.conn.Close()
			continue
		}
		
		remainingConns = append(remainingConns, poolConn)
	}
	
	// 有効なコネクションのみを保持
	p.idleConns = remainingConns
}

// プールのクローズ
func (p *ConnectionPool) Close() error {
	p.mu.Lock()
	
	if p.closed {
		p.mu.Unlock()
		return ErrPoolClosed
	}
	
	p.closed = true
	close(p.stopCh)
	
	// アイドルコネクションを閉じる
	for _, poolConn := range p.idleConns {
		poolConn.state = connectionClosed
		poolConn.conn.Close()
		atomic.AddInt32(&p.activeCount, -1)
		atomic.AddInt64(&p.metrics.closed, 1)
	}
	
	p.idleConns = nil
	p.mu.Unlock()
	
	// クリーンアップゴルーチンの終了を待機
	p.wg.Wait()
	
	return nil
}

// プールのメトリクス取得
func (p *ConnectionPool) GetMetrics() map[string]int64 {
	return map[string]int64{
		"active_connections":    int64(atomic.LoadInt32(&p.activeCount)),
		"created_connections":   atomic.LoadInt64(&p.metrics.created),
		"closed_connections":    atomic.LoadInt64(&p.metrics.closed),
		"reused_connections":    atomic.LoadInt64(&p.metrics.reused),
		"idle_timeout_closes":   atomic.LoadInt64(&p.metrics.idleTimeout),
		"max_lifetime_closes":   atomic.LoadInt64(&p.metrics.maxLifetime),
		"health_check_fails":    atomic.LoadInt64(&p.metrics.healthCheckFail),
		"failed_dials":          atomic.LoadInt64(&p.metrics.failedDials),
		"acquire_waits":         atomic.LoadInt64(&p.metrics.acquireWaits),
	}
}
```

### 5.1 アプリケーションメトリクスの収集と分析

ネットワークアプリケーションの状態を監視するためのメトリクス収集と分析システムを実装します。

```go
// Go 1.22
package monitoring

import (
	"encoding/json"
	"fmt"
	"net/http"
	"sync"
	"time"
)

// メトリクスの種類
type MetricType string

const (
	CounterMetric  MetricType = "counter"  // 単調増加するカウンター
	GaugeMetric    MetricType = "gauge"    // 任意に変化する値
	HistogramMetric MetricType = "histogram" // 値の分布
)

// メトリクス値
type MetricValue struct {
	Type        MetricType  `json:"type"`
	Name        string      `json:"name"`
	Description string      `json:"description"`
	Labels      []string    `json:"labels,omitempty"`
	Value       interface{} `json:"value"`
	Timestamp   time.Time   `json:"timestamp"`
}

// ヒストグラムデータ
type HistogramData struct {
	Buckets     map[float64]uint64 `json:"buckets"` // バケットの上限と対応するカウント
	Count       uint64             `json:"count"`   // 総サンプル数
	Sum         float64            `json:"sum"`     // 合計値
	Min         float64            `json:"min"`     // 最小値
	Max         float64            `json:"max"`     // 最大値
}

// メトリクスレジストリ
type MetricsRegistry struct {
	metrics map[string]*MetricValue
	mutex   sync.RWMutex
}

// 新規メトリクスレジストリの作成
func NewMetricsRegistry() *MetricsRegistry {
	return &MetricsRegistry{
		metrics: make(map[string]*MetricValue),
	}
}

// カウンターの登録
func (mr *MetricsRegistry) RegisterCounter(name, description string, labels ...string) {
	mr.mutex.Lock()
	defer mr.mutex.Unlock()
	
	mr.metrics[name] = &MetricValue{
		Type:        CounterMetric,
		Name:        name,
		Description: description,
		Labels:      labels,
		Value:       uint64(0),
		Timestamp:   time.Now(),
	}
}

// ゲージの登録
func (mr *MetricsRegistry) RegisterGauge(name, description string, labels ...string) {
	mr.mutex.Lock()
	defer mr.mutex.Unlock()
	
	mr.metrics[name] = &MetricValue{
		Type:        GaugeMetric,
		Name:        name,
		Description: description,
		Labels:      labels,
		Value:       float64(0),
		Timestamp:   time.Now(),
	}
}

// ヒストグラムの登録
func (mr *MetricsRegistry) RegisterHistogram(name, description string, buckets []float64, labels ...string) {
	mr.mutex.Lock()
	defer mr.mutex.Unlock()
	
	histogramData := HistogramData{
		Buckets: make(map[float64]uint64),
		Count:   0,
		Sum:     0,
		Min:     float64(0),
		Max:     float64(0),
	}
	
	// バケットの初期化
	for _, b := range buckets {
		histogramData.Buckets[b] = 0
	}
	
	mr.metrics[name] = &MetricValue{
		Type:        HistogramMetric,
		Name:        name,
		Description: description,
		Labels:      labels,
		Value:       histogramData,
		Timestamp:   time.Now(),
	}
}

// カウンターのインクリメント
func (mr *MetricsRegistry) IncrementCounter(name string, value uint64) {
	mr.mutex.Lock()
	defer mr.mutex.Unlock()
	
	if metric, exists := mr.metrics[name]; exists && metric.Type == CounterMetric {
		currentValue := metric.Value.(uint64)
		metric.Value = currentValue + value
		metric.Timestamp = time.Now()
	}
}

// ゲージの設定
func (mr *MetricsRegistry) SetGauge(name string, value float64) {
	mr.mutex.Lock()
	defer mr.mutex.Unlock()
	
	if metric, exists := mr.metrics[name]; exists && metric.Type == GaugeMetric {
		metric.Value = value
		metric.Timestamp = time.Now()
	}
}

// ヒストグラムへの値の追加
func (mr *MetricsRegistry) ObserveHistogram(name string, value float64) {
	mr.mutex.Lock()
	defer mr.mutex.Unlock()
	
	if metric, exists := mr.metrics[name]; exists && metric.Type == HistogramMetric {
		histData := metric.Value.(HistogramData)
		
		// カウントと合計の更新
		histData.Count++
		histData.Sum += value
		
		// 最小値と最大値の更新
		if histData.Count == 1 || value < histData.Min {
			histData.Min = value
		}
		if histData.Count == 1 || value > histData.Max {
			histData.Max = value
		}
		
		// バケットの更新
		for bucketLimit := range histData.Buckets {
			if value <= bucketLimit {
				histData.Buckets[bucketLimit]++
			}
		}
		
		metric.Value = histData
		metric.Timestamp = time.Now()
	}
}

// メトリクスの取得
func (mr *MetricsRegistry) GetMetric(name string) (*MetricValue, bool) {
	mr.mutex.RLock()
	defer mr.mutex.RUnlock()
	
	metric, exists := mr.metrics[name]
	if !exists {
		return nil, false
	}
	
	// コピーを返す
	metricCopy := *metric
	return &metricCopy, true
}

// すべてのメトリクスの取得
func (mr *MetricsRegistry) GetAllMetrics() []*MetricValue {
	mr.mutex.RLock()
	defer mr.mutex.RUnlock()
	
	metrics := make([]*MetricValue, 0, len(mr.metrics))
	for _, metric := range mr.metrics {
		metricCopy := *metric
		metrics = append(metrics, &metricCopy)
	}
	
	return metrics
}

// HTTPハンドラーの取得（メトリクスエンドポイント用）
func (mr *MetricsRegistry) Handler() http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		metrics := mr.GetAllMetrics()
		
		w.Header().Set("Content-Type", "application/json")
		json.NewEncoder(w).Encode(metrics)
	})
```

```go
// スパンのフェーズ
type SpanPhase string

const (
	PhaseStart  SpanPhase = "start"
	PhaseEnd    SpanPhase = "end"
	PhaseError  SpanPhase = "error"
	PhaseInfo   SpanPhase = "info"
	PhaseCancel SpanPhase = "cancel"
)

// スパンイベント
type SpanEvent struct {
	Timestamp time.Time  `json:"timestamp"`
	Phase     SpanPhase  `json:"phase"`
	Message   string     `json:"message,omitempty"`
	Fields    []KeyValue `json:"fields,omitempty"`
}

	ParentSpanID string       `json:"parent_span_id,omitempty"`
	Operation    string       `json:"operation"`

	EndTime      time.Time    `json:"end_time,omitempty"`

	serviceName string

}

// サンプラー

	ShouldSample(traceID string) bool
}

// 常にサンプリングするサンプラー
type AlwaysSampleSampler struct{}

func (s *AlwaysSampleSampler) ShouldSample(traceID string) bool {
	return true
}

// 確率的サンプラー
type ProbabilisticSampler struct {
	SamplingRate float64 // 0.0〜1.0
}

func (s *ProbabilisticSampler) ShouldSample(traceID string) bool {
	if s.SamplingRate >= 1.0 {
		return true
	}
	if s.SamplingRate <= 0.0 {
		return false
	}
	
	// トレースIDを使用して決定論的にサンプリング
	if len(traceID) < 2 {
		return rand.Float64() < s.SamplingRate
	}
	
	// トレースIDの最初の2バイトを数値に変換
	bytes, err := hex.DecodeString(traceID[:4])
	if err != nil || len(bytes) < 2 {
		return rand.Float64() < s.SamplingRate
	}
	
	value := float64(uint16(bytes[0])<<8|uint16(bytes[1])) / 65535.0
	return value < s.SamplingRate
}

// コンソールエクスポーター
type ConsoleExporter struct{}

func (e *ConsoleExporter) ExportSpan(span *Span) {

	fmt.Printf("  TraceID: %s, SpanID: %s, ParentSpanID: %s\n", 
		span.TraceID, span.SpanID, span.ParentSpanID)
	fmt.Printf("  開始: %s, 終了: %s, 所要時間: %s\n", 
		span.StartTime.Format(time.RFC3339Nano),
		span.EndTime.Format(time.RFC3339Nano),
		span.Duration)
	
	if span.Error {
		fmt.Printf("  エラー: %s\n", span.ErrorMessage)
	}
	
	if len(span.Tags) > 0 {
		fmt.Println("  タグ:")
		for _, tag := range span.Tags {
			fmt.Printf("    %s: %v\n", tag.Key, tag.Value)
		}
	}
	
	if len(span.Events) > 0 {
		fmt.Println("  イベント:")
		for _, event := range span.Events {
			fmt.Printf("    [%s] %s: %s\n",
				event.Timestamp.Format(time.RFC3339Nano),
				event.Phase, event.Message)
			
			if len(event.Fields) > 0 {
				for _, field := range event.Fields {
					fmt.Printf("      %s: %v\n", field.Key, field.Value)
				}
			}
		}

	fmt.Println()

// 新規トレーサーの作成
func NewTracer(serviceName string) *Tracer {
	return &Tracer{
		spans:       make([]*Span, 0),
		exporters:   make([]SpanExporter, 0),
		sampler:     &AlwaysSampleSampler{},
		serviceName: serviceName,
	}
}

// エクスポーターの追加
func (t *Tracer) AddExporter(exporter SpanExporter) {
	t.exporters = append(t.exporters, exporter)
}

// サンプラーの設定
func (t *Tracer) SetSampler(sampler Sampler) {
	t.sampler = sampler
}

// 新規トレースの開始
func (t *Tracer) StartTrace(ctx context.Context, operation string) (context.Context, *Span) {
	traceID := generateID()
	return t.StartSpanWithTraceID(ctx, operation, traceID, "")
}

// 新規スパンの開始（トレースIDとオプションの親スパンID指定）
func (t *Tracer) StartSpanWithTraceID(ctx context.Context, operation, traceID, parentSpanID string) (context.Context, *Span) {
	if traceID == "" {
		traceID = generateID()
	}
	
	spanID := generateID()
	
	// コンテキストから親スパンを取得
	if parent := SpanFromContext(ctx); parent != nil && parentSpanID == "" {
		parentSpanID = parent.SpanID
		traceID = parent.TraceID
	}
	
	// サンプリング判定
	if !t.sampler.ShouldSample(traceID) {
		// サンプリングしない場合はNoOpSpanを返す
		return ctx, nil
	}
	
	span := &Span{
		TraceID:      traceID,
		SpanID:       spanID,
		ParentSpanID: parentSpanID,
		Operation:    operation,
		StartTime:    time.Now(),
		Events:       make([]SpanEvent, 0),
		Tags:         make([]KeyValue, 0),
	}
	
	// サービス名のタグを追加
	span.SetTag("service.name", t.serviceName)
	
	// 開始イベントを記録
	span.AddEvent(PhaseStart, fmt.Sprintf("開始: %s", operation))
	
	// スパンをリストに追加
	t.spansMutex.Lock()
	t.spans = append(t.spans, span)
	t.spansMutex.Unlock()
	
	// 新しいコンテキストを作成
	newCtx := context.WithValue(ctx, traceContextKey{}, span)
	
	return newCtx, span
}

// コンテキストからスパンを取得
func SpanFromContext(ctx context.Context) *Span {
	if span, ok := ctx.Value(traceContextKey{}).(*Span); ok {
		return span
	}
	return nil
}

// スパンの終了
func (s *Span) Finish() {
	s.mutex.Lock()
	defer s.mutex.Unlock()
	
	if s.finished {
		return
	}
	
	s.EndTime = time.Now()
	s.Duration = s.EndTime.Sub(s.StartTime)
	s.finished = true
	
	// 終了イベントを記録
	s.addEventLocked(PhaseEnd, fmt.Sprintf("終了: %s", s.Operation))
}

// エラーの記録
func (s *Span) RecordError(err error) {
	if err == nil {
		return
	}
	
	s.mutex.Lock()
	defer s.mutex.Unlock()
	
	s.Error = true
	s.ErrorMessage = err.Error()
	
	// エラーイベントを記録
	s.addEventLocked(PhaseError, fmt.Sprintf("エラー: %s", err.Error()))
}

// イベントの追加
func (s *Span) AddEvent(phase SpanPhase, message string, fields ...KeyValue) {
	s.mutex.Lock()
	defer s.mutex.Unlock()
	
	s.addEventLocked(phase, message, fields...)
}

// イベントの追加（ロック済み）
func (s *Span) addEventLocked(phase SpanPhase, message string, fields ...KeyValue) {
	event := SpanEvent{
		Timestamp: time.Now(),
		Phase:     phase,
		Message:   message,
		Fields:    fields,
	}
	
	s.Events = append(s.Events, event)
}

// タグの設定
func (s *Span) SetTag(key string, value interface{}) {
	s.mutex.Lock()
	defer s.mutex.Unlock()
	
	// 既存のタグを更新または新規タグを追加
	for i, tag := range s.Tags {
		if tag.Key == key {
			s.Tags[i].Value = value
			return
		}
	}
	
	s.Tags = append(s.Tags, KeyValue{Key: key, Value: value})
}

// ロギング
func (s *Span) Log(message string, fields ...KeyValue) {
	s.AddEvent(PhaseInfo, message, fields...)
}

// トレースIDの生成
func generateID() string {
	randomBytes := make([]byte, 8)
	rand.Read(randomBytes)
	return hex.EncodeToString(randomBytes)
}

// HTTPリクエストにトレースを挿入
func InjectTraceToHTTPRequest(span *Span, req *http.Request) {
	if span == nil {
		return
	}
	
	req.Header.Set("X-Trace-ID", span.TraceID)
	req.Header.Set("X-Span-ID", span.SpanID)
	req.Header.Set("X-Parent-Span-ID", span.ParentSpanID)
}

// HTTPリクエストからトレースを抽出
func ExtractTraceFromHTTPRequest(req *http.Request) (string, string, string) {
	traceID := req.Header.Get("X-Trace-ID")
	spanID := req.Header.Get("X-Span-ID")
	parentSpanID := req.Header.Get("X-Parent-Span-ID")
	
	return traceID, spanID, parentSpanID
}

// HTTPハンドラーのトレーシングミドルウェア
func TracingMiddleware(tracer *Tracer) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			// リクエストからトレース情報を抽出
			traceID, _, parentSpanID := ExtractTraceFromHTTPRequest(r)
			
			// 新しいスパンを開始
			operation := fmt.Sprintf("HTTP %s %s", r.Method, r.URL.Path)
			ctx, span := tracer.StartSpanWithTraceID(r.Context(), operation, traceID, parentSpanID)
			
			// リクエスト情報をタグとして設定
			span.SetTag("http.method", r.Method)
			span.SetTag("http.url", r.URL.String())
			span.SetTag("http.remote_addr", r.RemoteAddr)
			span.SetTag("http.user_agent", r.UserAgent())
			
			// レスポンスライターをラップして状態コードを取得
			wrapped := wrapResponseWriter(w)
			
			// リクエスト処理
			next.ServeHTTP(wrapped, r.WithContext(ctx))
			
			// スパンを完了
			span.SetTag("http.status_code", wrapped.statusCode)
			
			// エラーステータスコードをエラーとして記録
			if wrapped.statusCode >= 400 {
				span.RecordError(fmt.Errorf("HTTPステータスコード: %d", wrapped.statusCode))
			}
			
			span.Finish()
		})
	}
}

// レスポンスライターのラッパー
type wrappedResponseWriter struct {
	http.ResponseWriter
	statusCode int
}

func wrapResponseWriter(w http.ResponseWriter) *wrappedResponseWriter {
	return &wrappedResponseWriter{w, http.StatusOK}
}

func (w *wrappedResponseWriter) WriteHeader(code int) {
	w.statusCode = code
	w.ResponseWriter.WriteHeader(code)
}
```

```go
// Go 1.22
package alerting

import (
	"context"
	"fmt"

// アラートの重要度レベル
type AlertSeverity int

const (
	SeverityInfo AlertSeverity = iota

	SeverityError
	SeverityCritical
)

// 重要度の文字列表現
func (a AlertSeverity) String() string {

		return "INFO"

// アラートの状態
type AlertState int

const (

	StateResolved

// 状態の文字列表現
func (a AlertState) String() string {
	switch a {
	case StateActive:
		return "ACTIVE"
	case StateAcknowledged:
		return "ACKNOWLEDGED"
	case StateResolved:
		return "RESOLVED"

		return "UNKNOWN"

// アラート
type Alert struct {
	ID          string        `json:"id"`
	Name        string        `json:"name"`
	Description string        `json:"description"`
	Severity    AlertSeverity `json:"severity"`
	State       AlertState    `json:"state"`
	Source      string        `json:"source"`
	CreatedAt   time.Time     `json:"created_at"`
	UpdatedAt   time.Time     `json:"updated_at"`
	ResolvedAt  time.Time     `json:"resolved_at,omitempty"`
	Labels      map[string]string `json:"labels,omitempty"`
	Annotations map[string]string `json:"annotations,omitempty"`
}

// アラートハンドラー
type AlertHandler interface {
	HandleAlert(alert *Alert)
}

// アラートルール
type AlertRule interface {
	// ルールの評価
	Evaluate(ctx context.Context) ([]*Alert, error)
	
	// ルールの名前
	Name() string
	
	// ルールの説明
	Description() string
}

// アラートマネージャー
type AlertManager struct {
	rules     []AlertRule
	handlers  []AlertHandler
	alerts    map[string]*Alert
	alertsMu  sync.RWMutex
	interval  time.Duration
	ctx       context.Context
	cancel    context.CancelFunc
	wg        sync.WaitGroup
}

// 新規アラートマネージャーの作成
func NewAlertManager(interval time.Duration) *AlertManager {
	ctx, cancel := context.WithCancel(context.Background())
	
	return &AlertManager{
		rules:    make([]AlertRule, 0),
		handlers: make([]AlertHandler, 0),
		alerts:   make(map[string]*Alert),
		interval: interval,
		ctx:      ctx,
		cancel:   cancel,
	}
}

// アラートルールの追加
func (am *AlertManager) AddRule(rule AlertRule) {
	am.rules = append(am.rules, rule)
}

// アラートハンドラーの追加
func (am *AlertManager) AddHandler(handler AlertHandler) {
	am.handlers = append(am.handlers, handler)
}

// モニタリングの開始
func (am *AlertManager) Start() {
	am.wg.Add(1)
	go am.run()
	log.Println("アラートマネージャーを開始しました")
}

// モニタリングの停止
func (am *AlertManager) Stop() {
	am.cancel()
	am.wg.Wait()
	log.Println("アラートマネージャーを停止しました")
}

// モニタリングループ
func (am *AlertManager) run() {
	defer am.wg.Done()
	
	ticker := time.NewTicker(am.interval)
	defer ticker.Stop()
	
	for {
		select {
		case <-am.ctx.Done():
			return
		case <-ticker.C:
			am.evaluateRules()
		}
	}
}

// ルールの評価
func (am *AlertManager) evaluateRules() {
	for _, rule := range am.rules {
		alerts, err := rule.Evaluate(am.ctx)
		if err != nil {
			log.Printf("ルール '%s' の評価中にエラーが発生しました: %v", rule.Name(), err)
			continue
		}
		
		for _, alert := range alerts {
			am.processAlert(alert)
		}
	}
}

// アラートの処理
func (am *AlertManager) processAlert(alert *Alert) {
	am.alertsMu.Lock()
	defer am.alertsMu.Unlock()
	
	// 既存のアラートを確認
	existingAlert, exists := am.alerts[alert.ID]
	
	if !exists {
		// 新しいアラート
		am.alerts[alert.ID] = alert
		am.notifyHandlers(alert)
	} else if existingAlert.State == StateResolved && alert.State == StateActive {
		// 解決済みのアラートが再び発生
		existingAlert.State = StateActive
		existingAlert.UpdatedAt = time.Now()
		existingAlert.ResolvedAt = time.Time{}
		am.notifyHandlers(existingAlert)
	} else if existingAlert.State != alert.State {
		// 状態の変更
		existingAlert.State = alert.State
		existingAlert.UpdatedAt = time.Now()
		
		if alert.State == StateResolved {
			existingAlert.ResolvedAt = time.Now()
		}
		
		am.notifyHandlers(existingAlert)
	}
}

// すべてのハンドラーに通知
func (am *AlertManager) notifyHandlers(alert *Alert) {
	for _, handler := range am.handlers {
		go handler.HandleAlert(alert)
	}
}

// アラートの承認
func (am *AlertManager) AcknowledgeAlert(id string) error {
	am.alertsMu.Lock()
	defer am.alertsMu.Unlock()
	
	alert, exists := am.alerts[id]
	if !exists {
		return fmt.Errorf("アラート ID '%s' が見つかりません", id)
	}
	
	if alert.State == StateActive {
		alert.State = StateAcknowledged
		alert.UpdatedAt = time.Now()
		am.notifyHandlers(alert)
	}
	
	return nil
}

// アラートの解決
func (am *AlertManager) ResolveAlert(id string) error {
	am.alertsMu.Lock()
	defer am.alertsMu.Unlock()
	
	alert, exists := am.alerts[id]
	if !exists {
		return fmt.Errorf("アラート ID '%s' が見つかりません", id)
	}
	
	if alert.State != StateResolved {
		alert.State = StateResolved
		alert.UpdatedAt = time.Now()
		alert.ResolvedAt = time.Now()
		am.notifyHandlers(alert)
	}
	
	return nil
}

// アクティブなアラートの取得
func (am *AlertManager) GetActiveAlerts() []*Alert {
	am.alertsMu.RLock()
	defer am.alertsMu.RUnlock()
	
	var activeAlerts []*Alert
	for _, alert := range am.alerts {
		if alert.State == StateActive || alert.State == StateAcknowledged {
			alertCopy := *alert
			activeAlerts = append(activeAlerts, &alertCopy)
		}
	}
	
	return activeAlerts
}

// すべてのアラートの取得
func (am *AlertManager) GetAllAlerts() []*Alert {
	am.alertsMu.RLock()
	defer am.alertsMu.RUnlock()
	
	var allAlerts []*Alert
	for _, alert := range am.alerts {
		alertCopy := *alert
		allAlerts = append(allAlerts, &alertCopy)
	}
	
	return allAlerts
}

	return r.description
}

func (r *ThresholdAlertRule) Evaluate(ctx context.Context) ([]*Alert, error) {
	value, err := r.metricFn(ctx)
	if err != nil {
		return nil, fmt.Errorf("メトリクス取得エラー: %w", err)
	}
	
	var triggered bool
	switch r.operator {
	case "gt":
		triggered = value > r.threshold
	case "lt":
		triggered = value < r.threshold
	case "eq":
		triggered = value == r.threshold
	case "ge":
		triggered = value >= r.threshold
	case "le":
		triggered = value <= r.threshold
	default:
		return nil, fmt.Errorf("不明な演算子: %s", r.operator)
	}
	
	if triggered {
		alertID := fmt.Sprintf("%s_%s_%f", r.name, r.operator, r.threshold)
		alert := &Alert{
			ID:          alertID,
			Name:        r.name,
			Description: fmt.Sprintf("%s: 現在値 %.2f %s しきい値 %.2f", 
				r.description, value, operatorToSymbol(r.operator), r.threshold),
			Severity:    r.severity,
			State:       StateActive,
			Source:      r.source,
			CreatedAt:   time.Now(),
			UpdatedAt:   time.Now(),
			Labels:      r.labels,
			Annotations: map[string]string{
				"current_value": fmt.Sprintf("%.2f", value),
				"threshold":     fmt.Sprintf("%.2f", r.threshold),
				"operator":      r.operator,
			},
		}
		return []*Alert{alert}, nil
	}
	
	// ルールがトリガーされなかった場合、解決済みアラートを返す
	alertID := fmt.Sprintf("%s_%s_%f", r.name, r.operator, r.threshold)
	resolvedAlert := &Alert{
		ID:          alertID,
		Name:        r.name,
		Description: fmt.Sprintf("%s: 現在値 %.2f %s しきい値 %.2f", 
			r.description, value, operatorToSymbol(r.operator), r.threshold),
		Severity:    r.severity,
		State:       StateResolved,
		Source:      r.source,
		CreatedAt:   time.Now(),
		UpdatedAt:   time.Now(),
		ResolvedAt:  time.Now(),
		Labels:      r.labels,
		Annotations: map[string]string{
			"current_value": fmt.Sprintf("%.2f", value),
			"threshold":     fmt.Sprintf("%.2f", r.threshold),
			"operator":      r.operator,
		},
	}
	return []*Alert{resolvedAlert}, nil
}

// 演算子の記号表現
func operatorToSymbol(operator string) string {
	switch operator {
	case "gt":
		return ">"
	case "lt":
		return "<"
	case "eq":
		return "="
	case "ge":
		return ">="
	case "le":
		return "<="
	default:
		return operator
	}
}
```
