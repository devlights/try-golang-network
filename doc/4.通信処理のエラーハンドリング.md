# Goにおける通信処理のエラーハンドリング

使用するGoバージョン：Go 1.22（2024年10月時点の最新安定版）

## 1. エラーの分類と対応

### コード例：エラー分類と再試行ロジック

```go
package main

import (
	"context"
	"errors"
	"fmt"
	"io"
	"net"
	"net/http"
	"syscall"
	"time"

	"golang.org/x/time/rate"
)

// エラータイプの定義
type ErrorType int

const (
	// 一時的なエラー - 再試行が有効
	ErrorTemporary ErrorType = iota
	// 永続的なエラー - 再試行しても解決しない
	ErrorPermanent
	// リソース制約エラー - システムリソースの問題
	ErrorResource
)

// カスタムエラー構造体
type NetworkError struct {
	Type    ErrorType
	Message string
	Err     error
}

func (e *NetworkError) Error() string {
	return fmt.Sprintf("%s: %v", e.Message, e.Err)
}

func (e *NetworkError) Unwrap() error {
	return e.Err
}

// エラーの種類を分類する関数
func classifyError(err error) ErrorType {
	if err == nil {
		return ErrorPermanent // nil は再試行の必要なし
	}

	// タイムアウトエラーは一時的
	var netErr net.Error
	if errors.As(err, &netErr) && netErr.Timeout() {
		return ErrorTemporary
	}

	// 特定のシステムエラー
	if errors.Is(err, syscall.ECONNRESET) || 
	   errors.Is(err, syscall.ECONNABORTED) || 
	   errors.Is(err, syscall.ECONNREFUSED) || 
	   errors.Is(err, io.ErrUnexpectedEOF) {
		return ErrorTemporary
	}

	// HTTPステータスコードによる分類
	var httpErr *http.Response
	if errors.As(err, &httpErr) {
		code := httpErr.StatusCode
		if code >= 500 && code != 501 { // 5xxは一時的な可能性が高い（501除く）
			return ErrorTemporary
		} else if code == 429 || code == 503 { // レート制限またはサービス一時停止
			return ErrorResource
		} else if code >= 400 && code < 500 { // 4xxは通常、クライアントエラー（永続的）
			return ErrorPermanent
		}
	}

	// デフォルトは永続的エラー
	return ErrorPermanent
}

// 再試行ポリシーの設定
type RetryPolicy struct {
	MaxRetries  int           // 最大再試行回数
	BaseDelay   time.Duration // 基本待機時間
	MaxDelay    time.Duration // 最大待機時間
	Multiplier  float64       // 待機時間の乗数（指数バックオフ用）
	Jitter      float64       // ジッター係数（0-1.0）
	RetryableFunc func(error) bool // エラーが再試行可能か判断する関数
}

// デフォルトの再試行ポリシー
func DefaultRetryPolicy() RetryPolicy {
	return RetryPolicy{
		MaxRetries:  5,
		BaseDelay:   100 * time.Millisecond,
		MaxDelay:    30 * time.Second,
		Multiplier:  2.0,
		Jitter:      0.2,
		RetryableFunc: func(err error) bool {
			errType := classifyError(err)
			return errType == ErrorTemporary || errType == ErrorResource
		},
	}
}

// 指数バックオフとジッターを使用した待機時間計算
func calculateBackoff(attempt int, policy RetryPolicy) time.Duration {
	if attempt <= 0 {
		return 0
	}

	// 指数バックオフ計算
	backoff := float64(policy.BaseDelay) * pow(policy.Multiplier, attempt-1)
	
	// 最大値を制限
	if backoff > float64(policy.MaxDelay) {
		backoff = float64(policy.MaxDelay)
	}
	
	// ジッターを追加 (均一分布)
	if policy.Jitter > 0 {
		backoff = backoff * (1 - policy.Jitter/2 + policy.Jitter*rand.Float64())
	}
	
	return time.Duration(backoff)
}

// 簡易的なpow関数（Mathパッケージを使わない場合）
func pow(a, b float64) float64 {
	if b == 0 {
		return 1
	}
	
	result := a
	for i := 2; i <= int(b); i++ {
		result *= a
	}
	return result
}

// 再試行ロジックを実装した関数
func DoWithRetry(ctx context.Context, policy RetryPolicy, operation func() error) error {
	var err error
	var nextDelay time.Duration

	for attempt := 1; attempt <= policy.MaxRetries+1; attempt++ {
		// 操作実行
		err = operation()
		
		// エラーがない場合は成功
		if err == nil {
			return nil
		}
		
		// 最大試行回数に達した、またはエラーが再試行不可能な場合
		if attempt > policy.MaxRetries || !policy.RetryableFunc(err) {
			return fmt.Errorf("operation failed after %d attempts: %w", attempt, err)
		}
		
		// 次の再試行までの待機時間を計算
		nextDelay = calculateBackoff(attempt, policy)
		
		// タイマーとコンテキストのキャンセル処理
		timer := time.NewTimer(nextDelay)
		select {
		case <-timer.C:
			// 待機時間経過、次の再試行へ
		case <-ctx.Done():
			// コンテキストがキャンセルされた場合
			timer.Stop()
			return fmt.Errorf("operation cancelled: %w", ctx.Err())
		}
	}
	
	return err // 通常はここには到達しない
}

// 使用例
func main() {
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	defer cancel()
	
	policy := DefaultRetryPolicy()
	
	err := DoWithRetry(ctx, policy, func() error {
		// 通信処理の例（HTTPリクエスト）
		client := &http.Client{
			Timeout: 10 * time.Second,
		}
		
		req, err := http.NewRequestWithContext(ctx, "GET", "https://api.example.com/data", nil)
		if err != nil {
			return &NetworkError{
				Type:    ErrorPermanent,
				Message: "リクエスト作成エラー",
				Err:     err,
			}
		}
		
		resp, err := client.Do(req)
		if err != nil {
			return &NetworkError{
				Type:    classifyError(err),
				Message: "HTTP通信エラー",
				Err:     err,
			}
		}
		defer resp.Body.Close()
		
		if resp.StatusCode >= 400 {
			return &NetworkError{
				Type:    classifyError(resp),
				Message: fmt.Sprintf("HTTPステータスエラー: %d", resp.StatusCode),
				Err:     fmt.Errorf("HTTP %d", resp.StatusCode),
			}
		}
		
		// 成功時の処理
		// ...
		
		return nil
	})
	
	if err != nil {
		fmt.Printf("エラー発生: %v\n", err)
	} else {
		fmt.Println("処理成功")
	}
}
```

## 2. サーキットブレーカー実装

```go
package circuitbreaker

import (
	"errors"
	"sync"
	"time"
)

// 回路の状態
type State int

const (
	StateClosed State = iota // 閉回路: 通常状態
	StateOpen               // 開回路: 障害状態
	StateHalfOpen           // 半開回路: 回復確認中
)

// サーキットブレーカーの設定
type Settings struct {
	Threshold          int           // 失敗許容回数
	FailureRateThreshold float64     // 失敗率しきい値 (0.0-1.0)
	MinimumRequests    int           // 評価に必要な最小リクエスト数
	OpenTimeout        time.Duration // 開回路状態の継続時間
	HalfOpenMaxCalls   int           // 半開回路状態での最大呼び出し数
}

// サーキットブレーカー
type CircuitBreaker struct {
	name        string
	settings    Settings
	state       State
	failures    int
	successes   int
	halfOpen    int
	totalCalls  int
	lastStateChange time.Time
	mutex       sync.RWMutex
}

// デフォルト設定
func DefaultSettings() Settings {
	return Settings{
		Threshold:          5,
		FailureRateThreshold: 0.5,  // 50%以上の失敗率でトリップ
		MinimumRequests:    10,
		OpenTimeout:        60 * time.Second,
		HalfOpenMaxCalls:   3,
	}
}

// 新しいサーキットブレーカーを作成
func NewCircuitBreaker(name string, settings Settings) *CircuitBreaker {
	return &CircuitBreaker{
		name:        name,
		settings:    settings,
		state:       StateClosed,
		lastStateChange: time.Now(),
	}
}

// 各リクエスト前に呼び出される関数
func (cb *CircuitBreaker) Allow() error {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()

	now := time.Now()

	switch cb.state {
	case StateClosed:
		// 閉回路状態: 通常通りリクエストを許可
		cb.totalCalls++
		return nil

	case StateOpen:
		// 開回路から半開回路への遷移判定
		if now.Sub(cb.lastStateChange) > cb.settings.OpenTimeout {
			cb.setState(StateHalfOpen, now)
			cb.halfOpen = 0
			cb.totalCalls++
			return nil
		}
		// まだタイムアウトしていない
		return errors.New("circuit breaker is open")

	case StateHalfOpen:
		// 半開回路状態: 限られた数のリクエストのみ許可
		if cb.halfOpen >= cb.settings.HalfOpenMaxCalls {
			return errors.New("too many requests in half-open state")
		}
		cb.halfOpen++
		cb.totalCalls++
		return nil
	}

	return nil
}

// リクエスト成功時に呼び出される関数
func (cb *CircuitBreaker) Success() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()

	switch cb.state {
	case StateClosed:
		// 閉回路状態: 成功カウント
		cb.successes++
		// 窓サイズに達したら評価
		if cb.totalCalls >= cb.settings.MinimumRequests {
			cb.resetCounters()
		}

	case StateHalfOpen:
		// 半開回路状態: 成功カウント
		cb.successes++
		// 回復条件を満たしたら閉回路に戻す
		if cb.successes >= cb.settings.HalfOpenMaxCalls {
			cb.setState(StateClosed, time.Now())
		}
	}
}

// リクエスト失敗時に呼び出される関数
func (cb *CircuitBreaker) Failure() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()

	now := time.Now()

	switch cb.state {
	case StateClosed:
		// 閉回路状態: 失敗カウント
		cb.failures++
		
		// しきい値判定（絶対数または失敗率）
		if cb.failures >= cb.settings.Threshold {
			cb.setState(StateOpen, now)
			return
		}
		
		// 失敗率判定（最小リクエスト数を超えた場合のみ）
		if cb.totalCalls >= cb.settings.MinimumRequests {
			failureRate := float64(cb.failures) / float64(cb.totalCalls)
			if failureRate >= cb.settings.FailureRateThreshold {
				cb.setState(StateOpen, now)
				return
			}
			// 窓サイズに達したらカウンタリセット
			cb.resetCounters()
		}

	case StateHalfOpen:
		// 半開回路状態での失敗は即時に開回路に戻す
		cb.setState(StateOpen, now)
	}
}

// 状態変更
func (cb *CircuitBreaker) setState(state State, now time.Time) {
	cb.state = state
	cb.lastStateChange = now
	cb.resetCounters()
}

// カウンタリセット
func (cb *CircuitBreaker) resetCounters() {
	cb.failures = 0
	cb.successes = 0
	cb.totalCalls = 0
}

// 現在の状態を取得
func (cb *CircuitBreaker) State() State {
	cb.mutex.RLock()
	defer cb.mutex.RUnlock()
	return cb.state
}

// 使用例
func ExampleUsage() {
	// サーキットブレーカーの作成
	cb := NewCircuitBreaker("api-service", DefaultSettings())

	// 処理の実行
	if err := cb.Allow(); err != nil {
		// サーキットブレーカーがオープン状態のため処理をスキップ
		return
	}

	// 実際の処理
	err := callExternalService()
	if err != nil {
		// 失敗を記録
		cb.Failure()
		return
	}

	// 成功を記録
	cb.Success()
}

// サーキットブレーカーを使用したHTTPクライアント実装
func createCircuitBreakerClient() *http.Client {
	// 各ホスト用のサーキットブレーカーを管理するマップ
	cbMap := make(map[string]*CircuitBreaker)
	cbMutex := sync.RWMutex{}
	
	// サーキットブレーカーを取得または作成
	getCircuitBreaker := func(host string) *CircuitBreaker {
		cbMutex.RLock()
		cb, exists := cbMap[host]
		cbMutex.RUnlock()
		
		if !exists {
			cbMutex.Lock()
			defer cbMutex.Unlock()
			
			// 二重確認
			if cb, exists = cbMap[host]; !exists {
				cb = NewCircuitBreaker(host, DefaultSettings())
				cbMap[host] = cb
			}
		}
		
		return cb
	}
	
	// カスタムトランスポート
	transport := &http.Transport{
		Proxy: http.ProxyFromEnvironment,
		DialContext: (&net.Dialer{
			Timeout:   30 * time.Second,
			KeepAlive: 30 * time.Second,
		}).DialContext,
		MaxIdleConns:          100,
		IdleConnTimeout:       90 * time.Second,
		TLSHandshakeTimeout:   10 * time.Second,
		ExpectContinueTimeout: 1 * time.Second,
	}
	
	// サーキットブレーカーをラップしたRoundTripper
	roundTripper := RoundTripFunc(func(req *http.Request) (*http.Response, error) {
		host := req.URL.Host
		cb := getCircuitBreaker(host)
		
		// サーキットブレーカーチェック
		if err := cb.Allow(); err != nil {
			return nil, fmt.Errorf("circuit breaker for %s is open: %w", host, err)
		}
		
		// 実際のHTTPリクエスト
		resp, err := transport.RoundTrip(req)
		
		// 結果に基づいてサーキットブレーカーの状態を更新
		if err != nil || (resp != nil && resp.StatusCode >= 500) {
			cb.Failure()
		} else {
			cb.Success()
		}
		
		return resp, err
	})
	
	// カスタムクライアント
	return &http.Client{
		Transport: roundTripper,
		Timeout:   time.Minute,
	}
}

// RoundTripFuncはhttp.RoundTripperインターフェースを実装
type RoundTripFunc func(*http.Request) (*http.Response, error)

func (f RoundTripFunc) RoundTrip(req *http.Request) (*http.Response, error) {
	return f(req)
}
```

## 3. グレースフルデグラデーション

```go
package degradation

import (
	"context"
	"net/http"
	"sync/atomic"
	"time"

	"golang.org/x/time/rate"
)

// サービス品質レベル
type QualityLevel int

const (
	QualityHigh QualityLevel = iota
	QualityMedium
	QualityLow
	QualityMinimal
)

// APIクライアントの設定
type APIClientSettings struct {
	// 各品質レベルでのレート制限（リクエスト/秒）
	RateLimitHigh     float64
	RateLimitMedium   float64
	RateLimitLow      float64
	RateLimitMinimal  float64
	
	// 各品質レベルでのタイムアウト
	TimeoutHigh       time.Duration
	TimeoutMedium     time.Duration
	TimeoutLow        time.Duration
	TimeoutMinimal    time.Duration
	
	// エラーレート閾値
	HighToMediumThreshold   float64 // この失敗率を超えるとミディアム品質に下がる
	MediumToLowThreshold    float64
	LowToMinimalThreshold   float64
	
	// 回復チェック間隔と閾値
	RecoveryCheckInterval   time.Duration
	RecoverySuccessThreshold float64 // この成功率を超えると品質レベルアップ
}

// デフォルト設定
func DefaultAPIClientSettings() APIClientSettings {
	return APIClientSettings{
		RateLimitHigh:    20.0,
		RateLimitMedium:  10.0,
		RateLimitLow:     5.0,
		RateLimitMinimal: 1.0,
		
		TimeoutHigh:      10 * time.Second,
		TimeoutMedium:    15 * time.Second,
		TimeoutLow:       30 * time.Second,
		TimeoutMinimal:   60 * time.Second,
		
		HighToMediumThreshold: 0.1,  // 10%の失敗率
		MediumToLowThreshold:  0.3,  // 30%の失敗率
		LowToMinimalThreshold: 0.5,  // 50%の失敗率
		
		RecoveryCheckInterval:   30 * time.Second,
		RecoverySuccessThreshold: 0.95, // 95%の成功率
	}
}

// 適応型APIクライアント
type AdaptiveAPIClient struct {
	client       *http.Client
	baseURL      string
	settings     APIClientSettings
	currentLevel QualityLevel
	limiter      *rate.Limiter
	
	// 統計情報
	successCount uint64
	failureCount uint64
	successRate  float64
	
	// リクエストオプション
	hooks        []RequestHook
}

// リクエスト前後の処理をカスタマイズするためのフック
type RequestHook interface {
	BeforeRequest(*http.Request, QualityLevel) error
	AfterRequest(*http.Response, error, QualityLevel)
}

// 新しいクライアント作成
func NewAdaptiveAPIClient(baseURL string, settings APIClientSettings) *AdaptiveAPIClient {
	client := &AdaptiveAPIClient{
		baseURL:      baseURL,
		settings:     settings,
		currentLevel: QualityHigh, // 最初は最高品質から開始
		client: &http.Client{
			Timeout: settings.TimeoutHigh,
		},
		limiter: rate.NewLimiter(rate.Limit(settings.RateLimitHigh), 1),
	}
	
	// 定期的な品質レベル回復チェック
	go client.recoveryCheck()
	
	return client
}

// 定期的な回復チェック
func (c *AdaptiveAPIClient) recoveryCheck() {
	ticker := time.NewTicker(c.settings.RecoveryCheckInterval)
	defer ticker.Stop()
	
	for range ticker.C {
		// 現在の成功率を計算
		success := atomic.LoadUint64(&c.successCount)
		failure := atomic.LoadUint64(&c.failureCount)
		total := success + failure
		
		if total == 0 {
			continue // データが不足している場合はスキップ
		}
		
		successRate := float64(success) / float64(total)
		c.successRate = successRate
		
		// 回復条件を満たした場合、品質レベルを上げる
		if successRate >= c.settings.RecoverySuccessThreshold {
			c.tryUpgradeQuality()
		}
		
		// カウンタをリセット
		atomic.StoreUint64(&c.successCount, 0)
		atomic.StoreUint64(&c.failureCount, 0)
	}
}

// 品質レベルを上げる
func (c *AdaptiveAPIClient) tryUpgradeQuality() {
	currentLevel := c.currentLevel
	
	switch currentLevel {
	case QualityMinimal:
		c.setQualityLevel(QualityLow)
	case QualityLow:
		c.setQualityLevel(QualityMedium)
	case QualityMedium:
		c.setQualityLevel(QualityHigh)
	}
}

// 品質レベルを下げる
func (c *AdaptiveAPIClient) degradeQuality() {
	currentLevel := c.currentLevel
	
	switch currentLevel {
	case QualityHigh:
		c.setQualityLevel(QualityMedium)
	case QualityMedium:
		c.setQualityLevel(QualityLow)
	case QualityLow:
		c.setQualityLevel(QualityMinimal)
	}
}

// 品質レベル設定
func (c *AdaptiveAPIClient) setQualityLevel(level QualityLevel) {
	c.currentLevel = level
	
	// 品質レベルに応じたパラメータ設定
	switch level {
	case QualityHigh:
		c.client.Timeout = c.settings.TimeoutHigh
		c.limiter.SetLimit(rate.Limit(c.settings.RateLimitHigh))
	case QualityMedium:
		c.client.Timeout = c.settings.TimeoutMedium
		c.limiter.SetLimit(rate.Limit(c.settings.RateLimitMedium))
	case QualityLow:
		c.client.Timeout = c.settings.TimeoutLow
		c.limiter.SetLimit(rate.Limit(c.settings.RateLimitLow))
	case QualityMinimal:
		c.client.Timeout = c.settings.TimeoutMinimal
		c.limiter.SetLimit(rate.Limit(c.settings.RateLimitMinimal))
	}
}

// リクエストフックを追加
func (c *AdaptiveAPIClient) AddHook(hook RequestHook) {
	c.hooks = append(c.hooks, hook)
}

// GETリクエスト
func (c *AdaptiveAPIClient) Get(ctx context.Context, path string) (*http.Response, error) {
	url := c.baseURL + path
	req, err := http.NewRequestWithContext(ctx, "GET", url, nil)
	if err != nil {
		return nil, err
	}
	
	return c.Do(req)
}

// POSTリクエスト（ボディ処理は割愛）
func (c *AdaptiveAPIClient) Post(ctx context.Context, path string, contentType string, body io.Reader) (*http.Response, error) {
	url := c.baseURL + path
	req, err := http.NewRequestWithContext(ctx, "POST", url, body)
	if err != nil {
		return nil, err
	}
	req.Header.Set("Content-Type", contentType)
	
	return c.Do(req)
}

// リクエスト実行
func (c *AdaptiveAPIClient) Do(req *http.Request) (*http.Response, error) {
	// レート制限に従う
	ctx := req.Context()
	if err := c.limiter.Wait(ctx); err != nil {
		return nil, err
	}
	
	// 現在の品質レベルを取得
	qualityLevel := c.currentLevel
	
	// リクエストの前処理（フック）
	for _, hook := range c.hooks {
		if err := hook.BeforeRequest(req, qualityLevel); err != nil {
			return nil, err
		}
	}
	
	// 実際のリクエスト実行
	resp, err := c.client.Do(req)
	
	// 結果に基づく統計カウント
	if err != nil || (resp != nil && resp.StatusCode >= 500) {
		atomic.AddUint64(&c.failureCount, 1)
		
		// 失敗状況に応じて品質レベルを調整
		failure := atomic.LoadUint64(&c.failureCount)
		success := atomic.LoadUint64(&c.successCount)
		total := failure + success
		
		if total > 10 { // 最小サンプル数
			failureRate := float64(failure) / float64(total)
			
			// 閾値に基づいて品質低下判定
			switch qualityLevel {
			case QualityHigh:
				if failureRate > c.settings.HighToMediumThreshold {
					c.degradeQuality()
				}
			case QualityMedium:
				if failureRate > c.settings.MediumToLowThreshold {
					c.degradeQuality()
				}
			case QualityLow:
				if failureRate > c.settings.LowToMinimalThreshold {
					c.degradeQuality()
				}
			}
		}
	} else {
		atomic.AddUint64(&c.successCount, 1)
	}
	
	// リクエストの後処理（フック）
	for _, hook := range c.hooks {
		hook.AfterRequest(resp, err, qualityLevel)
	}
	
	return resp, err
}

// 品質レベルに応じたヘッダ設定フック
type QualityAwareHeader struct{}

func (h *QualityAwareHeader) BeforeRequest(req *http.Request, level QualityLevel) error {
```

```go
// RoundTripper インターフェース実装
func (t *InstrumentedTransport) RoundTrip(req *http.Request) (*http.Response, error) {
	method := req.Method
	path := req.URL.Path
	
	// リクエストサイズ計測
	if req.ContentLength > 0 {
		t.metrics.requestSize.WithLabelValues(method, path).Observe(float64(req.ContentLength))
	}
	
	// リクエスト開始時間
	startTime := time.Now()
	
	// 実際のリクエスト
	resp, err := t.next.RoundTrip(req)
	
	// エラー処理
	if err != nil {
		errType := "network"
		errCode := "unknown"
		
		// エラータイプの分類
		var netErr net.Error
		if errors.As(err, &netErr) {
			if netErr.Timeout() {
				errType = "timeout"
			} else if netErr.Temporary() {
				errType = "temporary"
			}
		}
		
		// エラーカウント
		t.metrics.errorCounter.WithLabelValues(method, path, errType, errCode).Inc()
		
		// リクエスト時間計測（エラーの場合も）
		duration := time.Since(startTime).Seconds()
		t.metrics.requestDuration.WithLabelValues(method, path, "error").Observe(duration)
		
		return nil, err
	}
	
	// ステータスコード文字列
	status := http.StatusText(resp.StatusCode)
	
	// リクエスト時間計測
	duration := time.Since(startTime).Seconds()
	t.metrics.requestDuration.WithLabelValues(method, path, status).Observe(duration)
	
	// リクエストカウント増加
	t.metrics.requestCounter.WithLabelValues(method, path, status).Inc()
	
	// レスポンスサイズ計測
	if resp.ContentLength > 0 {
		t.metrics.responseSize.WithLabelValues(method, path, status).Observe(float64(resp.ContentLength))
	}
	
	// エラーレスポンスの場合はエラーカウント
	if resp.StatusCode >= 400 {
		errType := "http"
		errCode := status
		
		t.metrics.errorCounter.WithLabelValues(method, path, errType, errCode).Inc()
	}
	
	return resp, nil
}
```

```go
		halfOpenMax:     halfOpenMax,
		resetTimeout:    resetTimeout,
		state:           StateClosed,

func (cb *CircuitBreaker) Allow() error {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()

	now := time.Now()

	switch cb.state {
	case StateClosed:
		return nil
	case StateOpen:
		if now.Sub(cb.lastStateChange) > cb.resetTimeout {
			cb.state = StateHalfOpen
			cb.lastStateChange = now
			cb.successes = 0
			return nil
		}
		return fmt.Errorf("circuit breaker is open for %s", cb.name)
	case StateHalfOpen:
		if cb.successes < cb.halfOpenMax {
			return nil
		}
		return fmt.Errorf("too many requests in half-open state for %s", cb.name)
	}
	return nil
}

func (cb *CircuitBreaker) Success() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()

	switch cb.state {
	case StateClosed:
		cb.failures = 0
	case StateHalfOpen:
		cb.successes++
		if cb.successes >= cb.halfOpenMax {
			cb.state = StateClosed
			cb.lastStateChange = time.Now()
			cb.failures = 0
			cb.successes = 0
		}
	}
}

func (cb *CircuitBreaker) Failure() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()

	switch cb.state {
	case StateClosed:
		cb.failures++
		if cb.failures >= cb.threshold {
			cb.state = StateOpen
			cb.lastStateChange = time.Now()
		}
	case StateHalfOpen:
		cb.state = StateOpen
		cb.lastStateChange = time.Now()
	}
}

func (cb *CircuitBreaker) State() CircuitState {
	cb.mutex.RLock()
	defer cb.mutex.RUnlock()
	return cb.state

// ========== 再試行ポリシー ==========

type RetryPolicy struct {
	MaxRetries    int
	BaseDelay     time.Duration
	MaxDelay      time.Duration
	Multiplier    float64
	Jitter        float64
	RetryableFunc func(error) bool
}

func DefaultRetryPolicy() RetryPolicy {
	return RetryPolicy{
		MaxRetries: 3,
		BaseDelay:  100 * time.Millisecond,
		MaxDelay:   30 * time.Second,
		Multiplier: 2.0,
		Jitter:     0.2,
		RetryableFunc: func(err error) bool {
			if err == nil {
				return false
			}

			var netErr net.Error
			if errors.As(err, &netErr) && netErr.Timeout() {
				return true
			}

			if errors.Is(err, syscall.ECONNRESET) ||
				errors.Is(err, syscall.ECONNABORTED) ||
				errors.Is(err, syscall.ECONNREFUSED) ||
				errors.Is(err, io.ErrUnexpectedEOF) {
				return true
			}

			// HTTPエラーの場合
			var resp *http.Response
			if errors.As(err, &resp) {
				code := resp.StatusCode
				// 5xx系は一時的なエラーとして扱い再試行
				return code >= 500 && code != 501
			}

			return false
		},
	}
}

func calculateBackoff(attempt int, policy RetryPolicy) time.Duration {
	if attempt <= 0 {
		return 0
	}

	// 指数バックオフ計算
	backoff := float64(policy.BaseDelay)
	for i := 1; i < attempt; i++ {
		backoff *= policy.Multiplier
	}

	// 最大値を制限
	if backoff > float64(policy.MaxDelay) {
		backoff = float64(policy.MaxDelay)
	}

	// ジッターを追加
	if policy.Jitter > 0 {
		backoff = backoff * (1 - policy.Jitter/2 + policy.Jitter*rand.Float64())
	}

	return time.Duration(backoff)
}

func (c *ApiClient) qualityMonitor() {
	ticker := time.NewTicker(10 * time.Second)
	defer ticker.Stop()

	for range ticker.C {
		c.evaluateQuality()
	}
}

func (c *ApiClient) evaluateQuality() {
	c.qualityMutex.Lock()
	defer c.qualityMutex.Unlock()

	// メトリクスリセット時間を過ぎていたらリセット
	now := time.Now()
	if now.Sub(c.metricResetTime) >= c.metricResetInterval {
		c.metricResetTime = now
		
		// 十分なサンプルがあれば品質レベルを評価
		total := c.successCount + c.failureCount
		if total >= 10 {
			successRate := float64(c.successCount) / float64(total)
			
			// 現在の品質レベルに応じて評価
			switch c.currentQuality {
			case QualityHigh:
				if 1-successRate > c.options.QualityThresholds.HighToMedium {
					c.currentQuality = QualityMedium
					log.Printf("品質レベルを下げました: High -> Medium (成功率: %.2f%%)", successRate*100)
				}
			case QualityMedium:
				if 1-successRate > c.options.QualityThresholds.MediumToLow {
					c.currentQuality = QualityLow
					log.Printf("品質レベルを下げました: Medium -> Low (成功率: %.2f%%)", successRate*100)
				} else if successRate > c.options.QualityThresholds.RecoveryRate {
					c.currentQuality = QualityHigh
					log.Printf("品質レベルを上げました: Medium -> High (成功率: %.2f%%)", successRate*100)
				}
			case QualityLow:
				if 1-successRate > c.options.QualityThresholds.LowToMinimal {
					c.currentQuality = QualityMinimal
					log.Printf("品質レベルを下げました: Low -> Minimal (成功率: %.2f%%)", successRate*100)
				} else if successRate > c.options.QualityThresholds.RecoveryRate {
					c.currentQuality = QualityMedium
					log.Printf("品質レベルを上げました: Low -> Medium (成功率: %.2f%%)", successRate*100)
				}
			case QualityMinimal:
				if successRate > c.options.QualityThresholds.RecoveryRate {
					c.currentQuality = QualityLow
					log.Printf("品質レベルを上げました: Minimal -> Low (成功率: %.2f%%)", successRate*100)
				}
			}
		}
		
		// カウンタリセット
		c.successCount = 0
		c.failureCount = 0
	}
}

func (c *ApiClient) getCircuitBreaker(host string) *CircuitBreaker {
	c.circuitBreakerMutex.RLock()
	cb, exists := c.circuitBreakers[host]
	c.circuitBreakerMutex.RUnlock()

	if !exists {
		c.circuitBreakerMutex.Lock()
		defer c.circuitBreakerMutex.Unlock()
		
		// ダブルチェック
		if cb, exists = c.circuitBreakers[host]; !exists {
			cb = NewCircuitBreaker(
				host,
				c.options.CBThreshold,
				c.options.CBHalfOpenMax,
				c.options.CBResetTimeout,
			)
			c.circuitBreakers[host] = cb
		}
	}

	return cb
}

func (c *ApiClient) Do(req *http.Request) (*http.Response, error) {
	// URLをパースして必要なコンポーネントを取得
	host := req.URL.Host
	path := req.URL.Path
	method := req.Method

	// コンテキスト取得
	ctx := req.Context()

	// レート制限に従う
	if err := c.limiter.Wait(ctx); err != nil {
		return nil, fmt.Errorf("rate limit wait error: %w", err)
	}

	// サーキットブレーカーチェック
	cb := c.getCircuitBreaker(host)
	if err := cb.Allow(); err != nil {
		c.metrics.circuitBreakerState.WithLabelValues(host).Set(float64(StateOpen))
		c.metrics.errorCounter.WithLabelValues(method, host, path, "circuit_open").Inc()
		return nil, err
	}

	// サーキットブレーカー状態を記録
	c.metrics.circuitBreakerState.WithLabelValues(host).Set(float64(cb.State()))

	// 品質レベルに応じたヘッダー設定
	c.qualityMutex.RLock()
	quality := c.currentQuality
	c.qualityMutex.RUnlock()

	switch quality {
	case QualityHigh:
		// フルリクエスト
	case QualityMedium:
		req.Header.Set("Accept-Encoding", "gzip, deflate")
	case QualityLow, QualityMinimal:
		req.Header.Set("Accept-Encoding", "gzip, deflate")
		req.Header.Set("X-Reduced-Content", "true")
	}

	// 再試行ポリシー
	var resp *http.Response
	var err error
	var attempt int

	startTime := time.Now()

	for attempt = 0; attempt <= c.options.RetryPolicy.MaxRetries; attempt++ {
		if attempt > 0 {
			// 再試行前の遅延計算
			delay := calculateBackoff(attempt, c.options.RetryPolicy)
			
			// 再試行カウント
			c.metrics.retryCounter.WithLabelValues(method, host, path).Inc()
			
			// 遅延タイマー
			timer := time.NewTimer(delay)
			select {
			case <-timer.C:
				// 遅延終了、次の試行へ
			case <-ctx.Done():
				timer.Stop()
				return nil, ctx.Err()
			}
		}

		// 実際のリクエスト実行
		reqClone := req.Clone(ctx)
		resp, err = c.httpClient.Do(reqClone)

		// 成功の場合
		if err == nil && resp.StatusCode < 500 {
			break
		}

		// エラーの場合、再試行すべきか判断
		if !c.options.RetryPolicy.RetryableFunc(err) {
			break
		}

		// 最後の試行の場合は再試行しない
		if attempt == c.options.RetryPolicy.MaxRetries {
			break
		}
	}

	// リクエスト時間を記録
	duration := time.Since(startTime).Seconds()
	c.metrics.requestDuration.WithLabelValues(method, host, path).Observe(duration)

	// 結果処理
	if err != nil {
		// エラー記録
		c.metrics.errorCounter.WithLabelValues(method, host, path, "network").Inc()
		
		// サーキットブレーカー更新
		cb.Failure()
		
		// 品質メトリクス更新
		c.qualityMutex.Lock()
		c.failureCount++
		c.qualityMutex.Unlock()
		
		return nil, err
	}

	// HTTP エラーコード
	if resp.StatusCode >= 400 {
		errType := "http_4xx"
		if resp.StatusCode >= 500 {
			errType = "http_5xx"
		}
		
		c.metrics.errorCounter.WithLabelValues(method, host, path, errType).Inc()
		c.metrics.requestCounter.WithLabelValues(method, host, path, fmt.Sprintf("%d", resp.StatusCode)).Inc()
		
		// 5xx エラーはサーキットブレーカー更新
		if resp.StatusCode >= 500 {
			cb.Failure()
			
			// 品質メトリクス更新
			c.qualityMutex.Lock()
			c.failureCount++
			c.qualityMutex.Unlock()
		} else {
			// 4xx エラーは通常クライアントエラーなのでサーキットブレーカーは更新しない
			cb.Success()
			
			// 品質メトリクス更新（4xxはクライアントエラーなので成功とカウント）
			c.qualityMutex.Lock()
			c.successCount++
			c.qualityMutex.Unlock()
		}
	} else {
		// 成功を記録
		c.metrics.requestCounter.WithLabelValues(method, host, path, fmt.Sprintf("%d", resp.StatusCode)).Inc()
		
		// サーキットブレーカー更新
		cb.Success()
		
		// 品質メトリクス更新
		c.qualityMutex.Lock()
		c.successCount++
		c.qualityMutex.Unlock()
	}

	return resp, nil
}
```

## 6. 考慮すべき要点と実務上のポイント

### エラー処理の階層化

1. **ローカルエラー処理**: 
   - 個々のリクエストレベルでのエラー検出と分類
   - エラー情報の詳細な記録とコンテキスト保持

2. **サービスレベルエラー処理**:
   - サーキットブレーカーによる障害の分離
   - レート制限と輻輳制御

3. **システムレベルエラー処理**:
   - テレメトリによる障害検出とアラート
   - 回復メカニズムの自動化

### パフォーマンスへの影響

1. **エラー処理のオーバーヘッド**:
   - エラー処理コードが全体の約30%を占めることがある
   - ロギングは特にパフォーマンスに影響するため、適切なレベルで実施する

2. **メトリクス収集の影響**:
   - 高トラフィック環境では、サンプリングレートを下げることを検討
   - クリティカルパスのメトリクスは別ゴルーチンで非同期に処理

3. **再試行の制御**:
   - 過剰な再試行はシステム全体の負荷を増大させる
   - 予測可能な待機時間とジッターの適用が重要

### 実務向けチューニングのポイント

1. **高スループット環境向け設定**:
   - `MaxIdleConns`と`MaxConnsPerHost`を適切に設定（数百〜数千）
   - Keep-Alive接続の積極的な再利用

2. **信頼性重視環境向け設定**:
   - より保守的な再試行ポリシー（再試行間隔を長めに）
   - より低いサーキットブレーカー閾値

3. **レスポンスタイム重視環境向け設定**:
   - タイムアウトを短めに設定
   - フォールバック戦略の徹底実装

4. **リソース制約環境向け設定**:
   - オンデマンドでの接続確立
   - 低メモリ環境ではプールサイズを制限

```go
package main

import (
	"context"
	"errors"
	"fmt"
	"io"
	"log"
	"math/rand"

	"sync"
	"syscall"

// エラータイプの定義
type ErrorType int

const (
	// 一時的なエラー - 再試行が有効
	ErrorTemporary ErrorType = iota
	// 永続的なエラー - 再試行しても解決しない
	ErrorPermanent
	// リソース制約エラー - システムリソースの問題
	ErrorResource
)

// カスタムエラー構造体
type TCPError struct {
	Type    ErrorType
	Message string
	Err     error
}

func (e *TCPError) Error() string {
	return fmt.Sprintf("%s: %v", e.Message, e.Err)
}

func (e *TCPError) Unwrap() error {
	return e.Err
}

// エラーの種類を分類する関数
func classifyError(err error) ErrorType {
	if err == nil {
		return ErrorPermanent // nil は再試行の必要なし
	}

	// タイムアウトエラーは一時的
	var netErr net.Error
	if errors.As(err, &netErr) && netErr.Timeout() {
		return ErrorTemporary
	}

	// 特定のシステムエラー
	if errors.Is(err, syscall.ECONNRESET) ||
		errors.Is(err, syscall.ECONNABORTED) ||
		errors.Is(err, syscall.ECONNREFUSED) ||
		errors.Is(err, io.ErrUnexpectedEOF) {
		return ErrorTemporary
	}

	// 接続が閉じられた
	if errors.Is(err, io.EOF) {
		return ErrorTemporary
	}

	// 一時的なネットワークエラー
	if errors.Is(err, syscall.ETIMEDOUT) ||
		errors.Is(err, syscall.EHOSTUNREACH) ||
		errors.Is(err, syscall.ENETUNREACH) {
		return ErrorTemporary
	}

	// リソース制約エラー
	if errors.Is(err, syscall.EMFILE) || // ファイルディスクリプタ不足
		errors.Is(err, syscall.ENFILE) || // システムファイルテーブル不足
		errors.Is(err, syscall.ENOBUFS) || // バッファスペース不足
		errors.Is(err, syscall.ENOMEM) { // メモリ不足
		return ErrorResource
	}

	// デフォルトは永続的エラー
	return ErrorPermanent
}

// 基本的なTCPクライアント
type TCPClient struct {
	// 接続設定
	address          string
	connectTimeout   time.Duration
	readTimeout      time.Duration
	writeTimeout     time.Duration
	keepAliveTimeout time.Duration
	keepAliveCount   int
	bufferSize       int

	// 内部状態
	conn net.Conn
	mu   sync.Mutex
}

// 新しいTCPクライアント作成
func NewTCPClient(address string) *TCPClient {
	return &TCPClient{
		address:          address,
		connectTimeout:   5 * time.Second,
		readTimeout:      10 * time.Second,
		writeTimeout:     10 * time.Second,
		keepAliveTimeout: 30 * time.Second,
		keepAliveCount:   5,
		bufferSize:       4096,
	}
}

// 接続確立（エラーハンドリング付き）
func (c *TCPClient) Connect(ctx context.Context) error {
	c.mu.Lock()
	defer c.mu.Unlock()

	// 既に接続があれば閉じる
	if c.conn != nil {
		c.conn.Close()
		c.conn = nil
	}

	// コンテキスト付きの接続タイムアウト
	var dialer net.Dialer
	dialer.Timeout = c.connectTimeout
	dialer.KeepAlive = c.keepAliveTimeout

	var err error
	c.conn, err = dialer.DialContext(ctx, "tcp", c.address)
	if err != nil {
		errType := classifyError(err)
		return &TCPError{
			Type:    errType,
			Message: "TCP接続エラー",
			Err:     err,
		}
	}

	// TCPコネクション固有の設定
	tcpConn, ok := c.conn.(*net.TCPConn)
	if ok {
		// KeepAlive設定
		tcpConn.SetKeepAlive(true)
		tcpConn.SetKeepAlivePeriod(c.keepAliveTimeout)

		// Nagleアルゴリズム無効化（リアルタイム性を向上）
		tcpConn.SetNoDelay(true)

		// 各種バッファサイズ設定
		tcpConn.SetReadBuffer(c.bufferSize)
		tcpConn.SetWriteBuffer(c.bufferSize)
	}

	return nil
}

// データ送信（エラーハンドリング付き）
func (c *TCPClient) Send(ctx context.Context, data []byte) error {
	c.mu.Lock()
	defer c.mu.Unlock()

	if c.conn == nil {
		return &TCPError{
			Type:    ErrorPermanent,
			Message: "接続が確立されていません",
			Err:     errors.New("connection not established"),
		}
	}

	// 書き込みタイムアウト設定
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(c.writeTimeout)
	}
	if err := c.conn.SetWriteDeadline(deadline); err != nil {
		return &TCPError{
			Type:    ErrorTemporary,
			Message: "タイムアウト設定エラー",
			Err:     err,
		}
	}

	// データ送信
	n, err := c.conn.Write(data)
	if err != nil {
		errType := classifyError(err)
		return &TCPError{
			Type:    errType,
			Message: "データ送信エラー",
			Err:     err,
		}
	}

	// 部分的な書き込みチェック
	if n != len(data) {
		return &TCPError{
			Type:    ErrorTemporary,
			Message: "データの一部のみが送信されました",
			Err:     fmt.Errorf("sent %d bytes out of %d", n, len(data)),
		}
	}

	return nil
}

// データ受信（エラーハンドリング付き）
func (c *TCPClient) Receive(ctx context.Context, maxSize int) ([]byte, error) {
	c.mu.Lock()
	defer c.mu.Unlock()

	if c.conn == nil {
		return nil, &TCPError{
			Type:    ErrorPermanent,
			Message: "接続が確立されていません",
			Err:     errors.New("connection not established"),
		}
	}

	// 読み込みタイムアウト設定
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(c.readTimeout)
	}
	if err := c.conn.SetReadDeadline(deadline); err != nil {
		return nil, &TCPError{
			Type:    ErrorTemporary,
			Message: "タイムアウト設定エラー",
			Err:     err,
		}
	}

	// データ受信
	buffer := make([]byte, maxSize)
	n, err := c.conn.Read(buffer)
	if err != nil {
		errType := classifyError(err)
		return nil, &TCPError{
			Type:    errType,
			Message: "データ受信エラー",
			Err:     err,
		}
	}

	return buffer[:n], nil
}

// 接続を閉じる
func (c *TCPClient) Close() error {
	c.mu.Lock()
	defer c.mu.Unlock()

	if c.conn != nil {
		err := c.conn.Close()
		c.conn = nil
		if err != nil {
			return &TCPError{
				Type:    ErrorTemporary,
				Message: "接続クローズエラー",
				Err:     err,
			}
		}
	}
	return nil
}

// 接続状態チェック
func (c *TCPClient) IsConnected() bool {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.conn != nil
}

		log.Fatalf("受信エラー: %v", err)
```

## 2. TCP 通信向けコネクションプールとサーキットブレーカー

```go
package main

import (
	"context"
	"errors"
	"fmt"
	"log"
	"math/rand"
	"net"
	"sync"
	"sync/atomic"
	"time"
)

// ==== TCP コネクションプール ====

type PooledTCPConnection struct {
	conn      net.Conn
	createdAt time.Time
	lastUsed  time.Time
	inUse     bool
}

type TCPConnectionPool struct {
	address       string
	maxSize       int
	minSize       int
	maxLifetime   time.Duration
	maxIdleTime   time.Duration
	connectTimeout time.Duration

	mu            sync.Mutex
	connections   []*PooledTCPConnection
	connCount     int32
	
	// イベント通知用チャネル
	eventCh       chan struct{}
	closed        bool
}

func NewTCPConnectionPool(address string, maxSize int) *TCPConnectionPool {
	pool := &TCPConnectionPool{
		address:       address,
		maxSize:       maxSize,
		minSize:       2,
		maxLifetime:   10 * time.Minute,
		maxIdleTime:   5 * time.Minute,
		connectTimeout: 5 * time.Second,
		connections:   make([]*PooledTCPConnection, 0, maxSize),
		eventCh:       make(chan struct{}, 1),
	}

	// プール管理ゴルーチン起動
	go pool.maintain()

	return pool
}

// プールの状態維持
func (p *TCPConnectionPool) maintain() {
	ticker := time.NewTicker(30 * time.Second)
	defer ticker.Stop()

	for {
		select {
		case <-ticker.C:
			p.cleanIdleConnections()
		case <-p.eventCh:
			if p.closed {
				return
			}
		}
	}
}

// アイドル接続のクリーンアップ
func (p *TCPConnectionPool) cleanIdleConnections() {
	p.mu.Lock()
	defer p.mu.Unlock()

	if p.closed {
		return
	}

	now := time.Now()
	kept := make([]*PooledTCPConnection, 0, len(p.connections))

	for _, conn := range p.connections {
		if conn.inUse {
			// 使用中の接続は維持
			kept = append(kept, conn)
			continue
		}

		// 古すぎる接続は閉じる
		if now.Sub(conn.createdAt) > p.maxLifetime {
			conn.conn.Close()
			atomic.AddInt32(&p.connCount, -1)
			continue
		}

		// 長時間アイドル状態の接続は閉じる（ただし最小数は維持）
		if now.Sub(conn.lastUsed) > p.maxIdleTime && len(kept) >= p.minSize {
			conn.conn.Close()
			atomic.AddInt32(&p.connCount, -1)
			continue
		}

		kept = append(kept, conn)
	}

	p.connections = kept
}

// 接続の取得
func (p *TCPConnectionPool) GetConnection(ctx context.Context) (net.Conn, error) {
	p.mu.Lock()
	
	if p.closed {
		p.mu.Unlock()
		return nil, errors.New("connection pool is closed")
	}

	// 既存の空き接続を探す
	for _, conn := range p.connections {
		if !conn.inUse {
			conn.inUse = true
			conn.lastUsed = time.Now()
			p.mu.Unlock()
			return &PoolConnection{pool: p, conn: conn}, nil
		}
	}

	// プールサイズの制限を確認
	currentCount := atomic.LoadInt32(&p.connCount)
	if currentCount >= int32(p.maxSize) {
		p.mu.Unlock()
		return nil, errors.New("connection pool is full")
	}

	// 新しい接続作成
	p.mu.Unlock()

	dialer := net.Dialer{Timeout: p.connectTimeout}
	rawConn, err := dialer.DialContext(ctx, "tcp", p.address)
	if err != nil {
		return nil, fmt.Errorf("failed to create new connection: %w", err)
	}

	// TCP固有の設定
	if tcpConn, ok := rawConn.(*net.TCPConn); ok {
		tcpConn.SetKeepAlive(true)
		tcpConn.SetKeepAlivePeriod(1 * time.Minute)
		tcpConn.SetNoDelay(true)
	}

	// 接続をプールに追加
	pooledConn := &PooledTCPConnection{
		conn:      rawConn,
		createdAt: time.Now(),
		lastUsed:  time.Now(),
		inUse:     true,
	}

	p.mu.Lock()
	p.connections = append(p.connections, pooledConn)
	atomic.AddInt32(&p.connCount, 1)
	p.mu.Unlock()

	return &PoolConnection{pool: p, conn: pooledConn}, nil
}

// プールを閉じる
func (p *TCPConnectionPool) Close() {
	p.mu.Lock()
	defer p.mu.Unlock()

	if p.closed {
		return
	}

	p.closed = true

	// すべての接続を閉じる
	for _, conn := range p.connections {
		conn.conn.Close()
	}
	p.connections = nil

	// イベント通知
	select {
	case p.eventCh <- struct{}{}:
	default:
	}
	close(p.eventCh)
}

type CircuitState int

const (
	StateClosed CircuitState = iota
	StateOpen
	StateHalfOpen
)

type TCPCircuitBreaker struct {
	name              string
	failures          int32
	successes         int32
	threshold         int32
	halfOpenMax       int32
	resetTimeout      time.Duration
	state             CircuitState
	lastStateChange   time.Time
	mutex             sync.RWMutex
	
	// テレメトリ用
	totalRequests     int64
	totalFailures     int64
	consecutiveSuccess int64
	consecutiveFailure int64
}

func NewTCPCircuitBreaker(name string, threshold, halfOpenMax int, resetTimeout time.Duration) *TCPCircuitBreaker {
	return &TCPCircuitBreaker{
		name:            name,
		threshold:       int32(threshold),
		halfOpenMax:     int32(halfOpenMax),
		resetTimeout:    resetTimeout,
		state:           StateClosed,
		lastStateChange: time.Now(),
	}
}

func (cb *TCPCircuitBreaker) Allow() error {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()

	// テレメトリ更新
	atomic.AddInt64(&cb.totalRequests, 1)

	now := time.Now()
	state := cb.state

	switch state {
	case StateClosed:
		return nil
	case StateOpen:
		if now.Sub(cb.lastStateChange) > cb.resetTimeout {
			cb.state = StateHalfOpen
			cb.lastStateChange = now
			cb.successes = 0
			return nil
		}
		return fmt.Errorf("circuit breaker is open for %s", cb.name)
	case StateHalfOpen:
		if cb.successes < cb.halfOpenMax {
			return nil
		}
		return fmt.Errorf("half-open circuit breaker at capacity for %s", cb.name)
	}
	
	return nil
}

func (cb *TCPCircuitBreaker) Success() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()

	// テレメトリ更新
	atomic.AddInt64(&cb.consecutiveSuccess, 1)
	atomic.StoreInt64(&cb.consecutiveFailure, 0)

	switch cb.state {
	case StateClosed:
		cb.failures = 0
	case StateHalfOpen:
		cb.successes++
		if cb.successes >= cb.halfOpenMax {
			cb.state = StateClosed
			cb.lastStateChange = time.Now()
			cb.failures = 0
			cb.successes = 0
		}
	}
}

func (cb *TCPCircuitBreaker) Failure() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()

	// テレメトリ更新
	atomic.AddInt64(&cb.totalFailures, 1)
	atomic.StoreInt64(&cb.consecutiveSuccess, 0)
	atomic.AddInt64(&cb.consecutiveFailure, 1)

	switch cb.state {
	case StateClosed:
		cb.failures++
		if cb.failures >= cb.threshold {
			cb.state = StateOpen
			cb.lastStateChange = time.Now()
		}
	case StateHalfOpen:
		cb.state = StateOpen
		cb.lastStateChange = time.Now()
	}
}

func (cb *TCPCircuitBreaker) State() CircuitState {
	cb.mutex.RLock()
	defer cb.mutex.RUnlock()
	return cb.state
}

// テレメトリ取得
func (cb *TCPCircuitBreaker) Metrics() map[string]int64 {
	state := int64(cb.State())
	
	return map[string]int64{
		"state":               state,
		"total_requests":      atomic.LoadInt64(&cb.totalRequests),
		"total_failures":      atomic.LoadInt64(&cb.totalFailures),
		"consecutive_success": atomic.LoadInt64(&cb.consecutiveSuccess),
		"consecutive_failure": atomic.LoadInt64(&cb.consecutiveFailure),
	}
}

type EnhancedTCPClient struct {

	// コンポーネント

	retryBaseDelay   time.Duration
	retryMaxDelay    time.Duration
	retryMultiplier  float64
	retryJitter      float64
}

func NewEnhancedTCPClient(address string) *EnhancedTCPClient {
	client := &EnhancedTCPClient{
		address:         address,
		readTimeout:     10 * time.Second,
		writeTimeout:    10 * time.Second,
		connectTimeout:  5 * time.Second,
		maxRetries:      3,
		retryBaseDelay:  100 * time.Millisecond,
		retryMaxDelay:   5 * time.Second,
		retryMultiplier: 2.0,
		retryJitter:     0.2,
	}
	
	// コネクションプール初期化
	client.pool = NewTCPConnectionPool(address, 10)
	
	// サーキットブレーカー初期化
	client.circuitBreaker = NewTCPCircuitBreaker(
		address,
		5,    // 5回の失敗で開く
		3,    // 半開状態で3回の成功で閉じる
		30 * time.Second, // 30秒後に半開状態を試す
	)
	
	return client
}

// バックオフ時間計算
func (c *EnhancedTCPClient) calculateBackoff(attempt int) time.Duration {
	if attempt <= 0 {
		return 0
	}

	// 指数バックオフ計算
	backoff := float64(c.retryBaseDelay)
	for i := 1; i < attempt; i++ {
		backoff *= c.retryMultiplier
	}

	// 最大値を制限
	if backoff > float64(c.retryMaxDelay) {
		backoff = float64(c.retryMaxDelay)
	}

	// ジッター追加
	if c.retryJitter > 0 {
		backoff = backoff * (1 - c.retryJitter/2 + c.retryJitter*rand.Float64())
	}

	return time.Duration(backoff)
}

// 再試行付きTCP送受信
func (c *EnhancedTCPClient) SendWithRetry(ctx context.Context, data []byte) ([]byte, error) {
	// サーキットブレーカーチェック
	if err := c.circuitBreaker.Allow(); err != nil {
		return nil, fmt.Errorf("circuit breaker open: %w", err)
	}
	
	var lastErr error
	
	// 再試行ループ
	for attempt := 0; attempt <= c.maxRetries; attempt++ {
		if attempt > 0 {
			// 再試行前の遅延
			delay := c.calculateBackoff(attempt)
			timer := time.NewTimer(delay)
			
			select {
			case <-timer.C:
				// 遅延終了、次の試行へ
			case <-ctx.Done():
				timer.Stop()
				return nil, ctx.Err()
			}
			
			log.Printf("TCP通信再試行 %d/%d", attempt, c.maxRetries)
		}
		
		// コネクションプールから接続取得
		conn, err := c.pool.GetConnection(ctx)
		if err != nil {
			lastErr = fmt.Errorf("failed to get connection: %w", err)
			c.circuitBreaker.Failure()
			continue
		}
		
		// 必ず接続を返却
		defer conn.Close()
		
		// 書き込みタイムアウト設定
		writeDeadline := time.Now().Add(c.writeTimeout)
		if err := conn.SetWriteDeadline(writeDeadline); err != nil {
			lastErr = fmt.Errorf("failed to set write deadline: %w", err)
			c.circuitBreaker.Failure()
			continue
		}
		
		// データ送信
		_, err = conn.Write(data)
		if err != nil {
			lastErr = fmt.Errorf("failed to write data: %w", err)
			c.circuitBreaker.Failure()
			continue
		}
		
		// 読み込みタイムアウト設定
		readDeadline := time.Now().Add(c.readTimeout)
		if err := conn.SetReadDeadline(readDeadline); err != nil {
			lastErr = fmt.Errorf("failed to set read deadline: %w", err)
			c.circuitBreaker.Failure()
			continue
		}
		
		// レスポンス読み込み
		buffer := make([]byte, 4096)
		n, err := conn.Read(buffer)
		if err != nil {
			lastErr = fmt.Errorf("failed to read response: %w", err)
			c.circuitBreaker.Failure()
			continue
		}
		
		// 成功
		c.circuitBreaker.Success()
		return buffer[:n], nil
	}
	
	// すべての再試行が失敗
	return nil, fmt.Errorf("all retries failed: %w", lastErr)
}

// クライアント終了
func (c *EnhancedTCPClient) Close() {
	c.pool.Close()
}

// 使用例
func main() {
	client := NewEnhancedTCPClient("localhost:8080")
	defer client.Close()
	
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	
	// メッセージ送信と応答受信
	response, err := client.SendWithRetry(ctx, []byte("Hello, Server!"))
	if err != nil {
		log.Fatalf("通信エラー: %v", err)
	}
	
	log.Printf("応答受信: %s", string(response))
	
	// サーキットブレーカー状態の確認

	log.Printf("サーキットブレーカー状態: %v", metrics)
}
```

```go
package main

import (
	"bufio"
	"bytes"
	"context"

	"errors"
	"fmt"
	"hash/crc32"
	"io"
	"log"

	"sync"

// ==== メッセージフレーミングプロトコル ====

const (
	// プロトコル定数
	ProtocolVersion  byte = 1
	HeaderSize            = 12 // 1(version) + 1(flags) + 2(command) + 4(length) + 4(checksum)
	MaxMessageSize        = 10 * 1024 * 1024 // 10MB最大メッセージサイズ
	
	// フラグ
	FlagCompressed  byte = 1 << 0
	FlagEncrypted   byte = 1 << 1
	FlagHighPriority byte = 1 << 2
	
	// コマンド
	CmdRequest      uint16 = 1
	CmdResponse     uint16 = 2
	CmdPing         uint16 = 3
	CmdPong         uint16 = 4
	CmdError        uint16 = 5
	CmdClose        uint16 = 6
)

// メッセージ構造体
type Message struct {
	Version  byte
	Flags    byte
	Command  uint16
	Length   uint32
	Checksum uint32
	Payload  []byte
}

// メッセージをバイト列にエンコード
func (m *Message) Encode() ([]byte, error) {
	// ペイロードサイズチェック
	if len(m.Payload) > MaxMessageSize {
		return nil, fmt.Errorf("message payload too large: %d > %d", len(m.Payload), MaxMessageSize)
	}
	
	// バッファ確保
	buf := bytes.NewBuffer(make([]byte, 0, HeaderSize+len(m.Payload)))
	
	// バージョンとフラグを書き込み
	if err := buf.WriteByte(m.Version); err != nil {
		return nil, err
	}
	if err := buf.WriteByte(m.Flags); err != nil {
		return nil, err
	}
	
	// コマンドを書き込み
	if err := binary.Write(buf, binary.BigEndian, m.Command); err != nil {
		return nil, err
	}
	
	// ペイロード長を書き込み
	m.Length = uint32(len(m.Payload))
	if err := binary.Write(buf, binary.BigEndian, m.Length); err != nil {
		return nil, err
	}
	
	// CRC32チェックサム計算
	m.Checksum = crc32.ChecksumIEEE(m.Payload)
	if err := binary.Write(buf, binary.BigEndian, m.Checksum); err != nil {
		return nil, err
	}
	
	// ペイロード書き込み
	if _, err := buf.Write(m.Payload); err != nil {
		return nil, err
	}
	
	return buf.Bytes(), nil
}

// バイト列からメッセージをデコード
func DecodeMessage(data []byte) (*Message, error) {
	if len(data) < HeaderSize {
		return nil, errors.New("data too short for header")
	}
	
	msg := &Message{}
	
	// ヘッダー解析
	msg.Version = data[0]
	msg.Flags = data[1]
	
	msg.Command = binary.BigEndian.Uint16(data[2:4])
	msg.Length = binary.BigEndian.Uint32(data[4:8])
	msg.Checksum = binary.BigEndian.Uint32(data[8:12])
	
	// ペイロードサイズチェック
	if msg.Length > MaxMessageSize {
		return nil, fmt.Errorf("message payload too large: %d > %d", msg.Length, MaxMessageSize)
	}
	
	// ペイロード長検証
	expectedLen := HeaderSize + int(msg.Length)
	if len(data) < expectedLen {
		return nil, fmt.Errorf("data too short for payload: %d < %d", len(data), expectedLen)
	}
	
	// ペイロード取得
	msg.Payload = data[HeaderSize:expectedLen]
	
	// チェックサム検証
	actualChecksum := crc32.ChecksumIEEE(msg.Payload)
	if actualChecksum != msg.Checksum {
		return nil, fmt.Errorf("checksum mismatch: expected %d, got %d", msg.Checksum, actualChecksum)
	}
	
	return msg, nil
}

// ==== フレーミング処理 ====

// メッセージ読み込み
func ReadMessage(r io.Reader) (*Message, error) {
	// ヘッダーバッファ
	headerBuf := make([]byte, HeaderSize)
	
	// ヘッダー読み込み
	_, err := io.ReadFull(r, headerBuf)
	if err != nil {
		return nil, fmt.Errorf("failed to read header: %w", err)
	}
	
	// 長さフィールド取得
	payloadLen := binary.BigEndian.Uint32(headerBuf[4:8])
	
	if payloadLen > MaxMessageSize {
		return nil, fmt.Errorf("message too large: %d > %d", payloadLen, MaxMessageSize)
	}
	
	// ペイロード読み込み
	payloadBuf := make([]byte, payloadLen)
	_, err = io.ReadFull(r, payloadBuf)
	if err != nil {
		return nil, fmt.Errorf("failed to read payload: %w", err)
	}
	
	// 完全なメッセージ構築
	completeMsg := make([]byte, HeaderSize+payloadLen)
	copy(completeMsg, headerBuf)
	copy(completeMsg[HeaderSize:], payloadBuf)
	
	// メッセージデコード
	return DecodeMessage(completeMsg)
}

// メッセージ書き込み
func WriteMessage(w io.Writer, msg *Message) error {
	// メッセージエンコード
	data, err := msg.Encode()
	if err != nil {
		return err
	}
	
	// 一度に書き込み
	_, err = w.Write(data)
	return err
}

// ==== プロトコル対応TCPクライアント ====

type TCPProtocolClient struct {
	conn            net.Conn
	readTimeout     time.Duration
	writeTimeout    time.Duration
	mu              sync.Mutex
	reader          *bufio.Reader
	writer          *bufio.Writer
	lastActivity    time.Time
	pingInterval    time.Duration
	closed          bool
}

func NewTCPProtocolClient(conn net.Conn) *TCPProtocolClient {
	return &TCPProtocolClient{
		conn:         conn,
		readTimeout:  30 * time.Second,
		writeTimeout: 10 * time.Second,
		reader:       bufio.NewReader(conn),
		writer:       bufio.NewWriter(conn),
		pingInterval: 15 * time.Second,
		lastActivity: time.Now(),
	}
}

// 接続管理（キープアライブを含む）
func (c *TCPProtocolClient) Manage(ctx context.Context) {
	ticker := time.NewTicker(c.pingInterval)
	defer ticker.Stop()
	
	for {
		select {
		case <-ctx.Done():
			return
		case <-ticker.C:
			// アイドル時間をチェック
			c.mu.Lock()
			idleTime := time.Since(c.lastActivity)
			c.mu.Unlock()
			
			// キープアライブの送信（長時間アイドル状態の場合）
			if idleTime > c.pingInterval && !c.closed {
				if err := c.Ping(ctx); err != nil {
					log.Printf("キープアライブ失敗: %v", err)
					c.Close()
					return
				}
			}
		}
	}
}

// リクエスト送信と応答受信
func (c *TCPProtocolClient) Request(ctx context.Context, payload []byte, flags byte) ([]byte, error) {
	c.mu.Lock()
	defer c.mu.Unlock()
	
	if c.closed {
		return nil, errors.New("connection closed")
	}
	
	// アクティビティ記録
	c.lastActivity = time.Now()
	
	// リクエストメッセージ作成
	req := &Message{
		Version: ProtocolVersion,
		Flags:   flags,
		Command: CmdRequest,
		Payload: payload,
	}
	
	// コンテキストからタイムアウト取得
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(c.writeTimeout)
	}
	
	// 書き込みタイムアウト設定
	if err := c.conn.SetWriteDeadline(deadline); err != nil {
		return nil, fmt.Errorf("failed to set write deadline: %w", err)
	}
	
	// メッセージ送信
	if err := WriteMessage(c.writer, req); err != nil {
		return nil, fmt.Errorf("failed to write request: %w", err)
	}
	
	// バッファフラッシュ
	if err := c.writer.Flush(); err != nil {
		return nil, fmt.Errorf("failed to flush writer: %w", err)
	}
	
	// 読み込みタイムアウト設定
	if err := c.conn.SetReadDeadline(deadline); err != nil {
		return nil, fmt.Errorf("failed to set read deadline: %w", err)
	}
	
	// 応答受信
	resp, err := ReadMessage(c.reader)
	if err != nil {
		return nil, fmt.Errorf("failed to read response: %w", err)
	}
	
	// コマンド検証
	switch resp.Command {
	case CmdResponse:
		return resp.Payload, nil
	case CmdError:
		return nil, fmt.Errorf("server error: %s", string(resp.Payload))
	default:
		return nil, fmt.Errorf("unexpected response command: %d", resp.Command)
	}
}

// Ping送信（キープアライブ）
func (c *TCPProtocolClient) Ping(ctx context.Context) error {
	c.mu.Lock()
	defer c.mu.Unlock()
	
	if c.closed {
		return errors.New("connection closed")
	}
	
	// アクティビティ記録
	c.lastActivity = time.Now()
	
	// Pingメッセージ作成
	ping := &Message{
		Version: ProtocolVersion,
		Flags:   0,
		Command: CmdPing,
		Payload: []byte{},
	}
	
	// コンテキストからタイムアウト取得
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(5 * time.Second) // Pingには短いタイムアウト
	}
	
	// 書き込みタイムアウト設定
	if err := c.conn.SetWriteDeadline(deadline); err != nil {
		return fmt.Errorf("failed to set write deadline: %w", err)
	}
	
	// Ping送信
	if err := WriteMessage(c.writer, ping); err != nil {
		return fmt.Errorf("failed to write ping: %w", err)
	}
	
	// バッファフラッシュ
	if err := c.writer.Flush(); err != nil {
		return fmt.Errorf("failed to flush writer: %w", err)
	}
	
	// 読み込みタイムアウト設定
	if err := c.conn.SetReadDeadline(deadline); err != nil {
		return fmt.Errorf("failed to set read deadline: %w", err)
	}
	
	// Pong受信
	resp, err := ReadMessage(c.reader)
	if err != nil {
		return fmt.Errorf("failed to read pong: %w", err)
	}
	
	// Pong確認
	if resp.Command != CmdPong {
		return fmt.Errorf("unexpected response to ping: %d", resp.Command)
	}
	
	return nil
}

// 接続を閉じる
func (c *TCPProtocolClient) Close() error {
	c.mu.Lock()
	defer c.mu.Unlock()
	
	if c.closed {
		return nil
	}
	
	c.closed = true
	
	// クローズメッセージ送信（ベストエフォート）
	closeMsg := &Message{
		Version: ProtocolVersion,
		Flags:   0,
		Command: CmdClose,
		Payload: []byte{},
	}
	
	// 短いタイムアウトでクローズメッセージ送信試行
	c.conn.SetWriteDeadline(time.Now().Add(1 * time.Second))
	WriteMessage(c.writer, closeMsg)
	c.writer.Flush()
	
	return c.conn.Close()
}
```

```go
type QualityLevel int32

const (
	QualityHigh     QualityLevel = iota // 高品質（フル機能）

// 品質レベルごとの設定

// ==== 適応型TCPクライアント ====

type AdaptiveTCPClient struct {
	// 接続設定
	address          string
	baseConnectTimeout time.Duration
	
	// 品質管理
	currentQuality   QualityLevel
	qualityMutex     sync.RWMutex
	successCount     int64
	failureCount     int64
	latencies        []time.Duration
	latencyWindow    int // 移動平均ウィンドウサイズ
	latencyMutex     sync.Mutex
	
	// 品質変更閾値設定
	highToMediumThreshold   float64 // この失敗率を超えるとミディアム品質に下がる
	mediumToLowThreshold    float64
	lowToMinimalThreshold   float64
	recoverySuccessThreshold float64 // この成功率を超えると品質レベルアップ
	
	// TCP接続
	conn             net.Conn
	connMutex        sync.Mutex
	reconnectWait    time.Duration
	
	// イベント通知
	qualityChanged   chan QualityLevel
}

func NewAdaptiveTCPClient(address string) *AdaptiveTCPClient {
	client := &AdaptiveTCPClient{
		address:           address,
		baseConnectTimeout: 5 * time.Second,
		currentQuality:    QualityHigh,
		latencyWindow:     50,
		
		// 閾値設定
		highToMediumThreshold:   0.1,  // 10%の失敗率
		mediumToLowThreshold:    0.3,  // 30%の失敗率
		lowToMinimalThreshold:   0.5,  // 50%の失敗率
		recoverySuccessThreshold: 0.95, // 95%の成功率
		
		reconnectWait:     5 * time.Second,
		qualityChanged:    make(chan QualityLevel, 5),
	}
	
	// 品質モニタリング開始
	go client.monitorQuality()
	
	return client
}

// 品質モニタリング
func (c *AdaptiveTCPClient) monitorQuality() {
	ticker := time.NewTicker(10 * time.Second)
	defer ticker.Stop()
	
	for range ticker.C {
		c.evaluateQuality()
	}
}

// 品質評価と調整
func (c *AdaptiveTCPClient) evaluateQuality() {
	// 現在の成功率計算
	success := atomic.LoadInt64(&c.successCount)
	failure := atomic.LoadInt64(&c.failureCount)
	total := success + failure
	
	if total < 10 {
		return // 十分なサンプルがない
	}
	
	successRate := float64(success) / float64(total)
	
	// 平均レイテンシ計算
	c.latencyMutex.Lock()
	var avgLatency time.Duration
	if len(c.latencies) > 0 {
		var sum time.Duration
		for _, l := range c.latencies {
			sum += l
		}
		avgLatency = sum / time.Duration(len(c.latencies))
	}
	c.latencyMutex.Unlock()
	
	// 品質レベル決定ロジック
	c.qualityMutex.Lock()
	defer c.qualityMutex.Unlock()
	
	currentLevel := c.currentQuality
	var newLevel QualityLevel
	
	// 失敗率ベースの判定
	switch currentLevel {
	case QualityHigh:
		if 1-successRate > c.highToMediumThreshold {
			newLevel = QualityMedium
		} else {
			newLevel = QualityHigh
		}
	case QualityMedium:
		if 1-successRate > c.mediumToLowThreshold {
			newLevel = QualityLow
		} else if successRate > c.recoverySuccessThreshold {
			newLevel = QualityHigh
		} else {
			newLevel = QualityMedium
		}
	case QualityLow:
		if 1-successRate > c.lowToMinimalThreshold {
			newLevel = QualityMinimal
		} else if successRate > c.recoverySuccessThreshold {
			newLevel = QualityMedium
		} else {
			newLevel = QualityLow
		}
	case QualityMinimal:
		if successRate > c.recoverySuccessThreshold {
			newLevel = QualityLow
		} else {
			newLevel = QualityMinimal
		}
	}
	
	// レイテンシベースの追加判定
	if avgLatency > 200*time.Millisecond && newLevel == QualityHigh {
		newLevel = QualityMedium
	} else if avgLatency > 500*time.Millisecond && newLevel > QualityLow {
		newLevel = QualityLow
	} else if avgLatency > 1000*time.Millisecond {
		newLevel = QualityMinimal
	}
	
	// 品質レベル変更があれば適用
	if newLevel != currentLevel {
		log.Printf("品質レベル変更: %v -> %v (成功率: %.2f%%, 平均レイテンシ: %v)", 
			currentLevel, newLevel, successRate*100, avgLatency)
		c.currentQuality = newLevel
		
		// カウンタリセット
		atomic.StoreInt64(&c.successCount, 0)
		atomic.StoreInt64(&c.failureCount, 0)
		
		// 変更通知
		select {
		case c.qualityChanged <- newLevel:
		default:
			// チャネルがいっぱいの場合はドロップ
		}
	}
}

// 接続取得（必要に応じて再接続）
func (c *AdaptiveTCPClient) getConnection(ctx context.Context) (net.Conn, error) {
	c.connMutex.Lock()
	defer c.connMutex.Unlock()
	
	// 既存の接続がある場合はチェック
	if c.conn != nil {
		// 単純な接続チェック
		one := []byte{0}
		c.conn.SetWriteDeadline(time.Now().Add(1 * time.Second))
		_, err := c.conn.Write(one)
		if err == nil {
			return c.conn, nil
		}
		
		// 接続障害
		log.Printf("既存の接続が無効: %v", err)
		c.conn.Close()
		c.conn = nil
	}
	
	// 現在の品質レベルに基づくタイムアウト設定
	c.qualityMutex.RLock()
	qualitySettings := GetQualitySettings(c.currentQuality)
	c.qualityMutex.RUnlock()
	
	// 新しい接続
	dialer := net.Dialer{
		Timeout: qualitySettings.Timeout,
	}
	
	var err error
	c.conn, err = dialer.DialContext(ctx, "tcp", c.address)
	if err != nil {
		atomic.AddInt64(&c.failureCount, 1)
		return nil, fmt.Errorf("接続失敗: %w", err)
	}
	
	// TCP設定
	if tcpConn, ok := c.conn.(*net.TCPConn); ok {
		tcpConn.SetKeepAlive(true)
		tcpConn.SetKeepAlivePeriod(qualitySettings.KeepAliveInterval)
		tcpConn.SetNoDelay(true)
		tcpConn.SetReadBuffer(qualitySettings.BufferSize)
		tcpConn.SetWriteBuffer(qualitySettings.BufferSize)
	}
	
	log.Printf("新しい接続を確立しました: %s", c.address)
	return c.conn, nil
}

// 送信処理（品質レベルに適応）
func (c *AdaptiveTCPClient) Send(ctx context.Context, data []byte) error {
	startTime := time.Now()
	
	// 品質設定取得
	c.qualityMutex.RLock()
	qualitySettings := GetQualitySettings(c.currentQuality)
	c.qualityMutex.RUnlock()
	
	// 接続取得
	conn, err := c.getConnection(ctx)
	if err != nil {
		atomic.AddInt64(&c.failureCount, 1)
		return err
	}
	
	// 書き込みタイムアウト設定
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(qualitySettings.Timeout)
	}
	
	if err := conn.SetWriteDeadline(deadline); err != nil {
		atomic.AddInt64(&c.failureCount, 1)
		return fmt.Errorf("タイムアウト設定エラー: %w", err)
	}
	
	// データ送信
	_, err = conn.Write(data)
	if err != nil {
		atomic.AddInt64(&c.failureCount, 1)
		return fmt.Errorf("送信エラー: %w", err)
	}
	
	// 成功カウント
	atomic.AddInt64(&c.successCount, 1)
	
	// レイテンシ記録
	latency := time.Since(startTime)
	c.recordLatency(latency)
	
	return nil
}

// 受信処理（品質レベルに適応）
func (c *AdaptiveTCPClient) Receive(ctx context.Context) ([]byte, error) {
	startTime := time.Now()
	
	// 品質設定取得
	c.qualityMutex.RLock()
	qualitySettings := GetQualitySettings(c.currentQuality)
	c.qualityMutex.RUnlock()
	
	// 接続取得
	conn, err := c.getConnection(ctx)
	if err != nil {
		atomic.AddInt64(&c.failureCount, 1)
		return nil, err
	}
	
	// 読み込みタイムアウト設定
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(qualitySettings.Timeout)
	}
	
	if err := conn.SetReadDeadline(deadline); err != nil {
		atomic.AddInt64(&c.failureCount, 1)
		return nil, fmt.Errorf("タイムアウト設定エラー: %w", err)
	}
	
	// データ受信
	buffer := make([]byte, qualitySettings.BufferSize)
	n, err := conn.Read(buffer)
	if err != nil {
		atomic.AddInt64(&c.failureCount, 1)
		return nil, fmt.Errorf("受信エラー: %w", err)
	}
	
	// 成功カウント
	atomic.AddInt64(&c.successCount, 1)
	
	// レイテンシ記録
	latency := time.Since(startTime)
	c.recordLatency(latency)
	
	return buffer[:n], nil
}

// レイテンシ記録
func (c *AdaptiveTCPClient) recordLatency(latency time.Duration) {
	c.latencyMutex.Lock()
	defer c.latencyMutex.Unlock()
	
	// 移動平均ウィンドウ
	c.latencies = append(c.latencies, latency)
	if len(c.latencies) > c.latencyWindow {
		c.latencies = c.latencies[1:]
	}
}

// 品質レベル取得
func (c *AdaptiveTCPClient) GetQualityLevel() QualityLevel {
	c.qualityMutex.RLock()
	defer c.qualityMutex.RUnlock()
	return c.currentQuality
}

// 品質レベル強制設定
func (c *AdaptiveTCPClient) SetQualityLevel(level QualityLevel) {
	c.qualityMutex.Lock()
	defer c.qualityMutex.Unlock()
	
	if c.currentQuality != level {
		log.Printf("品質レベル手動変更: %v -> %v", c.currentQuality, level)
		c.currentQuality = level
		
		// 通知
		select {
		case c.qualityChanged <- level:
		default:
		}
	}
}

// 接続終了
func (c *AdaptiveTCPClient) Close() error {
	c.connMutex.Lock()
	defer c.connMutex.Unlock()
	
	if c.conn != nil {
		err := c.conn.Close()
		c.conn = nil
		return err
	}
	return nil
}

// ==== ジッター測定とネットワーク品質メトリクス ====

type NetworkMetrics struct {
	// 基本メトリクス
	latencyAvg       float64 // ミリ秒
	latencyStdDev    float64 // ミリ秒
	packetLossRate   float64 // 0.0-1.0
	jitter           float64 // ミリ秒（標準偏差）
	
	// 計算用データ
	latencySamples   []float64
	sampleCount      int
	lossCount        int
	sentCount        int
	
	mutex            sync.Mutex
}

func NewNetworkMetrics(windowSize int) *NetworkMetrics {
	return &NetworkMetrics{
		latencySamples: make([]float64, 0, windowSize),
		sampleCount:    windowSize,
	}
}

// レイテンシサンプル追加
func (nm *NetworkMetrics) AddLatencySample(latency time.Duration) {
	nm.mutex.Lock()
	defer nm.mutex.Unlock()
	
	latencyMs := float64(latency.Milliseconds())
	
	nm.latencySamples = append(nm.latencySamples, latencyMs)
	if len(nm.latencySamples) > nm.sampleCount {
		nm.latencySamples = nm.latencySamples[1:]
	}
	
	// メトリクス更新
	nm.updateMetrics()
}

// 送信パケットカウント
func (nm *NetworkMetrics) PacketSent() {
	nm.mutex.Lock()
	defer nm.mutex.Unlock()
	
	nm.sentCount++
}

// パケットロスカウント
func (nm *NetworkMetrics) PacketLost() {
	nm.mutex.Lock()
	defer nm.mutex.Unlock()
	
	nm.lossCount++
	nm.sentCount++
	
	// ロス率更新
	if nm.sentCount > 0 {
		nm.packetLossRate = float64(nm.lossCount) / float64(nm.sentCount)
	}
}

// メトリクス計算
func (nm *NetworkMetrics) updateMetrics() {
	sampleCount := len(nm.latencySamples)
	if sampleCount == 0 {
		return
	}
	
	// 平均計算
	var sum float64
	for _, l := range nm.latencySamples {
		sum += l
	}
	nm.latencyAvg = sum / float64(sampleCount)
	
	// 標準偏差（ジッター）計算
	var sqSum float64
	for _, l := range nm.latencySamples {
		diff := l - nm.latencyAvg
		sqSum += diff * diff
	}
	
	nm.latencyStdDev = math.Sqrt(sqSum / float64(sampleCount))
	nm.jitter = nm.latencyStdDev
}

// メトリクス取得
func (nm *NetworkMetrics) GetMetrics() map[string]float64 {
	nm.mutex.Lock()
	defer nm.mutex.Unlock()
	
	return map[string]float64{
		"latencyAvg":     nm.latencyAvg,
		"latencyStdDev":  nm.latencyStdDev,
		"packetLossRate": nm.packetLossRate,
		"jitter":         nm.jitter,
	}
}

func main() {
	// 適応型クライアント作成
	client := NewAdaptiveTCPClient("localhost:8080")

		startTime := time.Now()

			log.Printf("送信エラー: %v", err)
```

```go
		circuitBreakerState: promauto.NewGaugeVec(
			prometheus.GaugeOpts{
				Namespace: namespace,
				Name:      "circuit_breaker_state",
				Help:      "Circuit breaker state (0=closed, 1=half-open, 2=open)",
			},
			[]string{"remote_host", "port"},
		),

		// システムメトリクス
		cpuUsage: promauto.NewGauge(
			prometheus.GaugeOpts{
				Namespace: namespace,
				Name:      "cpu_usage_percent",
				Help:      "Current CPU usage percentage",
			},
		),
		memoryUsage: promauto.NewGauge(
			prometheus.GaugeOpts{
				Namespace: namespace,
				Name:      "memory_usage_bytes",
				Help:      "Current memory usage in bytes",
			},
		),
		goroutineCount: promauto.NewGauge(
			prometheus.GaugeOpts{
				Namespace: namespace,
				Name:      "goroutine_count",
				Help:      "Current number of goroutines",
			},
		),
		gcPauseTime: promauto.NewHistogram(

	go metrics.collectSystemMetrics()

	return metrics

// システムメトリクス収集
func (m *TCPMetrics) collectSystemMetrics() {
	ticker := time.NewTicker(5 * time.Second)
	defer ticker.Stop()

	var lastPauseNs uint64
	var lastNumGC uint32

	for range ticker.C {
		// CPU使用率
		cpuPercent, err := cpu.Percent(0, false)
		if err == nil && len(cpuPercent) > 0 {
			m.cpuUsage.Set(cpuPercent[0])
		}

		// メモリ使用量
		memStat, err := mem.VirtualMemory()
		if err == nil {
			m.memoryUsage.Set(float64(memStat.Used))
		}

		// ゴルーチン数
		m.goroutineCount.Set(float64(runtime.NumGoroutine()))

		// GC統計
		var memStats runtime.MemStats
		runtime.ReadMemStats(&memStats)

		// GCポーズタイム
		if lastPauseNs != 0 && memStats.PauseTotalNs > lastPauseNs {
			pauseNs := memStats.PauseTotalNs - lastPauseNs
			m.gcPauseTime.Observe(float64(pauseNs) / 1e9) // ナノ秒から秒へ変換
		}
		lastPauseNs = memStats.PauseTotalNs

		// GC回数とポーズタイム
		if lastNumGC != 0 && memStats.NumGC > lastNumGC {
			for i := lastNumGC; i < memStats.NumGC; i++ {
				idx := i % 256
				pause := float64(memStats.PauseNs[idx]) / 1e9 // ナノ秒から秒へ変換
				m.gcPauseTime.Observe(pause)
			}
		}
		lastNumGC = memStats.NumGC
	}
}

type InstrumentedTCPClient struct {
	address     string
	timeout     time.Duration
	metrics     *TCPMetrics
	conn        net.Conn
	connMutex   sync.Mutex
	connStart   time.Time
}

func NewInstrumentedTCPClient(address string, metrics *TCPMetrics) *InstrumentedTCPClient {
	return &InstrumentedTCPClient{
		address:   address,
		timeout:   10 * time.Second,
		metrics:   metrics,
	}
}

// TCP接続確立
func (c *InstrumentedTCPClient) Connect(ctx context.Context) error {
	c.connMutex.Lock()
	defer c.connMutex.Unlock()
	
	// 既存の接続があれば閉じる
	if c.conn != nil {
		duration := time.Since(c.connStart)
		c.metrics.ConnectionClosed(c.address, duration)
		c.conn.Close()
		c.conn = nil
	}
	
	// 接続試行記録
	c.metrics.ConnectionAttempt(c.address)
	
	// 接続タイムアウト設定
	dialer := net.Dialer{Timeout: c.timeout}
	
	var err error
	c.conn, err = dialer.DialContext(ctx, "tcp", c.address)
	if err != nil {
		c.metrics.ConnectionFailure(c.address, err)
		return fmt.Errorf("failed to connect to %s: %w", c.address, err)
	}
	
	// 接続成功記録
	c.metrics.ConnectionSuccess(c.address)
	c.connStart = time.Now()
	
	return nil
}

// TCP経由でデータ送信
func (c *InstrumentedTCPClient) Send(ctx context.Context, data []byte) error {
	c.connMutex.Lock()
	defer c.connMutex.Unlock()
	
	// 接続確認
	if c.conn == nil {
		return fmt.Errorf("not connected")
	}
	
	// メッセージタイプ判別（実際のアプリではより具体的な判別方法を）
	msgType := "data"
	if len(data) > 0 && data[0] == 1 {
		msgType = "control"
	}
	
	// 送信開始時間
	startTime := time.Now()
	
	// 書き込みタイムアウト設定
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(c.timeout)
	}
	if err := c.conn.SetWriteDeadline(deadline); err != nil {
		return fmt.Errorf("failed to set write deadline: %w", err)
	}
	
	// データ送信
	n, err := c.conn.Write(data)
	
	// 送信時間記録
	sendTime := time.Since(startTime)
	c.metrics.SendLatency(c.address, sendTime)
	
	if err != nil {
		c.metrics.SendError(c.address, err)
		return fmt.Errorf("failed to send data: %w", err)
	}
	
	// メトリクス更新
	c.metrics.BytesSent(c.address, n)
	c.metrics.MessageSent(c.address, msgType)
	
	return nil
}

// TCP経由でデータ受信
func (c *InstrumentedTCPClient) Receive(ctx context.Context, maxSize int) ([]byte, error) {
	c.connMutex.Lock()
	defer c.connMutex.Unlock()
	
	// 接続確認
	if c.conn == nil {
		return nil, fmt.Errorf("not connected")
	}
	
	// 受信開始時間
	startTime := time.Now()
	
	// 読み込みタイムアウト設定
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(c.timeout)
	}
	if err := c.conn.SetReadDeadline(deadline); err != nil {
		return nil, fmt.Errorf("failed to set read deadline: %w", err)
	}
	
	// データ受信
	buffer := make([]byte, maxSize)
	n, err := c.conn.Read(buffer)
	
	// 受信時間記録
	receiveTime := time.Since(startTime)
	c.metrics.ReceiveLatency(c.address, receiveTime)
	
	if err != nil {
		c.metrics.ReceiveError(c.address, err)
		return nil, fmt.Errorf("failed to receive data: %w", err)
	}
	
	// メッセージタイプ判別
	msgType := "data"
	if n > 0 && buffer[0] == 1 {
		msgType = "control"
	}
	
	// メトリクス更新
	c.metrics.BytesReceived(c.address, n)
	c.metrics.MessageReceived(c.address, msgType)
	
	return buffer[:n], nil
}

// リクエスト／レスポンスパターン
func (c *InstrumentedTCPClient) SendAndReceive(ctx context.Context, data []byte, maxResponseSize int) ([]byte, error) {
	startTime := time.Now()
	
	// 送信
	if err := c.Send(ctx, data); err != nil {
		return nil, err
	}
	
	// 受信
	response, err := c.Receive(ctx, maxResponseSize)
	
	// RTT記録
	rtt := time.Since(startTime)
	c.metrics.RoundTripTime(c.address, rtt)
	
	return response, err
}

// 接続終了
func (c *InstrumentedTCPClient) Close() error {
	c.connMutex.Lock()
	defer c.connMutex.Unlock()
	
	if c.conn != nil {
		duration := time.Since(c.connStart)
		c.metrics.ConnectionClosed(c.address, duration)
		
		err := c.conn.Close()
		c.conn = nil
		return err
	}
	
	return nil
}
```

```go
// エラータイプ
type ErrorType int

const (

	ErrorPermanent                  // 永続的なエラー（再試行不可）

// カスタムエラー構造体
type TCPError struct {

	Code    int
	Message string
	Err     error

func (e *TCPError) Error() string {
	if e.Err != nil {
		return fmt.Sprintf("[%d] %s: %v", e.Code, e.Message, e.Err)
	}
	return fmt.Sprintf("[%d] %s", e.Code, e.Message)

func (e *TCPError) Unwrap() error {

// エラーコード定義
const (
	ErrCodeUnknown         = 1000
	ErrCodeConnection      = 1001
	ErrCodeTimeout         = 1002
	ErrCodeIO              = 1003
	ErrCodeProtocol        = 1100
	ErrCodeAuthentication  = 1200
	ErrCodeAuthorization   = 1201
	ErrCodeResource        = 1300
	ErrCodeRateLimited     = 1301
	ErrCodeCircuitOpen     = 1400

// エラー分類関数
func classifyError(err error) (ErrorType, int) {
	if err == nil {
		return ErrorPermanent, 0
	}
	
	// 既に分類済みの場合
	var tcpErr *TCPError
	if errors.As(err, &tcpErr) {
		return tcpErr.Type, tcpErr.Code
	}
	
	// タイムアウトエラー
	var netErr net.Error
	if errors.As(err, &netErr) && netErr.Timeout() {
		return ErrorTemporary, ErrCodeTimeout
	}
	
	// 接続関連エラー
	if errors.Is(err, syscall.ECONNRESET) ||
	   errors.Is(err, syscall.ECONNABORTED) ||
	   errors.Is(err, syscall.ECONNREFUSED) {
		return ErrorTemporary, ErrCodeConnection
	}
	
	// IO関連エラー
	if errors.Is(err, io.EOF) ||
	   errors.Is(err, io.ErrUnexpectedEOF) {
		return ErrorTemporary, ErrCodeIO
	}
	
	// リソース制約エラー
	if errors.Is(err, syscall.EMFILE) ||
	   errors.Is(err, syscall.ENFILE) ||
	   errors.Is(err, syscall.ENOBUFS) ||
	   errors.Is(err, syscall.ENOMEM) {
		return ErrorResource, ErrCodeResource
	}
	
	// デフォルト
	return ErrorTemporary, ErrCodeUnknown
}

// ========== メッセージプロトコル ==========

// プロトコル定数
const (
	ProtocolVersion    byte = 1
	ProtocolHeaderSize      = 16 // バージョン(1) + フラグ(1) + コマンド(2) + 長さ(4) + シーケンス番号(4) + チェックサム(4)
	MaxMessageSize          = 10 * 1024 * 1024 // 10MB
	
	// フラグ定義
	FlagCompressed    byte = 1 << 0 // 圧縮されている
	FlagEncrypted     byte = 1 << 1 // 暗号化されている
	FlagHighPriority  byte = 1 << 2 // 高優先度
	FlagRequiresAck   byte = 1 << 3 // 確認応答が必要
	FlagIsAck         byte = 1 << 4 // これは確認応答
	FlagIsKeepAlive   byte = 1 << 5 // キープアライブメッセージ
	FlagIsError       byte = 1 << 6 // エラーメッセージ
	FlagReserved      byte = 1 << 7 // 予約済み
	
	// コマンド定義
	CmdData           uint16 = 0x0001 // データ転送
	CmdControl        uint16 = 0x0002 // 制御メッセージ
	CmdHandshake      uint16 = 0x0003 // ハンドシェイク
	CmdHeartbeat      uint16 = 0x0004 // ハートビート
	CmdError          uint16 = 0x00FF // エラー
)

// メッセージ構造体
type Message struct {
	Version       byte
	Flags         byte
	Command       uint16
	Length        uint32
	SequenceNum   uint32
	Checksum      uint32
	Payload       []byte
	ArrivalTime   time.Time // 内部使用のみ
}

// エンコード
func (m *Message) Encode() ([]byte, error) {
	if len(m.Payload) > MaxMessageSize {
		return nil, &TCPError{
			Type:    ErrorProtocol,
			Code:    ErrCodeProtocol,
			Message: "メッセージサイズが上限を超えています",
		}
	}
	
	buf := bytes.NewBuffer(make([]byte, 0, ProtocolHeaderSize+len(m.Payload)))
	
	// ヘッダー書き込み
	if err := buf.WriteByte(m.Version); err != nil {
		return nil, err
	}
	if err := buf.WriteByte(m.Flags); err != nil {
		return nil, err
	}
	
	if err := binary.Write(buf, binary.BigEndian, m.Command); err != nil {
		return nil, err
	}
	
	m.Length = uint32(len(m.Payload))
	if err := binary.Write(buf, binary.BigEndian, m.Length); err != nil {
		return nil, err
	}
	
	if err := binary.Write(buf, binary.BigEndian, m.SequenceNum); err != nil {
		return nil, err
	}
	
	// チェックサム計算（ペイロードのみ）
	m.Checksum = crc32.ChecksumIEEE(m.Payload)
	if err := binary.Write(buf, binary.BigEndian, m.Checksum); err != nil {
		return nil, err
	}
	
	// ペイロード書き込み
	if _, err := buf.Write(m.Payload); err != nil {
		return nil, err
	}
	
	return buf.Bytes(), nil
}

// デコード
func DecodeMessage(data []byte) (*Message, error) {
	if len(data) < ProtocolHeaderSize {
		return nil, &TCPError{
			Type:    ErrorProtocol,
			Code:    ErrCodeProtocol,
			Message: "不完全なメッセージヘッダー",
		}
	}
	
	msg := &Message{
		ArrivalTime: time.Now(),
	}
	
	// ヘッダー読み取り
	msg.Version = data[0]
	msg.Flags = data[1]
	
	msg.Command = binary.BigEndian.Uint16(data[2:4])
	msg.Length = binary.BigEndian.Uint32(data[4:8])
	msg.SequenceNum = binary.BigEndian.Uint32(data[8:12])
	msg.Checksum = binary.BigEndian.Uint32(data[12:16])
	
	// ペイロードサイズ検証
	if msg.Length > MaxMessageSize {
		return nil, &TCPError{
			Type:    ErrorProtocol,
			Code:    ErrCodeProtocol,
			Message: fmt.Sprintf("メッセージサイズが上限を超えています: %d > %d", msg.Length, MaxMessageSize),
		}
	}
	
	expectedLen := ProtocolHeaderSize + int(msg.Length)
	if len(data) < expectedLen {
		return nil, &TCPError{
			Type:    ErrorProtocol,
			Code:    ErrCodeProtocol,
			Message: fmt.Sprintf("不完全なメッセージペイロード: %d < %d", len(data), expectedLen),
		}
	}
	
	// ペイロード取得
	msg.Payload = data[ProtocolHeaderSize:expectedLen]
	
	// チェックサム検証
	if msg.Length > 0 {
		actualChecksum := crc32.ChecksumIEEE(msg.Payload)
		if actualChecksum != msg.Checksum {
			return nil, &TCPError{
				Type:    ErrorProtocol,
				Code:    ErrCodeProtocol,
				Message: fmt.Sprintf("チェックサムエラー: 期待値 %d, 実際 %d", msg.Checksum, actualChecksum),
			}
		}
	}
	
	return msg, nil
}

// メッセージ読み込み
func ReadMessage(reader io.Reader) (*Message, error) {
	// ヘッダー読み込み
	headerBuf := make([]byte, ProtocolHeaderSize)
	_, err := io.ReadFull(reader, headerBuf)
	if err != nil {
		return nil, &TCPError{
			Type:    classifyError(err),
			Code:    ErrCodeIO,
			Message: "ヘッダー読み込みエラー",
			Err:     err,
		}
	}
	
	// 長さフィールド取得
	payloadLen := binary.BigEndian.Uint32(headerBuf[4:8])
	
	if payloadLen > MaxMessageSize {
		return nil, &TCPError{
			Type:    ErrorProtocol,
			Code:    ErrCodeProtocol,
			Message: fmt.Sprintf("メッセージサイズが上限を超えています: %d > %d", payloadLen, MaxMessageSize),
		}
	}
	
	// ペイロード読み込み
	var payloadBuf []byte
	if payloadLen > 0 {
		payloadBuf = make([]byte, payloadLen)
		_, err = io.ReadFull(reader, payloadBuf)
		if err != nil {
			return nil, &TCPError{
				Type:    classifyError(err),
				Code:    ErrCodeIO,
				Message: "ペイロード読み込みエラー",
				Err:     err,
			}
		}
	} else {
		payloadBuf = []byte{}
	}
	
	// 完全なメッセージ構築
	completeMsg := make([]byte, ProtocolHeaderSize+payloadLen)
	copy(completeMsg, headerBuf)
	copy(completeMsg[ProtocolHeaderSize:], payloadBuf)
	
	// メッセージデコード
	return DecodeMessage(completeMsg)
}

// メッセージ書き込み
func WriteMessage(writer io.Writer, msg *Message) error {
	// メッセージエンコード
	data, err := msg.Encode()
	if err != nil {
		return err
	}
	
	// 一度に書き込み
	_, err = writer.Write(data)
	if err != nil {
		return &TCPError{
			Type:    classifyError(err),
			Code:    ErrCodeIO,
			Message: "メッセージ書き込みエラー",
			Err:     err,
		}
	}
	
	return nil
}

// ========== サーキットブレーカー ==========

type CircuitState int32

const (
	StateClosed CircuitState = iota
	StateHalfOpen
	StateOpen
)

type CircuitBreaker struct {
	name              string
	threshold         int32
	halfOpenRequests  int32
	resetTimeout      time.Duration
	
	failures          int32
	successes         int32
	state             CircuitState
	lastStateChange   time.Time
	mu                sync.RWMutex
	
	// テレメトリ
	totalRequests     int64
	totalFailures     int64
	consecutiveSuccess int64
	consecutiveFailure int64
}

func NewCircuitBreaker(name string, threshold, halfOpenRequests int, resetTimeout time.Duration) *CircuitBreaker {
	return &CircuitBreaker{
		name:             name,
		threshold:        int32(threshold),
		halfOpenRequests: int32(halfOpenRequests),
		resetTimeout:     resetTimeout,
		state:            StateClosed,
		lastStateChange:  time.Now(),
	}
}

func (cb *CircuitBreaker) Allow() error {
	cb.mu.Lock()
	defer cb.mu.Unlock()
	
	// リクエスト集計
	atomic.AddInt64(&cb.totalRequests, 1)
	
	now := time.Now()
	
	switch cb.state {
	case StateClosed:
		return nil // 閉回路状態ではすべて許可
		
	case StateOpen:
		// リセットタイムアウト経過で半開状態へ
		if now.Sub(cb.lastStateChange) > cb.resetTimeout {
			cb.setState(StateHalfOpen, now)
			return nil
		}
		
		// それ以外では拒否
		return &TCPError{
			Type:    ErrorTemporary,
			Code:    ErrCodeCircuitOpen,
			Message: fmt.Sprintf("サーキットブレーカーが開いています: %s", cb.name),
		}
		
	case StateHalfOpen:
		// 半開状態では限られた数のリクエストのみ許可
		if cb.successes < cb.halfOpenRequests {
			return nil
		}
		
		// 制限に達した場合は拒否
		return &TCPError{
			Type:    ErrorTemporary,
			Code:    ErrCodeCircuitOpen,
			Message: fmt.Sprintf("半開状態のリクエスト上限に達しました: %s", cb.name),
		}
	}
	
	return nil
}

func (cb *CircuitBreaker) Success() {
	cb.mu.Lock()
	defer cb.mu.Unlock()
	
	// テレメトリ更新
	atomic.AddInt64(&cb.consecutiveSuccess, 1)
	atomic.StoreInt64(&cb.consecutiveFailure, 0)
	
	switch cb.state {
	case StateClosed:
		cb.failures = 0 // 成功したらリセット
		
	case StateHalfOpen:
		cb.successes++
		if cb.successes >= cb.halfOpenRequests {
			// 閾値以上の成功で閉回路に戻す
			cb.setState(StateClosed, time.Now())
		}
	}
}

func (cb *CircuitBreaker) Failure() {
	cb.mu.Lock()
	defer cb.mu.Unlock()
	
	// テレメトリ更新
	atomic.AddInt64(&cb.totalFailures, 1)
	atomic.StoreInt64(&cb.consecutiveSuccess, 0)
	atomic.AddInt64(&cb.consecutiveFailure, 1)
	
	switch cb.state {
	case StateClosed:
		cb.failures++
		if cb.failures >= cb.threshold {
			// 閾値以上の失敗で開回路にする
			cb.setState(StateOpen, time.Now())
		}
		
	case StateHalfOpen:
		// 半開状態での失敗はすぐに開回路に戻す
		cb.setState(StateOpen, time.Now())
	}
}

func (cb *CircuitBreaker) setState(state CircuitState, when time.Time) {
	cb.state = state
	cb.lastStateChange = when
	cb.failures = 0
	cb.successes = 0
}

func (cb *CircuitBreaker) State() CircuitState {
	cb.mu.RLock()
	defer cb.mu.RUnlock()
	return cb.state
}

// テレメトリの取得
func (cb *CircuitBreaker) Metrics() map[string]interface{} {
	cb.mu.RLock()
	defer cb.mu.RUnlock()
	
	return map[string]interface{}{
		"name":                cb.name,
		"state":               cb.state,
		"total_requests":      atomic.LoadInt64(&cb.totalRequests),
		"total_failures":      atomic.LoadInt64(&cb.totalFailures),
		"consecutive_success": atomic.LoadInt64(&cb.consecutiveSuccess),
		"consecutive_failure": atomic.LoadInt64(&cb.consecutiveFailure),
		"time_in_state":       time.Since(cb.lastStateChange).String(),
	}
}

// ========== 接続プール ==========

type PooledConnection struct {
	conn      net.Conn
	createdAt time.Time
	lastUsed  time.Time
	inUse     bool
}

type ConnectionPool struct {
	address      string
	maxSize      int
	minSize      int
	maxLifetime  time.Duration
	maxIdleTime  time.Duration
	
	mu           sync.Mutex
	connections  []*PooledConnection
	connCount    int32
	
	dial         func(ctx context.Context) (net.Conn, error)
	closed       bool
	eventCh      chan struct{}
}

func NewConnectionPool(address string, maxSize int, dial func(ctx context.Context) (net.Conn, error)) *ConnectionPool {
	pool := &ConnectionPool{
		address:     address,
		maxSize:     maxSize,
		minSize:     2,
		maxLifetime: 30 * time.Minute,
		maxIdleTime: 5 * time.Minute,
		connections: make([]*PooledConnection, 0, maxSize),
		dial:        dial,
		eventCh:     make(chan struct{}, 1),
	}
	
	// プール管理ゴルーチン起動
	go pool.maintain()
	
	return pool
}

// プール管理
func (p *ConnectionPool) maintain() {
	ticker := time.NewTicker(30 * time.Second)
	defer ticker.Stop()
	
	for {
		select {
		case <-ticker.C:
			p.cleanIdleConnections()
		case <-p.eventCh:
			if p.closed {
				return
			}
		}
	}
}

// アイドル接続のクリーンアップ
func (p *ConnectionPool) cleanIdleConnections() {
	p.mu.Lock()
	defer p.mu.Unlock()
	
	if p.closed {
		return
	}
	
	now := time.Now()
	kept := make([]*PooledConnection, 0, len(p.connections))
	
	for _, conn := range p.connections {
		if conn.inUse {
			// 使用中の接続は維持
			kept = append(kept, conn)
			continue
		}
		
		// 接続が古すぎる場合は閉じる
		if now.Sub(conn.createdAt) > p.maxLifetime {
			conn.conn.Close()
			atomic.AddInt32(&p.connCount, -1)
			continue
		}
		
		// 長時間未使用の接続は閉じる（ただし最小数は維持）
		if now.Sub(conn.lastUsed) > p.maxIdleTime && len(kept) >= p.minSize {
			conn.conn.Close()
			atomic.AddInt32(&p.connCount, -1)
			continue
		}
		
		kept = append(kept, conn)
	}
	
	p.connections = kept
}

// 接続の取得
func (p *ConnectionPool) GetConnection(ctx context.Context) (net.Conn, error) {
	p.mu.Lock()
	
	if p.closed {
		p.mu.Unlock()
		return nil, errors.New("connection pool is closed")
	}
	
	// 既存の空き接続を探す
	for _, conn := range p.connections {
		if !conn.inUse {
			conn.inUse = true
			conn.lastUsed = time.Now()
			p.mu.Unlock()
			return &PoolConnection{pool: p, conn: conn}, nil
		}
	}
	
	// プールサイズの制限チェック
	currentCount := atomic.LoadInt32(&p.connCount)
	if currentCount >= int32(p.maxSize) {
		p.mu.Unlock()
		return nil, &TCPError{
			Type:    ErrorResource,
			Code:    ErrCodeResource,
			Message: "接続プールが上限に達しています",
		}
	}
	
	// 新しい接続を作成
	p.mu.Unlock()
	
	rawConn, err := p.dial(ctx)
	if err != nil {
		return nil, err
	}
	
	poolConn := &PooledConnection{
		conn:      rawConn,
		createdAt: time.Now(),
		lastUsed:  time.Now(),
		inUse:     true,
	}
	
	// 接続をプールに追加
	p.mu.Lock()
	p.connections = append(p.connections, poolConn)
	atomic.AddInt32(&p.connCount, 1)
	p.mu.Unlock()
	
	return &PoolConnection{pool: p, conn: poolConn}, nil
}

// プールの閉鎖
func (p *ConnectionPool) Close() {
	p.mu.Lock()
	defer p.mu.Unlock()
	
	if p.closed {
		return
	}
	
	p.closed = true
	
	// すべての接続を閉じる
	for _, conn := range p.connections {
		conn.conn.Close()
	}
	p.connections = nil
	
	// 通知
	select {
	case p.eventCh <- struct{}{}:
	default:
	}
	close(p.eventCh)
}

// プール接続ラッパー
type PoolConnection struct {
	pool *ConnectionPool
	conn *PooledConnection
}

func (pc *PoolConnection) Read(b []byte) (n int, err error) {
	return pc.conn.conn.Read(b)
}

func (pc *PoolConnection) Write(b []byte) (n int, err error) {
	return pc.conn.conn.Write(b)
}

func (pc *PoolConnection) Close() error {
	pc.pool.mu.Lock()
	defer pc.pool.mu.Unlock()
	
	// プールが閉じられている場合は完全にクローズ
	if pc.pool.closed {
		return pc.conn.conn.Close()
	}
	
	// それ以外はプールに戻す
	pc.conn.inUse = false
	pc.conn.lastUsed = time.Now()
	return nil
}

// ========== 再試行ポリシー ==========

type RetryPolicy struct {
	MaxRetries    int
	BaseDelay     time.Duration
	MaxDelay      time.Duration
	Multiplier    float64
	Jitter        float64
	RetryableFunc func(error) bool
}

func DefaultRetryPolicy() *RetryPolicy {
	return &RetryPolicy{
		MaxRetries: 3,
		BaseDelay:  100 * time.Millisecond,
		MaxDelay:   30 * time.Second,
		Multiplier: 2.0,
		Jitter:     0.2,
		RetryableFunc: func(err error) bool {
			if err == nil {
				return false
			}
			
			var tcpErr *TCPError
			if errors.As(err, &tcpErr) {
				return tcpErr.Type == ErrorTemporary || tcpErr.Type == ErrorResource
			}
			
			errType, _ := classifyError(err)
			return errType == ErrorTemporary || errType == ErrorResource
		},
	}
}

// バックオフ時間計算
func calculateBackoff(attempt int, policy *RetryPolicy) time.Duration {
	if attempt <= 0 {
		return 0
	}
	
	// 指数バックオフ
	backoff := float64(policy.BaseDelay)
	for i := 1; i < attempt; i++ {
		backoff *= policy.Multiplier
	}
	
	// 最大値を制限
	if backoff > float64(policy.MaxDelay) {
		backoff = float64(policy.MaxDelay)
	}
	
	// ジッター追加（均一分布）
	if policy.Jitter > 0 {
		backoff = backoff * (1 - policy.Jitter/2 + policy.Jitter*rand.Float64())
	}
	
	return time.Duration(backoff)
}

	for k, v := range metrics {
		t.circuitBreakerMetrics[k] = v
	}

// 平均計算ヘルパー
func calculateAvg(durations []time.Duration) time.Duration {
	if len(durations) == 0 {
		return 0
	}
	
	var sum time.Duration
	for _, d := range durations {
		sum += d
	}
	return sum / time.Duration(len(durations))
}

// 統計データ取得
func (t *Telemetry) GetStats() map[string]interface{} {
	// レイテンシデータコピー
	t.latencyMutex.Lock()
	sendLatencies := make([]time.Duration, len(t.sendLatencies))
	copy(sendLatencies, t.sendLatencies)
	
	receiveLatencies := make([]time.Duration, len(t.receiveLatencies))
	copy(receiveLatencies, t.receiveLatencies)
	
	rttLatencies := make([]time.Duration, len(t.rttLatencies))
	copy(rttLatencies, t.rttLatencies)
	t.latencyMutex.Unlock()
	
	// サーキットブレーカーデータコピー
	t.cbMutex.RLock()
	cbMetrics := make(map[string]interface{})
	for k, v := range t.circuitBreakerMetrics {
		cbMetrics[k] = v
	}
	t.cbMutex.RUnlock()
	
	// 平均計算
	avgSendLatency := calculateAvg(sendLatencies)
	avgReceiveLatency := calculateAvg(receiveLatencies)
	avgRtt := calculateAvg(rttLatencies)
	
	return map[string]interface{}{
		"connections": map[string]int64{
			"attempts":  t.connectionAttempts.Load(),
			"success":   t.connectionSuccess.Load(),
			"failures":  t.connectionFailures.Load(),
			"active":    t.activeConnections.Load(),
		},
		"transfer": map[string]int64{
			"bytes_sent":      t.bytesSent.Load(),
			"bytes_received":  t.bytesReceived.Load(),
			"messages_sent":   t.messagesSent.Load(),
			"messages_received": t.messagesReceived.Load(),
		},
		"errors": map[string]int64{
			"count":   t.errors.Load(),
			"retries": t.retries.Load(),
		},
		"latency": map[string]string{
			"avg_send":    avgSendLatency.String(),
			"avg_receive": avgReceiveLatency.String(),
			"avg_rtt":     avgRtt.String(),
		},
		"circuit_breaker": cbMetrics,
	}
}

// TCP クライアント
type TCPClient struct {
	config           *TCPClientConfig
	connPool         *ConnectionPool
	circuitBreaker   *CircuitBreaker
	telemetry        *Telemetry
	
	sequenceNum      uint32
	heartbeatTicker  *time.Ticker
	closeCh          chan struct{}
	
	// ロギング
	logger           *log.Logger
}

func NewTCPClient(config *TCPClientConfig) *TCPClient {
	if config == nil {
		config = DefaultTCPClientConfig()
	}
	
	client := &TCPClient{
		config:     config,
		telemetry:  NewTelemetry(),
		closeCh:    make(chan struct{}),
		logger:     log.New(os.Stderr, "[TCPClient] ", log.LstdFlags),
	}
	
	// サーキットブレーカー初期化
	if config.EnableCircuitBreaker {
		client.circuitBreaker = NewCircuitBreaker(
			config.Address,
			config.CBThreshold,
			config.CBHalfOpenRequests,
			config.CBResetTimeout,
		)
	}
	
	// 接続プール初期化
	if config.UseConnectionPool {
		client.connPool = NewConnectionPool(
			config.Address,
			config.MaxPoolSize,
			func(ctx context.Context) (net.Conn, error) {
				return client.createConnection(ctx)
			},
		)
	}
	
	return client
}

// 接続作成
func (c *TCPClient) createConnection(ctx context.Context) (net.Conn, error) {
	// テレメトリ
	if c.config.EnableTelemetry {
		c.telemetry.ConnectionAttempt()
	}
	
	if c.config.LogLevel >= 2 {
		c.logger.Printf("接続試行: %s", c.config.Address)
	}
	
	// ダイアル
	dialer := net.Dialer{
		Timeout: c.config.ConnectTimeout,
	}
	
	var conn net.Conn
	var err error
	
	// TLS使用判定
	if c.config.TLSConfig != nil {
		conn, err = tls.DialWithDialer(&dialer, "tcp", c.config.Address, c.config.TLSConfig)
	} else {
		conn, err = dialer.DialContext(ctx, "tcp", c.config.Address)
	}
	
	if err != nil {
		if c.config.EnableTelemetry {
			c.telemetry.ConnectionFailure()
		}
		
		if c.config.LogLevel >= 1 {
			c.logger.Printf("接続エラー: %v", err)
		}
		
		return nil, &TCPError{
			Type:    classifyError(err),
			Code:    ErrCodeConnection,
			Message: "TCP接続エラー",
			Err:     err,
		}
	}
	
	// TCP接続の場合、特別な設定を適用
	if tcpConn, ok := conn.(*net.TCPConn); ok {
		tcpConn.SetKeepAlive(true)
		tcpConn.SetKeepAlivePeriod(c.config.KeepAliveInterval)
		tcpConn.SetNoDelay(true) // Nagleアルゴリズム無効化
	}
	
	// テレメトリ
	if c.config.EnableTelemetry {
		c.telemetry.ConnectionSuccess()
	}
	
	if c.config.LogLevel >= 2 {
		c.logger.Printf("接続成功: %s -> %s", conn.LocalAddr(), conn.RemoteAddr())
	}
	
	return conn, nil
}

// 接続取得
func (c *TCPClient) getConnection(ctx context.Context) (net.Conn, error) {
	// サーキットブレーカーチェック
	if c.circuitBreaker != nil {
		if err := c.circuitBreaker.Allow(); err != nil {
			return nil, err
		}
	}
	
	// 接続プール使用時
	if c.connPool != nil {
		return c.connPool.GetConnection(ctx)
	}
	
	// プールなしの場合は直接接続
	return c.createConnection(ctx)
}

// 新しいシーケンス番号取得
func (c *TCPClient) nextSequence() uint32 {
	return atomic.AddUint32(&c.sequenceNum, 1)
}

// メッセージ送信（低レベルAPI）
func (c *TCPClient) sendMessage(ctx context.Context, conn net.Conn, msg *Message) error {
	// シーケンス番号設定（未設定の場合）
	if msg.SequenceNum == 0 {
		msg.SequenceNum = c.nextSequence()
	}
	
	// プロトコルバージョン設定
	msg.Version = ProtocolVersion
	
	// 書き込みタイムアウト設定
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(c.config.WriteTimeout)
	}
	if err := conn.SetWriteDeadline(deadline); err != nil {
		return &TCPError{
			Type:    ErrorTemporary,
			Code:    ErrCodeTimeout,
			Message: "書き込みタイムアウト設定エラー",
			Err:     err,
		}
	}
	
	// メッセージ送信
	startTime := time.Now()
	err := WriteMessage(conn, msg)
	duration := time.Since(startTime)
	
	// テレメトリ
	if c.config.EnableTelemetry {
		c.telemetry.SendLatency(duration)
		
		if err == nil {
			c.telemetry.MessageSent()
			msgSize := ProtocolHeaderSize + len(msg.Payload)
			c.telemetry.BytesSent(msgSize)
		} else {
			c.telemetry.Error()
		}
	}
	
	// ログ
	if err != nil {
		if c.config.LogLevel >= 1 {
			c.logger.Printf("送信エラー: seq=%d, cmd=%04x: %v", msg.SequenceNum, msg.Command, err)
		}
	} else if c.config.LogLevel >= 3 {
		c.logger.Printf("送信成功: seq=%d, cmd=%04x, size=%d, time=%v", 
			msg.SequenceNum, msg.Command, len(msg.Payload), duration)
	}
	
	return err
}

// メッセージ受信（低レベルAPI）
func (c *TCPClient) receiveMessage(ctx context.Context, conn net.Conn) (*Message, error) {
	// 読み込みタイムアウト設定
	deadline, ok := ctx.Deadline()
	if !ok {
		deadline = time.Now().Add(c.config.ReadTimeout)
	}
	if err := conn.SetReadDeadline(deadline); err != nil {
		return nil, &TCPError{
			Type:    ErrorTemporary,
			Code:    ErrCodeTimeout,
			Message: "読み込みタイムアウト設定エラー",
			Err:     err,
		}
	}
	
	// メッセージ受信
	startTime := time.Now()
	msg, err := ReadMessage(conn)
	duration := time.Since(startTime)
	
	// テレメトリ
	if c.config.EnableTelemetry {
		c.telemetry.ReceiveLatency(duration)
		
		if err == nil {
			c.telemetry.MessageReceived()
			msgSize := ProtocolHeaderSize + len(msg.Payload)
			c.telemetry.BytesReceived(msgSize)
		} else {
			c.telemetry.Error()
		}
	}
	
	// ログ
	if err != nil {
		if c.config.LogLevel >= 1 {
			c.logger.Printf("受信エラー: %v", err)
		}
	} else if c.config.LogLevel >= 3 {
		c.logger.Printf("受信成功: seq=%d, cmd=%04x, size=%d, time=%v", 
			msg.SequenceNum, msg.Command, len(msg.Payload), duration)
	}
	
	return msg, err
}

// データ送信（高レベルAPI）
func (c *TCPClient) SendData(ctx context.Context, data []byte, flags byte) error {
	// 再試行付き実行
	var lastErr error
	policy := c.config.RetryPolicy
	
	for attempt := 0; attempt <= policy.MaxRetries; attempt++ {
		if attempt > 0 {
			if c.config.EnableTelemetry {
				c.telemetry.Retry()
			}
			
			if c.config.LogLevel >= 2 {
				c.logger.Printf("再試行 %d/%d: %v", attempt, policy.MaxRetries, lastErr)
			}
			
			// バックオフ
			delay := calculateBackoff(attempt, policy)
			timer := time.NewTimer(delay)
			select {
			case <-timer.C:
				// 遅延完了
			case <-ctx.Done():
				timer.Stop()
				return ctx.Err()
			}
		}
		
		// 接続取得
		conn, err := c.getConnection(ctx)
		if err != nil {
			lastErr = err
			
			// サーキットブレーカー更新
			if c.circuitBreaker != nil {
				c.circuitBreaker.Failure()
			}
			
			// 再試行判定
			if !policy.RetryableFunc(err) {
				break
			}
			
			continue
		}
		
		// 接続の解放を保証
		connClosed := false
		defer func() {
			if !connClosed {
				conn.Close()
			}
		}()
		
		// メッセージ作成
		msg := &Message{
			Command: CmdData,
			Flags:   c.config.DefaultFlags | flags,
			Payload: data,
		}
		
		// メッセージ送信
		err = c.sendMessage(ctx, conn, msg)
		if err != nil {
			lastErr = err
			
			// サーキットブレーカー更新
			if c.circuitBreaker != nil {
				c.circuitBreaker.Failure()
			}
			
			conn.Close()
			connClosed = true
			
			// 再試行判定
			if !policy.RetryableFunc(err) {
				break
			}
			
			continue
		}
		
		// サーキットブレーカー更新
		if c.circuitBreaker != nil {
			c.circuitBreaker.Success()
		}
		
		// 接続をプールに返す
		conn.Close()
		connClosed = true
		
		return nil // 成功
	}
	
	// すべての再試行が失敗
	return lastErr
}

// データ送信と応答受信
func (c *TCPClient) SendAndReceive(ctx context.Context, data []byte, flags byte) ([]byte, error) {
	// 再試行付き実行
	var lastErr error
	policy := c.config.RetryPolicy
	
	for attempt := 0; attempt <= policy.MaxRetries; attempt++ {
		if attempt > 0 {
			if c.config.EnableTelemetry {
				c.telemetry.Retry()
			}
			
			if c.config.LogLevel >= 2 {
				c.logger.Printf("再試行 %d/%d: %v", attempt, policy.MaxRetries, lastErr)
			}
			
			// バックオフ
			delay := calculateBackoff(attempt, policy)
			timer := time.NewTimer(delay)
			select {
			case <-timer.C:
				// 遅延完了
			case <-ctx.Done():
				timer.Stop()
				return nil, ctx.Err()
			}
		}
		
		// RTT測定開始
		startTime := time.Now()
		
		// 接続取得
		conn, err := c.getConnection(ctx)
		if err != nil {
			lastErr = err
			
			// サーキットブレーカー更新
			if c.circuitBreaker != nil {
				c.circuitBreaker.Failure()
			}
			
			// 再試行判定
			if !policy.RetryableFunc(err) {
				break
			}
			
			continue
		}
		
		// 接続の解放を保証
		connClosed := false
		defer func() {
			if !connClosed {
				conn.Close()
			}
		}()
		
		// リクエストメッセージ作成
		msg := &Message{
			Command: CmdData,
			Flags:   c.config.DefaultFlags | flags,
			Payload: data,
		}
		
		// メッセージ送信
		err = c.sendMessage(ctx, conn, msg)
		if err != nil {
			lastErr = err
			
			// サーキットブレーカー更新
			if c.circuitBreaker != nil {
				c.circuitBreaker.Failure()
			}
			
			conn.Close()
			connClosed = true
			
			// 再試行判定
			if !policy.RetryableFunc(err) {
				break
			}
			
			continue
		}
		
		// 応答受信
		resp, err := c.receiveMessage(ctx, conn)
		if err != nil {
			lastErr = err
			
			// サーキットブレーカー更新
			if c.circuitBreaker != nil {
				c.circuitBreaker.Failure()
			}
			
			conn.Close()
			connClosed = true
			
			// 再試行判定
			if !policy.RetryableFunc(err) {
				break
			}
			
			continue
		}
		
		// RTT測定終了
		rtt := time.Since(startTime)
		if c.config.EnableTelemetry {
			c.telemetry.RoundTripTime(rtt)
		}
		
		// サーキットブレーカー更新
		if c.circuitBreaker != nil {
			c.circuitBreaker.Success()
		}
		
		// 接続をプールに返す
		conn.Close()
		connClosed = true
		
		// エラーレスポンスチェック
		if resp.Flags&FlagIsError != 0 {
			var errInfo struct {
				Code    int    `json:"code"`
				Message string `json:"message"`
			}
			
			if err := json.Unmarshal(resp.Payload, &errInfo); err == nil {
				return nil, &TCPError{
					Type:    ErrorPermanent,
					Code:    errInfo.Code,
					Message: errInfo.Message,
				}
			} else {
				return nil, &TCPError{
					Type:    ErrorPermanent,
					Code:    ErrCodeProtocol,
					Message: "サーバーエラー",
					Err:     errors.New(string(resp.Payload)),
				}
			}
		}
		
		return resp.Payload, nil // 成功
	}
	
	// すべての再試行が失敗
	return nil, lastErr
}

// ハートビート送信
func (c *TCPClient) SendHeartbeat(ctx context.Context) error {
	// 接続取得
	conn, err := c.getConnection(ctx)
	if err != nil {
		return err
	}
	defer conn.Close()
	
	// ハートビートメッセージ作成
	msg := &Message{
		Command: CmdHeartbeat,
		Flags:   FlagIsKeepAlive,
		Payload: []byte{},
	}
	
	// メッセージ送信
	return c.sendMessage(ctx, conn, msg)
}

// ハートビートスタート
func (c *TCPClient) StartHeartbeat() {
	if c.config.HeartbeatInterval <= 0 {
		return
	}
	
	c.heartbeatTicker = time.NewTicker(c.config.HeartbeatInterval)
	
	go func() {
		for {
			select {
			case <-c.heartbeatTicker.C:
				ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
				err := c.SendHeartbeat(ctx)
				cancel()
				
				if err != nil && c.config.LogLevel >= 2 {
					c.logger.Printf("ハートビートエラー: %v", err)
				}
				
			case <-c.closeCh:
				return
			}
		}
	}()
}

// テレメトリデータ取得
func (c *TCPClient) GetTelemetry() map[string]interface{} {
	if !c.config.EnableTelemetry {
		return map[string]interface{}{"enabled": false}
	}
	
	// サーキットブレーカーメトリクス取得
	if c.circuitBreaker != nil {
		cbMetrics := c.circuitBreaker.Metrics()
		c.telemetry.UpdateCircuitBreakerMetrics(cbMetrics)
	}
	
	return c.telemetry.GetStats()
}

// クライアント終了
func (c *TCPClient) Close() {
	// ハートビート停止
	if c.heartbeatTicker != nil {
		c.heartbeatTicker.Stop()
		close(c.closeCh)
	}
	
	// 接続プール停止
	if c.connPool != nil {
		c.connPool.Close()
	}
	
	if c.config.LogLevel >= 2 {
		c.logger.Println("クライアント終了")
	}
}

func main() {

	sigCh := make(chan os.Signal, 1)
```

## 7. 実務上のポイントとチューニング指針

### ネットワーク条件に応じた設定値の目安

| パラメータ | 高速LANネットワーク | インターネット（安定） | モバイルネットワーク |
|------------|-------------------|---------------------|---------------------|
| 接続タイムアウト | 1〜3秒 | 5〜10秒 | 10〜20秒 |
| 読み取りタイムアウト | 5〜10秒 | 15〜30秒 | 30〜60秒 |
| 書き込みタイムアウト | 2〜5秒 | 5〜15秒 | 15〜30秒 |
| キープアライブ間隔 | 30秒 | 15秒 | 60秒 |
| バッファサイズ | 16〜64KB | 8〜32KB | 4〜16KB |
| 最大再試行回数 | 1〜2 | 3〜5 | 5〜10 |
| 再試行間隔 | 初期: 50ms<br>最大: 1秒 | 初期: 100ms<br>最大: 5秒 | 初期: 500ms<br>最大: 30秒 |
| サーキットブレーカー閾値 | 10 | 5 | 3 |
| サーキットブレーカーリセット | 5秒 | 30秒 | 60秒 |

### トラブルシューティング手法

1. **段階的な診断アプローチ**:
   - 通信レイヤー → プロトコルレイヤー → アプリケーションレイヤーの順に調査
   - 低レベルから高レベルへ、クリーンな診断パスを構築

2. **ログレベルによる問題特定**:
   - エラー時のみ一時的にログレベルを上げる
   - タイミング情報と詳細なエラーコンテキストを記録

3. **テレメトリデータの活用**:
   - 問題発生前の傾向を分析（レイテンシ上昇、エラー率上昇）
   - 障害時の特徴パターンを識別するための「フィンガープリント」作成

### パフォーマンス最適化

1. **コネクションプールのチューニング**:
   - 平均接続数 × 1.5 程度のプールサイズを確保
   - アイドル接続の保持期間を実測RTTの10倍程度に設定
   - 高負荷環境では最小プールサイズを設定して即時応答を可能に

2. **バッファ管理**:
   - 平均メッセージサイズに合わせたバッファ初期サイズ設定
   - 大きなメッセージのための拡張バッファ戦略
   - ゼロコピー技術の活用（可能な場合）

3. **CPU効率**:
   - ロックの最小化（特にホットパス）
   - 適切なゴルーチン数の管理
   - コンテキスト切り替えコストの最小化

### 運用のベストプラクティス

1. **段階的デプロイメント**:
   - 小規模でのテスト導入
   - メトリクス基準でのスケールアップ
   - 問題の早期発見のための部分的ロールアウト

2. **継続的モニタリング**:
   - 重要メトリクス（エラー率、レイテンシ、帯域幅使用量）の監視
   - 閾値ベースのアラート設定
   - 定期的なパフォーマンス評価

3. **定期的な健全性チェック**:
   - 接続プールの使用状況
   - メモリ使用量の監視
   - ゴルーチンリークの検出

### セキュリティ考慮事項

1. **TLS設定の最適化**:
   - セッション再利用の有効化
   - 最小限の安全なプロトコルバージョンと暗号スイートの指定
   - 証明書検証の適切な設定

2. **認証とプロトコルセキュリティ**:
   - 初期ハンドシェイクでの認証
   - センシティブデータの暗号化フラグ管理
   - メッセージ認証コード（MAC）の適用

3. **リソース制約防止**:
   - レート制限の実装
   - 過大なメッセージサイズの拒否
   - リソース枯渇攻撃への対策

これらの指針とサンプルコードを適切に活用することで、高信頼性・高性能なTCP通信システムを構築できます。実際のデプロイメント前には、必ず徹底したテストと性能評価を行い、特定の環境に合わせた調整を行ってください。
