## 6.1 ノンブロッキングI/Oとイベント駆動設計

### 概念

ノンブロッキングI/Oとイベント駆動設計は、高性能なネットワークアプリケーションを構築するための重要な概念です。Go言語では、これらの概念が言語設計に組み込まれています。

**ノンブロッキングI/O**とは、I/O操作がプログラムの実行を停止（ブロック）せずに、別のタスクを実行できるようにする仕組みです。Go言語では、goroutineとchannelを使用してこれを実現します。

**イベント駆動設計**は、システムがイベント（接続要求、データ受信など）に応じて処理を行う設計パターンです。これにより、リソースを効率的に使用し、スケーラブルなアプリケーションを構築できます。

### 実装例

```go
package main

import (
	"fmt"
	"log"
	"net"
	"sync"
	"time"
)

// イベントタイプの定義
type EventType int

const (
	EventConnect EventType = iota
	EventData
	EventClose
)

// イベントオブジェクト
type Event struct {
	Type    EventType
	Conn    net.Conn
	Data    []byte
	ErrChan chan error
}

// イベントハンドラーインターフェース
type EventHandler interface {
	HandleConnect(conn net.Conn) error
	HandleData(conn net.Conn, data []byte) error
	HandleClose(conn net.Conn) error
}

// サーバー構造体
type Server struct {
	listener     net.Listener
	eventQueue   chan Event
	connections  map[net.Conn]bool
	handler      EventHandler
	wg           sync.WaitGroup
	shutdownChan chan struct{}
	mu           sync.RWMutex
}

// デフォルトのイベントハンドラー実装
type DefaultHandler struct{}

func (h *DefaultHandler) HandleConnect(conn net.Conn) error {
	log.Printf("新規接続: %s", conn.RemoteAddr())
	return nil
}

func (h *DefaultHandler) HandleData(conn net.Conn, data []byte) error {
	log.Printf("データ受信: %s から %d バイト", conn.RemoteAddr(), len(data))
	// エコーサーバーとして動作
	_, err := conn.Write(data)
	return err
}

func (h *DefaultHandler) HandleClose(conn net.Conn) error {
	log.Printf("接続クローズ: %s", conn.RemoteAddr())
	return nil
}

// 新しいサーバーを作成
func NewServer(address string, handler EventHandler) (*Server, error) {
	listener, err := net.Listen("tcp", address)
	if err != nil {
		return nil, fmt.Errorf("リスナーの作成に失敗しました: %w", err)
	}

	server := &Server{
		listener:     listener,
		eventQueue:   make(chan Event, 1000), // イベントキューのバッファサイズを設定
		connections:  make(map[net.Conn]bool),
		handler:      handler,
		shutdownChan: make(chan struct{}),
	}

	return server, nil
}

// サーバー起動
func (s *Server) Start() error {
	defer s.listener.Close()

	// イベント処理ワーカーの起動
	for i := 0; i < 5; i++ { // ワーカー数の調整
		s.wg.Add(1)
		go s.eventProcessor()
	}

	// 接続の受け入れ
	go s.acceptConnections()

	// シャットダウン待機
	<-s.shutdownChan
	s.wg.Wait()
	return nil
}

// 接続受け入れループ
func (s *Server) acceptConnections() {
	defer s.wg.Done()
	s.wg.Add(1)

	for {
		select {
		case <-s.shutdownChan:
			return
		default:
			s.listener.(*net.TCPListener).SetDeadline(time.Now().Add(500 * time.Millisecond))
			conn, err := s.listener.Accept()
			if err != nil {
				if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
					continue
				}
				log.Printf("接続受け入れエラー: %v", err)
				continue
			}

			// 新しい接続をマップに追加
			s.mu.Lock()
			s.connections[conn] = true
			s.mu.Unlock()

			// 接続イベントをキューに入れる
			s.eventQueue <- Event{
				Type:    EventConnect,
				Conn:    conn,
				ErrChan: make(chan error, 1),
			}

			// 接続ごとに読み取りループを開始
			s.wg.Add(1)
			go s.readLoop(conn)
		}
	}
}

// クライアント接続からの読み取りループ
func (s *Server) readLoop(conn net.Conn) {
	defer func() {
		s.mu.Lock()
		delete(s.connections, conn)
		s.mu.Unlock()
		conn.Close()
		s.wg.Done()
	}()

	buffer := make([]byte, 4096)
	for {
		conn.SetReadDeadline(time.Now().Add(500 * time.Millisecond))
		n, err := conn.Read(buffer)
		if err != nil {
			if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
				select {
				case <-s.shutdownChan:
					// クローズイベントをキューに入れる
					s.eventQueue <- Event{
						Type:    EventClose,
						Conn:    conn,
						ErrChan: make(chan error, 1),
					}
					return
				default:
					continue
				}
			}

			// クローズイベントをキューに入れる
			s.eventQueue <- Event{
				Type:    EventClose,
				Conn:    conn,
				ErrChan: make(chan error, 1),
			}
			return
		}

		// データイベントをキューに入れる
		data := make([]byte, n)
		copy(data, buffer[:n])
		s.eventQueue <- Event{
			Type:    EventData,
			Conn:    conn,
			Data:    data,
			ErrChan: make(chan error, 1),
		}
	}
}

// イベント処理ワーカー
func (s *Server) eventProcessor() {
	defer s.wg.Done()

	for {
		select {
		case <-s.shutdownChan:
			return
		case event := <-s.eventQueue:
			var err error
			switch event.Type {
			case EventConnect:
				err = s.handler.HandleConnect(event.Conn)
			case EventData:
				err = s.handler.HandleData(event.Conn, event.Data)
			case EventClose:
				err = s.handler.HandleClose(event.Conn)
			}

			if event.ErrChan != nil {
				event.ErrChan <- err
				close(event.ErrChan)
			}
		}
	}
}

// サーバーのシャットダウン
func (s *Server) Shutdown() {
	close(s.shutdownChan)

	// すべての接続をクローズ
	s.mu.RLock()
	for conn := range s.connections {
		conn.Close()
	}
	s.mu.RUnlock()
}

func main() {
	handler := &DefaultHandler{}
	server, err := NewServer(":8080", handler)
	if err != nil {
		log.Fatalf("サーバー作成エラー: %v", err)
	}

	log.Println("サーバー起動中...")
	if err := server.Start(); err != nil {
		log.Fatalf("サーバー実行エラー: %v", err)
	}
}
```

### ベストプラクティス

1. **イベントキューのバッファサイズを適切に設定する**: システムの要件に合わせて調整します。
2. **ワーカー数の調整**: CPUコア数やワークロードに応じて調整します。
3. **デッドラインとタイムアウト**: すべてのI/O操作にタイムアウトを設定します。
4. **エラー処理**: すべてのエラーを適切に処理し、リカバリー戦略を実装します。
5. **メモリ使用量の最適化**: バッファサイズを適切に設定し、不要なメモリ割り当てを避けます。

## 6.2 並行処理とchannelを活用したネットワークアプリケーション

### 概念

Go言語の強みの一つは、軽量なgoroutineとchannelを使用した並行処理モデルです。ネットワークアプリケーションでは、この並行処理モデルを活用して、複数のクライアント接続を効率的に処理できます。

**Goroutine**は、Go言語の軽量スレッドで、数千から数百万のgoroutineを同時に実行できます。これにより、各クライアント接続に対して個別のgoroutineを割り当てることができます。

**Channel**は、goroutine間でデータを安全に共有するための通信メカニズムです。ネットワークアプリケーションでは、イベントの伝播、ワーカープール間の作業分配、同期などに使用されます。

### 実装例: 並行TCPチャットサーバー

```go
package main

import (
	"bufio"
	"context"
	"fmt"
	"log"
	"net"
	"os"
	"os/signal"
	"sync"
	"syscall"
	"time"
)

// クライアント構造体
type Client struct {
	conn     net.Conn
	nickname string
	outgoing chan []byte
	quitting chan struct{}
}

// チャットサーバー構造体
type ChatServer struct {
	clients    map[*Client]bool
	register   chan *Client
	unregister chan *Client
	broadcast  chan []byte
	mu         sync.RWMutex
	wg         sync.WaitGroup
}

// 新しいチャットサーバーを作成
func NewChatServer() *ChatServer {
	return &ChatServer{
		clients:    make(map[*Client]bool),
		register:   make(chan *Client),
		unregister: make(chan *Client),
		broadcast:  make(chan []byte, 256),
	}
}

// クライアントを登録
func (s *ChatServer) registerClient(client *Client) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.clients[client] = true
}

// クライアントの登録を解除
func (s *ChatServer) unregisterClient(client *Client) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, ok := s.clients[client]; ok {
		delete(s.clients, client)
		close(client.outgoing)
	}
}

// すべてのクライアントにメッセージをブロードキャスト
func (s *ChatServer) broadcastMessage(message []byte) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	for client := range s.clients {
		select {
		case client.outgoing <- message:
		default:
			// クライアントの送信バッファがいっぱいの場合、クライアントを切断
			s.unregister <- client
		}
	}
}

// サーバーの実行
func (s *ChatServer) Run(ctx context.Context) {
	for {
		select {
		case <-ctx.Done():
			return
		case client := <-s.register:
			s.registerClient(client)
			message := []byte(fmt.Sprintf("システム: %s がチャットに参加しました\n", client.nickname))
			s.broadcastMessage(message)
		case client := <-s.unregister:
			s.unregisterClient(client)
			message := []byte(fmt.Sprintf("システム: %s がチャットを退出しました\n", client.nickname))
			s.broadcastMessage(message)
		case message := <-s.broadcast:
			s.broadcastMessage(message)
		}
	}
}

// 新しいクライアントを処理
func (s *ChatServer) handleClient(conn net.Conn) {
	defer func() {
		if r := recover(); r != nil {
			log.Printf("クライアント処理中にパニック発生: %v", r)
		}
		s.wg.Done()
	}()

	// ニックネームの要求
	conn.SetDeadline(time.Now().Add(30 * time.Second))
	conn.Write([]byte("ニックネームを入力してください: "))
	scanner := bufio.NewScanner(conn)
	if !scanner.Scan() {
		conn.Close()
		return
	}
	nickname := scanner.Text()
	conn.SetDeadline(time.Time{}) // タイムアウトをリセット

	client := &Client{
		conn:     conn,
		nickname: nickname,
		outgoing: make(chan []byte, 100),
		quitting: make(chan struct{}),
	}

	// クライアントを登録
	s.register <- client

	// 読み込みgoroutine
	s.wg.Add(1)
	go func() {
		defer func() {
			close(client.quitting)
			s.wg.Done()
		}()

		scanner := bufio.NewScanner(conn)
		for scanner.Scan() {
			text := scanner.Text()
			if text == "/quit" {
				return
			}
			message := []byte(fmt.Sprintf("%s: %s\n", client.nickname, text))
			s.broadcast <- message
		}
	}()

	// 書き込みgoroutine
	for {
		select {
		case <-client.quitting:
			s.unregister <- client
			conn.Close()
			return
		case message, ok := <-client.outgoing:
			if !ok {
				conn.Close()
				return
			}
			conn.SetWriteDeadline(time.Now().Add(5 * time.Second))
			_, err := conn.Write(message)
			if err != nil {
				return
			}
		}
	}
}

// リスナーの起動
func (s *ChatServer) Serve(ctx context.Context, address string) error {
	listener, err := net.Listen("tcp", address)
	if err != nil {
		return fmt.Errorf("リスナーの作成に失敗しました: %w", err)
	}
	defer listener.Close()

	log.Printf("チャットサーバーを %s で起動しました", address)

	// サーバーgoroutineを開始
	serverCtx, serverCancel := context.WithCancel(ctx)
	defer serverCancel()
	go s.Run(serverCtx)

	// シグナル処理
	sigChan := make(chan os.Signal, 1)
	signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
	go func() {
		<-sigChan
		log.Println("シャットダウンシグナルを受信しました...")
		serverCancel()
	}()

	// 接続を受け付ける
	for {
		select {
		case <-ctx.Done():
			log.Println("サーバーをシャットダウンしています...")
			s.wg.Wait()
			return nil
		default:
			listener.(*net.TCPListener).SetDeadline(time.Now().Add(500 * time.Millisecond))
			conn, err := listener.Accept()
			if err != nil {
				if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
					continue
				}
				log.Printf("接続受け入れエラー: %v", err)
				continue
			}

			s.wg.Add(1)
			go s.handleClient(conn)
		}
	}
}

func main() {
	server := NewChatServer()
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()

	err := server.Serve(ctx, ":8080")
	if err != nil {
		log.Fatalf("サーバーエラー: %v", err)
	}
}
```

### ベストプラクティス

1. **適切なgoroutineの使用**: 各クライアント接続に対して、読み込みと書き込み用の独立したgoroutineを使用します。
2. **channel設計**: バッファサイズを適切に選び、channelのブロッキングを回避します。
3. **同期処理**: `sync.WaitGroup`を使用して、すべてのgoroutineが終了するのを待ちます。
4. **コンテキスト使用**: `context.Context`を使用して、goroutineのキャンセルとタイムアウトを管理します。
5. **デッドラインとタイムアウト**: すべてのI/O操作にタイムアウトを設定します。
6. **エラー処理**: すべてのエラーを適切に処理し、パニックを回復します。

## 6.3 コンテキストを使用したキャンセル処理とタイムアウト

### 概念

`context`パッケージは、Go 1.7で標準ライブラリに追加され、APIの境界を越えてキャンセル処理とタイムアウトを制御するための方法を提供します。ネットワークプログラミングでは、リクエストのキャンセル、タイムアウト、値の伝播に特に有用です。

主な特徴：
- **キャンセル伝播**: 親コンテキストがキャンセルされると、すべての子コンテキストも自動的にキャンセルされます。
- **タイムアウト制御**: 指定した期間後に自動的にキャンセルされるコンテキストを作成できます。
- **値の伝播**: リクエストスコープの値をAPIの境界を越えて伝播できます。

### 実装例: コンテキスト対応HTTPクライアント

```go
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"io"
	"log"
	"net/http"
	"net/http/httptrace"
	"sync"
	"time"
)

// レスポンス構造体
type Response struct {
	URL      string        `json:"url"`
	Status   int           `json:"status"`
	Size     int           `json:"size"`
	Duration time.Duration `json:"duration"`
	Error    string        `json:"error,omitempty"`
}

// HTTPクライアント構造体
type HTTPClient struct {
	client       *http.Client
	maxRetries   int
	retryBackoff time.Duration
	timeout      time.Duration
}

// 新しいHTTPクライアントを作成
func NewHTTPClient(timeout time.Duration, maxRetries int, retryBackoff time.Duration) *HTTPClient {
	return &HTTPClient{
		client: &http.Client{
			Transport: &http.Transport{
				MaxIdleConns:        100,
				MaxIdleConnsPerHost: 20,
				IdleConnTimeout:     90 * time.Second,
				DisableCompression:  false,
			},
			Timeout: timeout,
		},
		maxRetries:   maxRetries,
		retryBackoff: retryBackoff,
		timeout:      timeout,
	}
}

// 複数のURLに並行リクエストを送信
func (c *HTTPClient) FetchAll(ctx context.Context, urls []string) ([]Response, error) {
	// ctx がキャンセルされた場合に備えて最初にチェック
	select {
	case <-ctx.Done():
		return nil, ctx.Err()
	default:
	}

	responses := make([]Response, len(urls))
	var wg sync.WaitGroup
	wg.Add(len(urls))

	for i, url := range urls {
		go func(i int, url string) {
			defer wg.Done()
			response, err := c.Fetch(ctx, url)
			if err != nil {
				log.Printf("URL %s の取得に失敗しました: %v", url, err)
				responses[i] = Response{
					URL:   url,
					Error: err.Error(),
				}
				return
			}
			responses[i] = response
		}(i, url)
	}

	// タイムアウトを考慮した待機
	done := make(chan struct{})
	go func() {
		wg.Wait()
		close(done)
	}()

	select {
	case <-done:
		return responses, nil
	case <-ctx.Done():
		return nil, ctx.Err()
	}
}

// 単一のURLにリクエストを送信
func (c *HTTPClient) Fetch(ctx context.Context, url string) (Response, error) {
	var response Response
	response.URL = url

	start := time.Now()
	var body []byte
	var status int
	var err error

	// リトライロジック
	for attempt := 0; attempt <= c.maxRetries; attempt++ {
		if attempt > 0 {
			select {
			case <-ctx.Done():
				return Response{}, ctx.Err()
			case <-time.After(c.retryBackoff * time.Duration(attempt)):
				// 指数バックオフで待機
			}
		}

		// トレース情報を追加したコンテキストを作成
		traceCtx := httptrace.WithClientTrace(ctx, &httptrace.ClientTrace{
			GotConn: func(info httptrace.GotConnInfo) {
				log.Printf("接続を確立: %v, 再利用: %v", info.Conn.RemoteAddr(), info.Reused)
			},
			ConnectStart: func(network, addr string) {
				log.Printf("接続開始: %s %s", network, addr)
			},
			ConnectDone: func(network, addr string, err error) {
				log.Printf("接続完了: %s %s %v", network, addr, err)
			},
		})

		body, status, err = c.doRequest(traceCtx, url)
		if err == nil {
			break
		}

		// 一時的なエラーのみリトライ
		if !isTemporaryError(err) {
			return Response{
				URL:      url,
				Status:   status,
				Duration: time.Since(start),
				Error:    err.Error(),
			}, err
		}

		log.Printf("一時的なエラーが発生しました (%s) - リトライ中 %d/%d", err, attempt+1, c.maxRetries)
	}

	if err != nil {
		return Response{
			URL:      url,
			Status:   status,
			Duration: time.Since(start),
			Error:    err.Error(),
		}, err
	}

	return Response{
		URL:      url,
		Status:   status,
		Size:     len(body),
		Duration: time.Since(start),
	}, nil
}

// 実際のHTTPリクエストを実行
func (c *HTTPClient) doRequest(ctx context.Context, url string) ([]byte, int, error) {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
	if err != nil {
		return nil, 0, fmt.Errorf("リクエスト作成エラー: %w", err)
	}

	// ユーザーエージェントの設定
	req.Header.Set("User-Agent", "GoHTTPClient/1.0")

	resp, err := c.client.Do(req)
	if err != nil {
		return nil, 0, fmt.Errorf("リクエスト実行エラー: %w", err)
	}
	defer resp.Body.Close()

	// コンテキストのキャンセルをチェック
	select {
	case <-ctx.Done():
		return nil, resp.StatusCode, ctx.Err()
	default:
	}

	// レスポンスボディを読み込む
	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, resp.StatusCode, fmt.Errorf("レスポンス読み込みエラー: %w", err)
	}

	return body, resp.StatusCode, nil
}

// 一時的なエラーかどうかを判定
func isTemporaryError(err error) bool {
	if netErr, ok := err.(net.Error); ok && netErr.Temporary() {
		return true
	}

	// タイムアウトエラーは一時的と見なす
	if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
		return true
	}

	// その他の一時的なエラー条件をここで追加

	return false
}

func main() {
	client := NewHTTPClient(10*time.Second, 3, 1*time.Second)

	// URLリスト
	urls := []string{
		"https://www.google.com",
		"https://www.github.com",
		"https://www.example.com",
		"https://nonexistentdomainxyz.com", // 存在しないドメイン
	}

	// タイムアウト付きのコンテキストを作成
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()

	log.Println("リクエスト開始...")
	responses, err := client.FetchAll(ctx, urls)
	if err != nil {
		log.Fatalf("リクエストエラー: %v", err)
	}

	// レスポンスをJSON形式で出力
	output, _ := json.MarshalIndent(responses, "", "  ")
	fmt.Println(string(output))
}
```

### ベストプラクティス

1. **コンテキストの伝播**: 関数呼び出し間でコンテキストを適切に伝播させます。
2. **タイムアウト設定**: ネットワーク操作には常に適切なタイムアウトを設定します。
3. **コンテキスト値の適切な使用**: リクエストスコープの値のみをコンテキストに保存します。
4. **キャンセルシグナルの処理**: コンテキストのキャンセルを定期的にチェックし、リソースを適切に解放します。
5. **エラー処理**: コンテキストのエラー（`context.Canceled`または`context.DeadlineExceeded`）を適切に処理します。

## 6.4 コネクションプールとリソース管理

### 概念

コネクションプールは、ネットワーク接続を再利用するための仕組みです。接続のオープンとクローズにはコストがかかるため、接続を再利用することでパフォーマンスを向上させることができます。Go言語の標準ライブラリには、HTTPクライアントやデータベース接続用のプール機能が組み込まれていますが、TCPやその他のプロトコル用のカスタムプールを実装することもできます。

### 実装例: カスタムTCPコネクションプール

```go
package main

import (
	"context"
	"errors"
	"fmt"
	"log"
	"net"
	"sync"
	"time"
)

// コネクション構造体
type Connection struct {
	conn       net.Conn
	lastUsed   time.Time
	inUse      bool
	timeoutSet bool
}

// ダイヤル関数の型定義
type DialFunc func(ctx context.Context) (net.Conn, error)

// コネクションプール構造体
type ConnectionPool struct {
	connections  []*Connection
	dial         DialFunc
	mu           sync.Mutex
	maxIdle      int
	maxOpen      int
	idleTimeout  time.Duration
	connTimeout  time.Duration
	waitTimeout  time.Duration
	waitingConns []chan *Connection
	closed       bool
}

// プールオプション関数の型定義
type PoolOption func(*ConnectionPool)

// 最大アイドル接続数を設定するオプション
func WithMaxIdle(maxIdle int) PoolOption {
	return func(p *ConnectionPool) {
		p.maxIdle = maxIdle
	}
}

// 最大オープン接続数を設定するオプション
func WithMaxOpen(maxOpen int) PoolOption {
	return func(p *ConnectionPool) {
		p.maxOpen = maxOpen
	}
}

// アイドルタイムアウトを設定するオプション
func WithIdleTimeout(timeout time.Duration) PoolOption {
	return func(p *ConnectionPool) {
		p.idleTimeout = timeout
	}
}

// 接続タイムアウトを設定するオプション
func WithConnTimeout(timeout time.Duration) PoolOption {
	return func(p *ConnectionPool) {
		p.connTimeout = timeout
	}
}

// 待機タイムアウトを設定するオプション
func WithWaitTimeout(timeout time.Duration) PoolOption {
	return func(p *ConnectionPool) {
		p.waitTimeout = timeout
	}
}

// 新しいコネクションプールを作成
func NewConnectionPool(dial DialFunc, options ...PoolOption) *ConnectionPool {
	pool := &ConnectionPool{
		dial:        dial,
		maxIdle:     10,
		maxOpen:     100,
		idleTimeout: 5 * time.Minute,
		connTimeout: 30 * time.Second,
		waitTimeout: 1 * time.Minute,
	}

	// オプションを適用
	for _, option := range options {
		option(pool)
	}

	// アイドル接続のクリーンアップを開始
	go pool.cleanupIdleConnections()

	return pool
}

// アイドル接続のクリーンアップ
func (p *ConnectionPool) cleanupIdleConnections() {
	ticker := time.NewTicker(30 * time.Second)
	defer ticker.Stop()

	for range ticker.C {
		p.mu.Lock()
		if p.closed {
			p.mu.Unlock()
			return
		}

		var idleCount int
		currentTime := time.Now()
		var remainConns []*Connection

		for _, conn := range p.connections {
			if conn.inUse {
				remainConns = append(remainConns, conn)
				continue
			}

			// アイドルタイムアウトをチェック
			if currentTime.Sub(conn.lastUsed) > p.idleTimeout {
				conn.conn.Close()
				continue
			}

			idleCount++
			// 最大アイドル接続数を超えている場合は閉じる
			if idleCount > p.maxIdle {
				conn.conn.Close()
				continue
			}

			remainConns = append(remainConns, conn)
		}

		p.connections = remainConns
		p.mu.Unlock()
	}
}

// 接続を取得
func (p *ConnectionPool) Get(ctx context.Context) (net.Conn, error) {
	p.mu.Lock()

	// プールが閉じられていないかチェック
	if p.closed {
		p.mu.Unlock()
		return nil, errors.New("コネクションプールは閉じられています")
	}

	// アイドル接続を探す
	for _, conn := range p.connections {
		if !conn.inUse {
			conn.inUse = true
			p.mu.Unlock()
			return &PoolConn{conn: conn.conn, original: conn, pool: p}, nil
		}
	}

	// 最大オープン接続数をチェック
	if p.maxOpen > 0 && len(p.connections) >= p.maxOpen {
		// 待機リストに追加
		req := make(chan *Connection, 1)
		p.waitingConns = append(p.waitingConns, req)
		p.mu.Unlock()

		// タイムアウトまたはコンテキストのキャンセルを待機
		var timeoutCh <-chan time.Time
		if p.waitTimeout > 0 {
			timer := time.NewTimer(p.waitTimeout)
			defer timer.Stop()
			timeoutCh = timer.C
		}

		select {
		case <-ctx.Done():
			// リクエストをキャンセル
			p.mu.Lock()
			var newWaitingConns []chan *Connection
			for _, c := range p.waitingConns {
				if c != req {
					newWaitingConns = append(newWaitingConns, c)
				}
			}
			p.waitingConns = newWaitingConns
			p.mu.Unlock()
			return nil, ctx.Err()
		case <-timeoutCh:
			// タイムアウト
			p.mu.Lock()
			var newWaitingConns []chan *Connection
			for _, c := range p.waitingConns {
				if c != req {
					newWaitingConns = append(newWaitingConns, c)
				}
			}
			p.waitingConns = newWaitingConns
			p.mu.Unlock()
			return nil, errors.New("接続の取得がタイムアウトしました")
		case conn := <-req:
			// 接続が利用可能になった
			if conn == nil {
				return nil, errors.New("コネクションプールは閉じられています")
			}
			return &PoolConn{conn: conn.conn, original: conn, pool: p}, nil
		}
	}

	// 新しい接続を作成
	p.mu.Unlock()

	// 接続タイムアウトのためのコンテキスト
	connCtx := ctx
	if p.connTimeout > 0 {
		var cancel context.CancelFunc
		connCtx, cancel = context.WithTimeout(ctx, p.connTimeout)
		defer cancel()
	}

	conn, err := p.dial(connCtx)
	if err != nil {
		return nil, fmt.Errorf("接続の作成に失敗しました: %w", err)
	}

	// 新しい接続をプールに追加
	newConn := &Connection{
		conn:     conn,
		lastUsed: time.Now(),
		inUse:    true,
	}

	p.mu.Lock()
	// プールが閉じられていないかもう一度チェック
	if p.closed {
		p.mu.Unlock()
		conn.Close()
		return nil, errors.New("コネクションプールは閉じられています")
	}
	p.connections = append(p.connections, newConn)
	p.mu.Unlock()

	return &PoolConn{conn: conn, original: newConn, pool: p}, nil
}

// プールを閉じる
func (p *ConnectionPool) Close() error {
	p.mu.Lock()
	defer p.mu.Unlock()

	if p.closed {
		return errors.New("コネクションプールは既に閉じられています")
	}

	p.closed = true

	// すべての接続を閉じる
	for _, conn := range p.connections {
		conn.conn.Close()
	}
	p.connections = nil

	// 待機中のリクエストをキャンセル
	for _, req := range p.waitingConns {
		close(req)
	}
	p.waitingConns = nil

	return nil
}

// プールの状態を取得
func (p *ConnectionPool) Stats() (open, idle, waiting int) {
	p.mu.Lock()
	defer p.mu.Unlock()

	open = len(p.connections)
	for _, conn := range p.connections {
		if !conn.inUse {
			idle++
		}
	}
	waiting = len(p.waitingConns)

	return
}

// プール接続ラッパー
type PoolConn struct {
	conn     net.Conn
	original *Connection
	pool     *ConnectionPool
	closed   bool
}

// Read実装
func (c *PoolConn) Read(b []byte) (n int, err error) {
	return c.conn.Read(b)
}

// Write実装
func (c *PoolConn) Write(b []byte) (n int, err error) {
	return c.conn.Write(b)
}

// Close実装（接続をプールに返す）
func (c *PoolConn) Close() error {
	if c.closed {
		return nil
	}
	c.closed = true

	c.pool.mu.Lock()
	defer c.pool.mu.Unlock()

	// プールが閉じられている場合は実際に閉じる
	if c.pool.closed {
		return c.conn.Close()
	}

	c.original.lastUsed = time.Now()
	c.original.inUse = false

	// 待機中のリクエストがあれば、この接続を渡す
	if len(c.pool.waitingConns) > 0 {
		req := c.pool.waitingConns[0]
		c.pool.waitingConns = c.pool.waitingConns[1:]
		c.original.inUse = true
		req <- c.original
	}

	return nil
}

// LocalAddr実装
func (c *PoolConn) LocalAddr() net.Addr {
	return c.conn.LocalAddr()
}

// RemoteAddr実装
func (c *PoolConn) RemoteAddr() net.Addr {
	return c.conn.RemoteAddr()
}

// SetDeadline実装
func (c *PoolConn) SetDeadline(t time.Time) error {
	c.original.timeoutSet = !t.IsZero()
	return c.conn.SetDeadline(t)
}

// SetReadDeadline実装
func (c *PoolConn) SetReadDeadline(t time.Time) error {
	c.original.timeoutSet = !t.IsZero()
	return c.conn.SetReadDeadline(t)
}

// SetWriteDeadline実装
func (c *PoolConn) SetWriteDeadline(t time.Time) error {
	c.original.timeoutSet = !t.IsZero()
	return c.conn.SetWriteDeadline(t)
}

// Timeoutが設定されているかチェック
func (c *PoolConn) HasTimeout() bool {
	return c.original.timeoutSet
}

// 使用例
func main() {
	// TCPサーバーアドレス
	serverAddr := "example.com:80"

	// ダイヤル関数
	dialFunc := func(ctx context.Context) (net.Conn, error) {
		var d net.Dialer
		return d.DialContext(ctx, "tcp", serverAddr)
	}

	// コネクションプールの作成
	pool := NewConnectionPool(
		dialFunc,
		WithMaxIdle(5),
		WithMaxOpen(20),
		WithIdleTimeout(2*time.Minute),
		WithConnTimeout(10*time.Second),
		WithWaitTimeout(30*time.Second),
	)
	defer pool.Close()

	// 接続の取得と使用
	ctx := context.Background()
	conn, err := pool.Get(ctx)
	if err != nil {
		log.Fatalf("接続の取得に失敗しました: %v", err)
	}

	// 接続の使用
	_, err = conn.Write([]byte("GET / HTTP/1.1\r\nHost: example.com\r\n\r\n"))
	if err != nil {
		log.Printf("書き込みエラー: %v", err)
	}

	// プールに接続を戻す
	conn.Close()

	// プールの統計情報を表示
	open, idle, waiting := pool.Stats()
	log.Printf("プール統計: オープン = %d, アイドル = %d, 待機中 = %d", open, idle, waiting)
}
```

### ベストプラクティス

1. **適切なプールサイズの設定**: システム要件とリソース制約に基づいて、`maxIdle`と`maxOpen`を適切に設定します。
2. **タイムアウトの設定**: アイドルタイムアウト、接続タイムアウト、待機タイムアウトを適切に設定します。
3. **リソースのクリーンアップ**: 使用されなくなった接続を定期的にクリーンアップします。
4. **エラー処理**: すべてのエラーを適切に処理し、リカバリー戦略を実装します。
5. **プール統計のモニタリング**: プールの使用状況を監視し、必要に応じて調整します。

## 6.5 バックプレッシャーとフロー制御

### 概念

バックプレッシャーは、システムの一部が処理できる以上のデータを受信したときに、上流のコンポーネントに処理速度を落とすよう通知するメカニズムです。これにより、システム全体の安定性とレスポンス性を維持します。

フロー制御は、データの送信速度を受信側の処理能力に合わせて調整するプロセスです。Go言語では、バッファ付きchannelを使用したバックプレッシャーメカニズムを実装できます。

### 実装例: バックプレッシャーを持つワーカープール

```go
package main

import (
	"context"
	"errors"
	"fmt"
	"log"
	"math/rand"
	"net/http"
	"sync"
	"sync/atomic"
	"time"
)

// ジョブの定義
type Job struct {
	ID      int
	Payload string
}

// 結果の定義
type Result struct {
	JobID   int
	Output  string
	Error   error
	Latency time.Duration
}

// ワーカープールの定義
type WorkerPool struct {
	jobQueue         chan Job
	resultQueue      chan Result
	workerCount      int
	activeWorkers    int32
	maxQueueSize     int
	processingTime   time.Duration
	processedCount   int64
	rejectedCount    int64
	backpressureFunc func(queueFillPercentage float64) bool
	mu               sync.RWMutex
	wg               sync.WaitGroup
	ctx              context.Context
	cancel           context.CancelFunc
}

// 新しいワーカープールのオプション
type WorkerPoolOption func(*WorkerPool)

// ワーカー数を設定するオプション
func WithWorkerCount(count int) WorkerPoolOption {
	return func(wp *WorkerPool) {
		wp.workerCount = count
	}
}

// キューサイズを設定するオプション
func WithQueueSize(size int) WorkerPoolOption {
	return func(wp *WorkerPool) {
		wp.maxQueueSize = size
	}
}

// 処理時間を設定するオプション
func WithProcessingTime(duration time.Duration) WorkerPoolOption {
	return func(wp *WorkerPool) {
		wp.processingTime = duration
	}
}

// バックプレッシャー関数を設定するオプション
func WithBackpressureFunc(fn func(queueFillPercentage float64) bool) WorkerPoolOption {
	return func(wp *WorkerPool) {
		wp.backpressureFunc = fn
	}
}

// 新しいワーカープールを作成
func NewWorkerPool(options ...WorkerPoolOption) *WorkerPool {
	ctx, cancel := context.WithCancel(context.Background())
	
	wp := &WorkerPool{
		workerCount:    5,
		maxQueueSize:   100,
		processingTime: 200 * time.Millisecond,
		ctx:            ctx,
		cancel:         cancel,
		backpressureFunc: func(queueFillPercentage float64) bool {
			// デフォルトでは、キューが80%以上埋まっていたら拒否
			return queueFillPercentage >= 0.8
		},
	}

	// オプションを適用
	for _, option := range options {
		option(wp)
	}

	wp.jobQueue = make(chan Job, wp.maxQueueSize)
	wp.resultQueue = make(chan Result, wp.maxQueueSize)

	return wp
}

// プールを開始
func (wp *WorkerPool) Start() {
	for i := 0; i < wp.workerCount; i++ {
		wp.wg.Add(1)
		go wp.worker(i)
	}

	// 統計情報の定期的な表示
	go func() {
		ticker := time.NewTicker(5 * time.Second)
		defer ticker.Stop()

		for {
			select {
			case <-wp.ctx.Done():
				return
			case <-ticker.C:
				wp.printStats()
			}
		}
	}()
}

// ワーカーゴルーチン
func (wp *WorkerPool) worker(id int) {
	defer wp.wg.Done()
	
	log.Printf("ワーカー %d 開始", id)
	
	for {
		select {
		case <-wp.ctx.Done():
			log.Printf("ワーカー %d 終了", id)
			return
		case job, ok := <-wp.jobQueue:
			if !ok {
				log.Printf("ワーカー %d: ジョブキューが閉じられています", id)
				return
			}
			
			atomic.AddInt32(&wp.activeWorkers, 1)
			result := wp.processJob(job)
			atomic.AddInt32(&wp.activeWorkers, -1)
			
			select {
			case wp.resultQueue <- result:
				// 結果を送信
			case <-wp.ctx.Done():
				log.Printf("ワーカー %d: 結果の送信中にキャンセルされました", id)
				return
			}
		}
	}
}

// ジョブを処理
func (wp *WorkerPool) processJob(job Job) Result {
	start := time.Now()
	
	// ジョブの処理をシミュレート
	processingTime := wp.processingTime
	if rand.Float64() < 0.2 {
		// 20%の確率で処理時間が2倍になる
		processingTime *= 2
	}
	
	// 処理をシミュレート
	timer := time.NewTimer(processingTime)
	select {
	case <-wp.ctx.Done():
		timer.Stop()
		return Result{
			JobID:   job.ID,
			Error:   errors.New("処理中にキャンセルされました"),
			Latency: time.Since(start),
		}
	case <-timer.C:
		// 処理完了
	}
	
	// 処理結果を作成
	output := fmt.Sprintf("Job %d processed with payload: %s", job.ID, job.Payload)
	atomic.AddInt64(&wp.processedCount, 1)
	
	return Result{
		JobID:   job.ID,
		Output:  output,
		Latency: time.Since(start),
	}
}

// ジョブをサブミット
func (wp *WorkerPool) Submit(job Job) (bool, error) {
	// バックプレッシャーチェック
	queueFillPercentage := float64(len(wp.jobQueue)) / float64(wp.maxQueueSize)
	if wp.backpressureFunc(queueFillPercentage) {
		atomic.AddInt64(&wp.rejectedCount, 1)
		return false, fmt.Errorf("バックプレッシャーによりジョブが拒否されました (キュー使用率: %.2f%%)", queueFillPercentage*100)
	}
	
	select {
	case <-wp.ctx.Done():
		return false, errors.New("ワーカープールは停止中です")
	case wp.jobQueue <- job:
		return true, nil
	default:
		// キューがいっぱいの場合
		atomic.AddInt64(&wp.rejectedCount, 1)
		return false, errors.New("ジョブキューがいっぱいです")
	}
}

// 結果を取得
func (wp *WorkerPool) GetResult(ctx context.Context) (Result, error) {
	select {
	case <-ctx.Done():
		return Result{}, ctx.Err()
	case <-wp.ctx.Done():
		return Result{}, errors.New("ワーカープールは停止中です")
	case result, ok := <-wp.resultQueue:
		if !ok {
			return Result{}, errors.New("結果キューが閉じられています")
		}
		return result, nil
	}
}

// プールの統計情報を取得
func (wp *WorkerPool) Stats() (queueSize, queueCapacity, activeWorkers int, processedJobs, rejectedJobs int64) {
	wp.mu.RLock()
	defer wp.mu.RUnlock()
	
	return len(wp.jobQueue), cap(wp.jobQueue), int(atomic.LoadInt32(&wp.activeWorkers)),
		atomic.LoadInt64(&wp.processedCount), atomic.LoadInt64(&wp.rejectedCount)
}

// 統計情報を表示
func (wp *WorkerPool) printStats() {
	queueSize, queueCapacity, activeWorkers, processedJobs, rejectedJobs := wp.Stats()
	queueFillPercentage := float64(queueSize) / float64(queueCapacity) * 100
	
	log.Printf("ワーカープール統計: キュー = %d/%d (%.1f%%), アクティブワーカー = %d/%d, 処理済み = %d, 拒否 = %d",
		queueSize, queueCapacity, queueFillPercentage, activeWorkers, wp.workerCount, processedJobs, rejectedJobs)
}

// プールを停止
func (wp *WorkerPool) Stop() {
	wp.cancel()
	close(wp.jobQueue)
	wp.wg.Wait()
	close(wp.resultQueue)
	log.Println("ワーカープールが停止しました")
}

// HTTPハンドラーを作成
func createHTTPHandler(pool *WorkerPool) http.HandlerFunc {
	var jobID int64
	
	return func(w http.ResponseWriter, r *http.Request) {
		// POSTメソッドのみ許可
		if r.Method != http.MethodPost {
			http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
			return
		}
		
		// 新しいジョブを作成
		id := int(atomic.AddInt64(&jobID, 1))
		job := Job{
			ID:      id,
			Payload: fmt.Sprintf("Request from %s at %s", r.RemoteAddr, time.Now().Format(time.RFC3339)),
		}
		
		// ジョブをサブミット
		accepted, err := pool.Submit(job)
		if err != nil {
			if !accepted {
				// バックプレッシャーまたはキューいっぱいのエラー
				w.Header().Set("Retry-After", "5") // 5秒後に再試行を推奨
				http.Error(w, err.Error(), http.StatusServiceUnavailable)
				return
			}
			http.Error(w, err.Error(), http.StatusInternalServerError)
			return
		}
		
		// 非同期処理のため、即座に成功レスポンスを返す
		w.WriteHeader(http.StatusAccepted)
		fmt.Fprintf(w, "Job %d submitted successfully", id)
	}
}

// 結果表示のHTTPハンドラー
func createResultsHandler(pool *WorkerPool) http.HandlerFunc {

		result, err := pool.GetResult(ctx)
```

### ベストプラクティス

1. **動的バックプレッシャー**: キューの状態に応じて拒否率を動的に調整します。
2. **適切なヘッダーの設定**: 拒否された場合は`Retry-After`ヘッダーを設定します。
3. **モニタリングと指標**: バックプレッシャーの動作を監視し、必要に応じて調整します。
4. **グレースフルデグラデーション**: システムの負荷が高い場合でも、一部の機能を維持します。
5. **リソース管理**: メモリとCPU使用量を監視し、過負荷を防ぎます。

## 6.6 リトライとサーキットブレーカーパターン

### 概念

**リトライパターン**は、一時的な障害が発生した場合に、操作を再試行する仕組みです。再試行間隔を指数関数的に増やす「指数バックオフ」と組み合わせることで、システムの負荷を軽減します。

**サーキットブレーカーパターン**は、連続した障害が発生した場合に、一定期間リクエストを拒否することでシステムを保護するデザインパターンです。回路遮断機（サーキットブレーカー）には、「閉」（通常操作）、「開」（リクエスト拒否）、「半開」（限定的に許可）の3つの状態があります。

### 実装例: リトライとサーキットブレーカー

```go
package main

import (
	"context"
	"errors"
	"fmt"
	"log"
	"math/rand"
	"net/http"
	"sync"
	"sync/atomic"
	"time"
)

// サーキットブレーカーの状態
type State int

const (
	StateClosed State = iota // 閉（通常）
	StateOpen                // 開（リクエスト拒否）
	StateHalfOpen           // 半開（限定的に許可）
)

// StateToString は状態を文字列に変換
func (s State) String() string {
	switch s {
	case StateClosed:
		return "閉（通常）"
	case StateOpen:
		return "開（リクエスト拒否）"
	case StateHalfOpen:
		return "半開（限定的に許可）"
	default:
		return "不明"
	}
}

// サーキットブレーカー設定
type CircuitBreakerConfig struct {
	FailureThreshold         int           // 開状態に移行するための連続失敗数のしきい値
	SuccessThreshold         int           // 半開状態から閉状態に戻るための連続成功数のしきい値
	ResetTimeout             time.Duration // 開状態から半開状態に移行するまでの時間
	MaxRequestsInHalfOpen    int           // 半開状態で許可するリクエスト数
	FailureRateThreshold     float64       // 失敗率のしきい値（0.0 - 1.0）
	MinimumRequestsThreshold int           // 判断するために必要な最小リクエスト数
	TrackingPeriod           time.Duration // 成功/失敗を追跡する期間
}

// リトライ設定
type RetryConfig struct {
	MaxRetries     int           // 最大再試行回数
	InitialBackoff time.Duration // 初期バックオフ時間
	MaxBackoff     time.Duration // 最大バックオフ時間
	BackoffFactor  float64       // バックオフ係数
	Jitter         float64       // ジッター（0.0 - 1.0）
}

// サーキットブレーカー
type CircuitBreaker struct {
	name     string
	config   CircuitBreakerConfig
	state    State
	failures int32
	successes int32
	
	// ローリングウィンドウ用のカウンター
	totalRequests int32
	totalFailures int32
	
	// リクエスト履歴
	requestHistory []*requestResult
	historyMutex   sync.RWMutex
	
	lastStateChange time.Time
	mutex           sync.RWMutex
	
	// 半開状態でのリクエストカウンター
	halfOpenRequests int32
}

// リクエスト結果
type requestResult struct {
	timestamp time.Time
	success   bool
}

// サーキットブレーカーの作成
func NewCircuitBreaker(name string, config CircuitBreakerConfig) *CircuitBreaker {
	return &CircuitBreaker{
		name:            name,
		config:          config,
		state:           StateClosed,
		lastStateChange: time.Now(),
		requestHistory:  make([]*requestResult, 0, 100),
	}
}

// 現在の状態を取得
func (cb *CircuitBreaker) State() State {
	cb.mutex.RLock()
	defer cb.mutex.RUnlock()
	return cb.state
}

// 状態の変更
func (cb *CircuitBreaker) changeState(newState State) {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()
	
	prevState := cb.state
	if prevState != newState {
		log.Printf("[%s] 状態変更: %s -> %s", cb.name, prevState, newState)
		cb.state = newState
		cb.lastStateChange = time.Now()
		
		// 状態変更時にカウンターをリセット
		if newState == StateHalfOpen {
			atomic.StoreInt32(&cb.halfOpenRequests, 0)
			atomic.StoreInt32(&cb.successes, 0)
		}
	}
}

// リクエスト可能かどうかを確認
func (cb *CircuitBreaker) AllowRequest() bool {
	cb.mutex.RLock()
	defer cb.mutex.RUnlock()
	
	now := time.Now()
	
	switch cb.state {
	case StateClosed:
		return true
	case StateOpen:
		// リセットタイムアウトを経過していれば半開状態に変更
		if now.Sub(cb.lastStateChange) > cb.config.ResetTimeout {
			cb.mutex.RUnlock()
			cb.changeState(StateHalfOpen)
			cb.mutex.RLock()
			return true
		}
		return false
	case StateHalfOpen:
		// 半開状態では限定的にリクエストを許可
		return atomic.LoadInt32(&cb.halfOpenRequests) < int32(cb.config.MaxRequestsInHalfOpen)
	default:
		return false
	}
}

// リクエスト成功を記録
func (cb *CircuitBreaker) RecordSuccess() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()
	
	// リクエスト履歴に成功を追加
	cb.historyMutex.Lock()
	cb.requestHistory = append(cb.requestHistory, &requestResult{
		timestamp: time.Now(),
		success:   true,
	})
	// 古い履歴をクリーンアップ
	cb.cleanupHistory()
	cb.historyMutex.Unlock()
	
	atomic.AddInt32(&cb.totalRequests, 1)
	
	switch cb.state {
	case StateHalfOpen:
		atomic.AddInt32(&cb.halfOpenRequests, 1)
		successes := atomic.AddInt32(&cb.successes, 1)
		
		// 連続成功が閾値に達したら閉状態に戻す
		if int(successes) >= cb.config.SuccessThreshold {
			cb.state = StateClosed
			cb.failures = 0
			cb.successes = 0
			cb.lastStateChange = time.Now()
			log.Printf("[%s] 半開状態から閉状態に戻りました", cb.name)
		}
	case StateClosed:
		// 閉状態では失敗カウンターをリセット
		atomic.StoreInt32(&cb.failures, 0)
	}
}

// リクエスト失敗を記録
func (cb *CircuitBreaker) RecordFailure() {
	cb.mutex.Lock()
	defer cb.mutex.Unlock()
	
	// リクエスト履歴に失敗を追加
	cb.historyMutex.Lock()
	cb.requestHistory = append(cb.requestHistory, &requestResult{
		timestamp: time.Now(),
		success:   false,
	})
	// 古い履歴をクリーンアップ
	cb.cleanupHistory()
	cb.historyMutex.Unlock()
	
	atomic.AddInt32(&cb.totalRequests, 1)
	atomic.AddInt32(&cb.totalFailures, 1)
	
	switch cb.state {
	case StateClosed:
		failures := atomic.AddInt32(&cb.failures, 1)
		
		// 失敗率をチェック
		failureRate := cb.getFailureRate()
		totalRequests := atomic.LoadInt32(&cb.totalRequests)
		
		// 閾値を超える連続失敗または失敗率が高い場合は開状態に変更
		if int(failures) >= cb.config.FailureThreshold || 
			(failureRate >= cb.config.FailureRateThreshold && 
			totalRequests >= int32(cb.config.MinimumRequestsThreshold)) {
			cb.state = StateOpen
			cb.lastStateChange = time.Now()
			log.Printf("[%s] 閉状態から開状態に変更しました (failures: %d, failureRate: %.2f%%)",
				cb.name, failures, failureRate*100)
		}
	case StateHalfOpen:
		atomic.AddInt32(&cb.halfOpenRequests, 1)
		// 半開状態での失敗は即座に開状態に戻す
		cb.state = StateOpen
		cb.lastStateChange = time.Now()
		log.Printf("[%s] 半開状態から開状態に戻りました", cb.name)
	}
}

// 古い履歴をクリーンアップ
func (cb *CircuitBreaker) cleanupHistory() {
	now := time.Now()
	cutoffTime := now.Add(-cb.config.TrackingPeriod)
	
	// 追跡期間より古い履歴を削除
	var newHistory []*requestResult
	for _, result := range cb.requestHistory {
		if result.timestamp.After(cutoffTime) {
			newHistory = append(newHistory, result)
		}
	}
	cb.requestHistory = newHistory
	
	// 統計情報を更新
	var failures int32
	for _, result := range newHistory {
		if !result.success {
			failures++
		}
	}
	
	totalRequests := int32(len(newHistory))
	atomic.StoreInt32(&cb.totalRequests, totalRequests)
	atomic.StoreInt32(&cb.totalFailures, failures)
}

// 直近の失敗率を計算
func (cb *CircuitBreaker) getFailureRate() float64 {
	cb.historyMutex.RLock()
	defer cb.historyMutex.RUnlock()
	
	totalRequests := len(cb.requestHistory)
	if totalRequests == 0 {
		return 0.0
	}
	
	var failures int
	for _, result := range cb.requestHistory {
		if !result.success {
			failures++
		}
	}
	
	return float64(failures) / float64(totalRequests)
}

// リトライ処理を実行
func Retry(ctx context.Context, config RetryConfig, operationName string, operation func() error) error {
	var err error
	var backoff = config.InitialBackoff
	
	for i := 0; i <= config.MaxRetries; i++ {
		// 最初の試行か再試行かをログに出力
		if i == 0 {
			log.Printf("[%s] 実行中...", operationName)
		} else {
			log.Printf("[%s] 再試行中 (%d/%d)...", operationName, i, config.MaxRetries)
		}
		
		// 操作を実行
		err = operation()
		if err == nil {
			// 成功したら終了
			return nil
		}
		
		// 一時的なエラーかどうかをチェック
		if !isRetryableError(err) {
			log.Printf("[%s] 再試行不可能なエラー: %v", operationName, err)
			return err
		}
		
		// 最後の試行だったら終了
		if i == config.MaxRetries {
			break
		}
		
		// バックオフ時間を計算（ジッターを含む）
		jitterBackoff := calculateBackoffWithJitter(backoff, config.Jitter)
		log.Printf("[%s] バックオフ: %v", operationName, jitterBackoff)
		
		// コンテキストのキャンセルかタイムアウトをチェック
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-time.After(jitterBackoff):
			// バックオフ時間の経過後に続行
		}
		
		// 次のバックオフ時間を計算（指数関数的に増加）
		backoff = time.Duration(float64(backoff) * config.BackoffFactor)
		if backoff > config.MaxBackoff {
			backoff = config.MaxBackoff
		}
	}
	
	return fmt.Errorf("最大再試行回数 (%d) に達しました: %w", config.MaxRetries, err)
}

// ジッターを含むバックオフ時間を計算
func calculateBackoffWithJitter(backoff time.Duration, jitter float64) time.Duration {
	if jitter <= 0 {
		return backoff
	}
	
	// バックオフ時間の±jitter%の範囲でランダムに変動
	jitterAmount := float64(backoff) * jitter
	random := rand.Float64()*jitterAmount*2 - jitterAmount
	jitteredBackoff := float64(backoff) + random
	
	if jitteredBackoff < 0 {
		return 0
	}
	
	return time.Duration(jitteredBackoff)
}

// 再試行可能なエラーかどうかをチェック
func isRetryableError(err error) bool {
	// タイムアウトエラー
	if errors.Is(err, context.DeadlineExceeded) {
		return true
	}
	
	// 一時的なネットワークエラー
	var netErr net.Error
	if errors.As(err, &netErr) {
		return netErr.Temporary() || netErr.Timeout()
	}
	
	// HTTP関連の再試行可能なエラー
	if errors.Is(err, io.EOF) || errors.Is(err, io.ErrUnexpectedEOF) {
		return true
	}
	
	// 特定のHTTPステータスコード
	if httpErr, ok := err.(*httpError); ok {
		statusCode := httpErr.StatusCode
		return statusCode >= 500 || statusCode == 429
	}
	
	// その他のケースは再試行不可能とする
	return false
}

// HTTPエラー型
type httpError struct {
	StatusCode int
	Message    string
}

func (e *httpError) Error() string {
	return fmt.Sprintf("HTTP エラー %d: %s", e.StatusCode, e.Message)
}

// サービス呼び出し関数
type ServiceCall func(ctx context.Context) (interface{}, error)

// サーキットブレーカーとリトライを組み合わせた実行
func ExecuteWithCircuitBreaker(
	ctx context.Context,
	cb *CircuitBreaker,
	retryConfig RetryConfig,
	operationName string,
	serviceCall ServiceCall,
) (interface{}, error) {
	// リクエストが許可されるかチェック
	if !cb.AllowRequest() {
		return nil, fmt.Errorf("サーキットブレーカー '%s' が開いています: リクエストは拒否されました", cb.name)
	}
	
	var result interface{}
	var err error
	
	// リトライロジックと組み合わせて実行
	err = Retry(ctx, retryConfig, operationName, func() error {
		var callErr error
		result, callErr = serviceCall(ctx)
		return callErr
	})
	
	// 結果に基づいて成功または失敗を記録
	if err != nil {
		cb.RecordFailure()
		return nil, err
	}
	
	cb.RecordSuccess()
	return result, nil
}

// HTTPクライアントラッパー
type HTTPClient struct {
	client *http.Client
	cb     *CircuitBreaker
	retry  RetryConfig
}

// 新しいHTTPクライアントを作成
func NewHTTPClient(cbConfig CircuitBreakerConfig, retryConfig RetryConfig) *HTTPClient {
	return &HTTPClient{
		client: &http.Client{
			Timeout: 10 * time.Second,
			Transport: &http.Transport{
				MaxIdleConns:        100,
				MaxIdleConnsPerHost: 20,
				IdleConnTimeout:     90 * time.Second,
			},
		},
		cb:    NewCircuitBreaker("HTTPClient", cbConfig),
		retry: retryConfig,
	}
}

// HTTPリクエストを実行
func (c *HTTPClient) Do(ctx context.Context, req *http.Request) (*http.Response, error) {
	result, err := ExecuteWithCircuitBreaker(
		ctx,
		c.cb,
		c.retry,
		fmt.Sprintf("HTTPRequest_%s", req.URL.String()),
		func(ctx context.Context) (interface{}, error) {
			// コンテキストをリクエストに適用
			reqWithCtx := req.WithContext(ctx)
			resp, err := c.client.Do(reqWithCtx)
			if err != nil {
				return nil, err
			}
			
			// エラーステータスコードをエラーに変換
			if resp.StatusCode >= 400 {
				defer resp.Body.Close()
				bodyBytes, _ := io.ReadAll(resp.Body)
				bodyStr := string(bodyBytes)
				return nil, &httpError{
					StatusCode: resp.StatusCode,
					Message:    bodyStr,
				}
			}
			
			return resp, nil
		},
	)
	
	if err != nil {
		return nil, err
	}
	
	return result.(*http.Response), nil
}

func main() {
	// サーキットブレーカー設定
	cbConfig := CircuitBreakerConfig{
		FailureThreshold:         5,        // 5回連続失敗で開状態に移行
		SuccessThreshold:         2,        // 2回連続成功で閉状態に戻る
		ResetTimeout:             10 * time.Second, // 開状態から10秒後に半開状態に移行
		MaxRequestsInHalfOpen:    3,        // 半開状態で最大3リクエストを許可
		FailureRateThreshold:     0.5,      // 50%以上の失敗率で開状態に移行
		MinimumRequestsThreshold: 10,       // 最小10リクエストから判断
		TrackingPeriod:           1 * time.Minute, // 1分間の履歴を追跡
	}
	
	// リトライ設定
	retryConfig := RetryConfig{
		MaxRetries:     3,                // 最大3回再試行
		InitialBackoff: 100 * time.Millisecond, // 初期バックオフ100ms
		MaxBackoff:     2 * time.Second,  // 最大バックオフ2秒
		BackoffFactor:  2.0,              // 指数バックオフ（2倍ずつ増加）
		Jitter:         0.2,              // 20%のジッター
	}
	
	// HTTPクライアントを作成
	client := NewHTTPClient(cbConfig, retryConfig)
	
	// サンプルリクエスト
	req, _ := http.NewRequest("GET", "https://httpbin.org/status/200,500", nil)
	ctx := context.Background()
	
	// リクエストを実行
	for i := 0; i < 20; i++ {
		log.Printf("リクエスト %d/20", i+1)
		resp, err := client.Do(ctx, req)
		if err != nil {
			log.Printf("リクエストエラー: %v", err)
			time.Sleep(500 * time.Millisecond)
			continue
		}
		
		// レスポンスボディを読み取り
		body, _ := io.ReadAll(resp.Body)
		resp.Body.Close()
		log.Printf("レスポンス: %d - %s", resp.StatusCode, string(body))
		time.Sleep(500 * time.Millisecond)
	}
	
	// サーキットブレーカーの状態を確認
	log.Printf("最終的なサーキットブレーカーの状態: %s", client.cb.State())
}
```
